    30→    30→/**
    31→    31→ * Task workflow actions (spec §3: POST /tasks/{id}/assign|status|progress|
    32→    32→ * stages|inspect|comment|block|unblock|update).
    33→    33→ *
    34→    34→ * Every action is a 1:1 port of the Next.js reference implementation:
    35→    35→ * - PATCH src/app/api/atelier/tasks/[id]/route.ts — assign, update, cancel,
    36→    36→ *   block, unblock, submit, inspect, ready, deliver (status-transition
    37→    37→ *   gates, notifications, audit codes ported verbatim);
    38→    38→ * - POST src/app/api/atelier/tasks/[id]/progress/route.ts — progress;
    39→    39→ * - POST src/app/api/atelier/tasks/[id]/stages/route.ts — checklist;
    40→    40→ * - POST src/app/api/atelier/tasks/[id]/comments/route.ts — notes/issues.
    41→    41→ * The `status` endpoint additionally offers "start" and "reopen" verbs per
    42→    42→ * the controller/view contract.
    43→    43→ *
    44→    44→ * Access rules: a task may only be touched by its assignee or a supervisor
    45→    45→ * (MANAGEMENT / ADMIN); STAFF users reaching another user's task get 403.
    46→    46→ * Terminal tasks (DELIVERED / CANCELLED) reject every action, like the
    47→    47→ * outer gate of the Next.js PATCH handler.
    48→    48→ *
    49→    49→ * Deliberately does not extend a base Controller class, mirroring
    50→    50→ * LoginController / LocaleController.
    51→    51→ */
    52→    52→class TaskActionController
    53→    53→{
    54→    54→    /**
    55→    55→     * Statuses in which the checklist / progress can be updated by the
    56→    56→     * assignee (supervisors may always correct, mirroring route.ts).
    57→    57→     */
    58→    58→    private const IN_WORK_STATUSES = ['ASSIGNED', 'ONGOING', 'CHANGES_REQUESTED'];
    59→    59→
    60→    60→    /**
    61→    61→     * POST /tasks/{id}/assign — assign or reassign the task (route perm
    62→    62→     * production.assign: MANAGEMENT / ADMIN only).
    63→    63→     *
    64→    64→     * Ported from the ASSIGN case of tasks/[id]/route.ts: the new status
    65→    65→     * only moves UNASSIGNED -> ASSIGNED; the assignment history row is
    66→    66→     * typed ASSIGN or REASSIGN; the new assignee is notified (message
    67→    67→     * mirrors the reference implementation exactly — the old assignee is
    68→    68→     * NOT notified, matching route.ts).
    69→    69→     */
    70→    70→    public function assign(Request $request, string $id): RedirectResponse
    71→    71→    {
    72→    72→        $me = $request->user();
    73→    73→        $task = $this->loadTask($id);
    74→    74→        $this->authorize($me, $task);
    75→    75→
    76→    76→        if ($task->isTerminal()) {
    77→    77→            return $this->closedTaskError($task);
    78→    78→        }
    79→    79→
    80→    80→        $data = $request->validate([
    81→    81→            'staffId' => [
    82→    82→                'required',
    83→    83→                'string',
    84→    84→                Rule::exists('users', 'id')->where('is_active', true),
    85→    85→            ],
    86→    86→            'note' => ['nullable', 'string', 'max:1000'],
    87→    87→        ]);
    88→    88→
    89→    89→        $staff = User::query()->findOrFail($data['staffId']);
    90→    90→
    91→    91→        if (! in_array($staff->role, Permissions::assignableRoles(), true)) {
    92→    92→            return $this->actionError($task, __('td.toast.selectStaff'));
    93→    93→        }
    94→    94→
    95→    95→        if ($staff->id === $task->assigned_staff_id) {
    96→    96→            return $this->actionError($task, __('td.toast.selectStaff'));
    97→    97→        }
    98→    98→
    99→    99→        $reassign = $task->assigned_staff_id !== null;
   100→   100→        $fromStatus = $task->status;
   101→   101→        $newStatus = $fromStatus === 'UNASSIGNED' ? 'ASSIGNED' : $fromStatus;
   102→   102→        $note = trim((string) ($data['note'] ?? ''));
   103→   103→
   104→   104→        DB::transaction(function () use ($task, $staff, $me, $reassign, $newStatus, $note): void {
   105→   105→            $task->forceFill([
   106→   106→                'assigned_staff_id' => $staff->id,
   107→   107→                'status' => $newStatus,
   108→   108→            ])->save();
   109→   109→
   110→   110→            TaskAssignment::query()->create([
   111→   111→                'task_id' => $task->id,
   112→   112→                'staff_id' => $staff->id,
   113→   113→                'staff_name' => $staff->name,
   114→   114→                'assigned_by_id' => $me->id,
   115→   115→                'assigned_by_name' => $me->name,
   116→   116→                'type' => $reassign ? 'REASSIGN' : 'ASSIGN',
   117→   117→                'note' => $note !== '' ? $note : null,
   118→   118→            ]);
   119→   119→        });
   120→   120→
   121→   121→        if ($newStatus !== $fromStatus) {
   122→   122→            $this->recordStatusChange($task, $fromStatus, $newStatus, $me, $note !== '' ? $note : null);
   123→   123→        }
   124→   124→
   125→   125→        $message = $this->taskLabel($task).' has been '.($reassign ? 'reassigned' : 'assigned')
   126→   126→            .' to you.'.($note !== '' ? ' Note: '.$note : '');
   127→   127→
   128→   128→        Notifier::users(
   129→   129→            [$staff->id],
   130→   130→            $reassign ? 'TASK_REASSIGNED' : 'TASK_ASSIGNED',
   131→   131→            $message,
   132→   132→            $task->id,
   133→   133→            $task->order_id
   134→   134→        );
   135→   135→
   136→   136→        Audit::log($me, $reassign ? 'TASK_REASSIGNED' : 'TASK_ASSIGNED', 'TASK', $task->id, [
   137→   137→            'staff' => $staff->name,
   138→   138→            'note' => $note !== '' ? $note : null,
   139→   139→        ]);
   140→   140→
   141→   141→        return redirect()
   142→   142→            ->route('tasks.show', $task->id)
   143→   143→            ->with('success', Trans::fill($reassign ? 'td.toast.reassignedTo' : 'td.toast.assignedTo', ['name' => $staff->name]));
   144→   144→    }

   146→   146→    /**
   147→   147→     * POST /tasks/{id}/status — workflow transitions driven by the hidden
   148→   148→     * `action` field: start|submit|ready|deliver|reopen|cancel (view
   149→   149→     * contract). Gates, side effects, notifications and audit codes are
   150→   150→     * ported from the SUBMIT / READY / DELIVER / CANCEL cases of
   151→   151→     * tasks/[id]/route.ts; "start" and "reopen" implement the contract
   152→   152→     * additions (reopen mirrors the REOPENED inspection decision).
   153→   153→     */
   154→   154→    public function status(Request $request, string $id): RedirectResponse
   155→   155→    {
   156→   156→        $me = $request->user();
   157→   157→        $task = $this->loadTask($id);
   158→   158→        $this->authorize($me, $task);
   159→   159→
   160→   160→        if ($task->isTerminal()) {
   161→   161→            return $this->closedTaskError($task);
   162→   162→        }
   163→   163→
   164→   164→        $data = $request->validate([
   165→   165→            'action' => ['required', 'string', Rule::in(['start', 'submit', 'ready', 'deliver', 'reopen', 'cancel'])],
   166→   166→            'note' => ['nullable', 'string', 'max:1000'],
   167→   167→        ]);
   168→   168→
   169→   169→        $action = (string) $data['action'];
   170→   170→        $note = trim((string) ($data['note'] ?? ''));
   171→   171→        $supervisor = $this->isSupervisor($me);
   172→   172→
   173→   173→        switch ($action) {
   174→   174→            case 'start': {
   175→   175→                // Contract: assignee or supervisor, from ASSIGNED or
   176→   176→                // CHANGES_REQUESTED -> ONGOING.
   177→   177→                if (! in_array($task->status, ['ASSIGNED', 'CHANGES_REQUESTED'], true)) {
   178→   178→                    return $this->actionError($task, __('td.actionFailed'));
   179→   179→                }
   180→   180→
   181→   181→                $fromStatus = $task->status;
   182→   182→                $task->forceFill(['status' => 'ONGOING'])->save();
   183→   183→                $this->recordStatusChange($task, $fromStatus, 'ONGOING', $me, $note !== '' ? $note : null);
   184→   184→
   185→   185→                return redirect()
   186→   186→                    ->route('tasks.show', $task->id)
   187→   187→                    ->with('success', Trans::fill('td.hist.statusChange', [
   188→   188→                        'from' => __('labels.status.'.$fromStatus),
   189→   189→                        'to' => __('labels.status.ONGOING'),
   190→   190→                    ]));
   191→   191→            }
   192→   192→
   193→   193→            case 'submit': {
   194→   194→                if (! in_array($task->status, self::IN_WORK_STATUSES, true)) {
   195→   195→                    return $this->actionError($task, __('td.actionFailed'));
   196→   196→                }
   197→   197→
   198→   198→                // Assignees must reach 100% (or complete every applicable
   199→   199→                // stage) before submitting; supervisors may force-submit.
   200→   200→                if (! $supervisor && ! $this->readyForInspection($task)) {
   201→   201→                    return $this->actionError($task, __('td.reach100'));
   202→   202→                }
   203→   203→
   204→   204→                $fromStatus = $task->status;
   205→   205→                $task->forceFill(['status' => 'AWAITING_INSPECTION'])->save();
   206→   206→                $this->recordStatusChange($task, $fromStatus, 'AWAITING_INSPECTION', $me, $note !== '' ? $note : null);
   207→   207→
   208→   208→                if ($note !== '') {
   209→   209→                    TaskComment::query()->create([
   210→   210→                        'task_id' => $task->id,
   211→   211→                        'user_id' => $me->id,
   212→   212→                        'user_name' => $me->name,
   213→   213→                        'message' => 'Submitted for inspection: '.$note,
   214→   214→                    ]);
   215→   215→                }
   216→   216→
   217→   217→                Notifier::management(
   218→   218→                    'SUBMITTED_FOR_INSPECTION',
   219→   219→                    $me->name.' submitted '.$this->taskLabel($task).' for inspection.',
   220→   220→                    $task->id,
   221→   221→                    $task->order_id
   222→   222→                );
   223→   223→
   224→   224→                Audit::log($me, 'SUBMITTED_FOR_INSPECTION', 'TASK', $task->id, [
   225→   225→                    'note' => $note !== '' ? $note : null,
   226→   226→                ]);
   227→   227→
   228→   228→                return redirect()
   229→   229→                    ->route('tasks.show', $task->id)
   230→   230→                    ->with('success', __('td.toast.submitted'));
   231→   231→            }
   232→   232→
   233→   233→            case 'ready': {
   234→   234→                if (! $supervisor || ! Permissions::allows($me, 'production.inspect')) {
   235→   235→                    abort(403);
   236→   236→                }
   237→   237→
   238→   238→                if ($task->status !== 'APPROVED') {
   239→   239→                    return $this->actionError($task, __('td.actionFailed'));
   240→   240→                }
   241→   241→
   242→   242→                $fromStatus = $task->status;
   243→   243→                $task->forceFill(['status' => 'READY_FOR_PICKUP'])->save();
   244→   244→                $this->recordStatusChange($task, $fromStatus, 'READY_FOR_PICKUP', $me);
   245→   245→
   246→   246→                Notifier::management(
   247→   247→                    'READY_FOR_PICKUP',
   248→   248→                    $this->taskLabel($task).' is approved and ready for customer pickup.',
   249→   249→                    $task->id,
   250→   250→                    $task->order_id
   251→   251→                );
   252→   252→
   253→   253→                if ($task->assigned_staff_id !== null) {
   254→   254→                    Notifier::users(
   255→   255→                        [$task->assigned_staff_id],
   256→   256→                        'READY_FOR_PICKUP',
   257→   257→                        $this->taskLabel($task).' is ready for customer pickup.',
   258→   258→                        $task->id,
   259→   259→                        $task->order_id
   260→   260→                    );
   261→   261→                }
   262→   262→
   263→   263→                Audit::log($me, 'TASK_READY_FOR_PICKUP', 'TASK', $task->id);
   264→   264→
   265→   265→                return redirect()
   266→   266→                    ->route('tasks.show', $task->id)
   267→   267→                    ->with('success', __('td.toast.ready'));
   268→   268→            }
   269→   269→
   270→   270→            case 'deliver': {
   271→   271→                if (! $supervisor || ! Permissions::allows($me, 'production.inspect')) {
   272→   272→                    abort(403);
   273→   273→                }
   274→   274→
   275→   275→                if ($task->status !== 'READY_FOR_PICKUP') {
   276→   276→                    return $this->actionError($task, __('td.actionFailed'));
   277→   277→                }
   278→   278→
   279→   279→                $fromStatus = $task->status;
   280→   280→                $task->forceFill([
   281→   281→                    'status' => 'DELIVERED',
   282→   282→                    'delivered_at' => now(),
   283→   283→                ])->save();
   284→   284→                $this->recordStatusChange($task, $fromStatus, 'DELIVERED', $me, $note !== '' ? $note : null);
   285→   285→
   286→   286→                $message = $this->taskLabel($task).' has been delivered/collected by the customer.';
   287→   287→
   288→   288→                Notifier::management('TASK_DELIVERED', $message, $task->id, $task->order_id);
   289→   289→
   290→   290→                if ($task->assigned_staff_id !== null) {
   291→   291→                    Notifier::users([$task->assigned_staff_id], 'TASK_DELIVERED', $message, $task->id, $task->order_id);
   292→   292→                }
   293→   293→
   294→   294→                Audit::log($me, 'TASK_DELIVERED', 'TASK', $task->id, [
   295→   295→                    'note' => $note !== '' ? $note : null,
   296→   296→                ]);
   297→   297→
   298→   298→                return redirect()
   299→   299→                    ->route('tasks.show', $task->id)
   300→   300→                    ->with('success', __('td.toast.delivered'));
   301→   301→            }
   302→   302→
   303→   303→            case 'reopen': {
   304→   304→                if (! $supervisor) {
   305→   305→                    abort(403);
   306→   306→                }
   307→   307→
   308→   308→                if (! in_array($task->status, ['APPROVED', 'READY_FOR_PICKUP', 'AWAITING_INSPECTION', 'CHANGES_REQUESTED'], true)) {
   309→   309→                    return $this->actionError($task, __('td.actionFailed'));
   310→   310→                }
   311→   311→
   312→   312→                $reason = trim((string) $request->input('reason', ''));
   313→   313→
   314→   314→                if ($reason === '' || mb_strlen($reason) < 3) {
   315→   315→                    throw ValidationException::withMessages([
   316→   316→                        'reason' => Trans::fill('td.toast.reasonRequiredDesc', [
   317→   317→                            'field' => __('td.dialog.reopenPlaceholder'),
   318→   318→                            'min' => 3,
   319→   319→                        ]),
   320→   320→                    ]);
   321→   321→                }
   322→   322→
   323→   323→                $fromStatus = $task->status;
   324→   324→                $task->forceFill([
   325→   325→                    'status' => 'ONGOING',
   326→   326→                    'completed_at' => null,
   327→   327→                ])->save();
   328→   328→                $this->recordStatusChange($task, $fromStatus, 'ONGOING', $me, $reason);
   329→   329→
   330→   330→                if ($task->assigned_staff_id !== null) {
   331→   331→                    Notifier::users(
   332→   332→                        [$task->assigned_staff_id],
   333→   333→                        'TASK_REOPENED',
   334→   334→                        $this->taskLabel($task).' has been reopened for production.',
   335→   335→                        $task->id,
   336→   336→                        $task->order_id
   337→   337→                    );
   338→   338→                }
   339→   339→
   340→   340→                return redirect()
   341→   341→                    ->route('tasks.show', $task->id)
   342→   342→                    ->with('success', __('td.toast.reopened'));
   343→   343→            }
   344→   344→
   345→   345→            case 'cancel': {
   346→   346→                if (! $supervisor || ! Permissions::allows($me, 'production.cancel')) {
   347→   347→                    abort(403);
   348→   348→                }
   349→   349→
   350→   350→                $reason = trim((string) $request->input('reason', ''));
   351→   351→
   352→   352→                if ($reason === '' || mb_strlen($reason) < 3) {
   353→   353→                    throw ValidationException::withMessages([
   354→   354→                        'reason' => Trans::fill('td.toast.reasonRequiredDesc', [
   355→   355→                            'field' => __('td.reason.cancel'),
   356→   356→                            'min' => 3,
   357→   357→                        ]),
   358→   358→                    ]);
   359→   359→                }
   360→   360→
   361→   361→                $fromStatus = $task->status;
   362→   362→                $task->forceFill([
   363→   363→                    'status' => 'CANCELLED',
   364→   364→                    'cancel_reason' => $reason,
   365→   365→                ])->save();
   366→   366→                $this->recordStatusChange($task, $fromStatus, 'CANCELLED', $me, $reason);
   367→   367→
   368→   368→                if ($task->assigned_staff_id !== null) {
   369→   369→                    Notifier::users(
   370→   370→                        [$task->assigned_staff_id],
   371→   371→                        'TASK_CANCELLED',
   372→   372→                        $this->taskLabel($task).' has been cancelled. Reason: '.$reason,
   373→   373→                        $task->id,
   374→   374→                        $task->order_id
   375→   375→                    );
   376→   376→                }
   377→   377→
   378→   378→                Audit::log($me, 'TASK_CANCELLED', 'TASK', $task->id, ['reason' => $reason]);
   379→   379→
   380→   380→                return redirect()
   381→   381→                    ->route('tasks.show', $task->id)
   382→   382→                    ->with('success', __('td.toast.cancelled'));
   383→   383→            }
   384→   384→        }
   385→   385→
   386→   386→        return $this->actionError($task, __('td.actionFailed'));
   387→   387→    }

   389→   389→    /**
   390→   390→     * POST /tasks/{id}/progress — manual numerical progress update
   391→   391→     * (ported from tasks/[id]/progress/route.ts). 100% never auto-approves:
   392→   392→     * inspection submission stays an explicit action. The first positive
   393→   393→     * value on an ASSIGNED task moves it to ONGOING.
   394→   394→     */
   395→   395→    public function progress(Request $request, string $id): RedirectResponse
   396→   396→    {
   397→   397→        $me = $request->user();
   398→   398→        $task = $this->loadTask($id);
   399→   399→        $this->authorize($me, $task);
   400→   400→
   401→   401→        $supervisor = $this->isSupervisor($me);
   402→   402→
   403→   403→        if ($task->isTerminal()) {
   404→   404→            return $this->closedTaskError($task);
   405→   405→        }
   406→   406→
   407→   407→        if (! $supervisor && ! in_array($task->status, self::IN_WORK_STATUSES, true)) {
   408→   408→            return $this->actionError($task, __('td.toast.progressFailed'));
   409→   409→        }
   410→   410→
   411→   411→        // View contract names the field "value"; fall back to the
   412→   412→        // reference implementation's "progress" name just in case.
   413→   413→        if ($request->missing('value') && $request->has('progress')) {
   414→   414→            $request->merge(['value' => $request->input('progress')]);
   415→   415→        }
   416→   416→
   417→   417→        $data = $request->validate([
   418→   418→            'value' => ['required', 'integer', 'min:0', 'max:100'],
   419→   419→            'note' => ['nullable', 'string', 'max:500'],
   420→   420→        ]);
   421→   421→
   422→   422→        $value = (int) $data['value'];
   423→   423→        $previous = (int) $task->progress;
   424→   424→
   425→   425→        if ($value === $previous) {
   426→   426→            return $this->actionError($task, __('td.toast.progressFailed'));
   427→   427→        }
   428→   428→
   429→   429→        $note = trim((string) ($data['note'] ?? ''));
   430→   430→        $fromStatus = $task->status;
   431→   431→        $newStatus = $fromStatus === 'ASSIGNED' && $value > 0 ? 'ONGOING' : $fromStatus;
   432→   432→
   433→   433→        DB::transaction(function () use ($task, $me, $value, $previous, $newStatus, $note): void {
   434→   434→            $task->forceFill([
   435→   435→                'progress' => $value,
   436→   436→                'status' => $newStatus,
   437→   437→            ])->save();
   438→   438→
   439→   439→            TaskProgressHistory::query()->create([
   440→   440→                'task_id' => $task->id,
   441→   441→                'user_id' => $me->id,
   442→   442→                'user_name' => $me->name,
   443→   443→                'previous_progress' => $previous,
   444→   444→                'new_progress' => $value,
   445→   445→                'source' => 'MANUAL',
   446→   446→                'note' => $note !== '' ? $note : null,
   447→   447→            ]);
   448→   448→        });
   449→   449→
   450→   450→        if ($newStatus !== $fromStatus) {
   451→   451→            $this->recordStatusChange($task, $fromStatus, $newStatus, $me, 'Production started');
   452→   452→        }
   453→   453→
   454→   454→        Audit::log($me, 'PROGRESS_UPDATED', 'TASK', $task->id, [
   455→   455→            'from' => $previous,
   456→   456→            'to' => $value,
   457→   457→            'note' => $note !== '' ? $note : null,
   458→   458→            'correction' => $supervisor,
   459→   459→        ]);
   460→   460→
   461→   461→        return redirect()
   462→   462→            ->route('tasks.show', $task->id)
   463→   463→            ->with('success', Trans::fill('td.toast.progressUpdated', ['value' => $value]));
   464→   464→    }

   929→   929→        }
   930→   930→
   931→   931→        $instructionsInput = trim((string) ($data['instructions'] ?? ''));
   932→   932→
   933→   933→        if ($instructionsInput !== '' && $instructionsInput !== (string) $task->instructions) {
   934→   934→            $update['instructions'] = $instructionsInput;
   935→   935→            $changes['instructions'] = 'updated';
   936→   936→            $instructionsChanged = true;
   937→   937→        }
   938→   938→
   939→   939→        $garmentPieceInput = trim((string) ($data['garmentPiece'] ?? ''));
   940→   940→
   941→   941→        if ($garmentPieceInput !== '' && $garmentPieceInput !== (string) $task->garment_piece) {
   942→   942→            $clash = ProductionTask::query()
   943→   943→                ->where('order_id', $task->order_id)
   944→   944→                ->where('garment_piece', $garmentPieceInput)
   945→   945→                ->where('status', '!=', 'CANCELLED')
   946→   946→                ->where('id', '!=', $task->id)
   947→   947→                ->exists();
   948→   948→
   949→   949→            if ($clash) {
   950→   950→                throw ValidationException::withMessages([
   951→   951→                    'garmentPiece' => 'Piece "'.$garmentPieceInput.'" already exists on this order.',
   952→   952→                ]);
   953→   953→            }
   954→   954→
   955→   955→            $update['garment_piece'] = $garmentPieceInput;
   956→   956→        }
   957→   957→
   958→   958→        if ($update === []) {
   959→   959→            return $this->actionError($task, __('td.toast.nothingToUpdate'));
   960→   960→        }
   961→   961→
   962→   962→        $task->forceFill($update)->save();
   963→   963→
   964→   964→        if ($deadlineChanged && $task->assigned_staff_id !== null) {
   965→   965→            Notifier::users(
   966→   966→                [$task->assigned_staff_id],
   967→   967→                'DEADLINE_CHANGED',
   968→   968→                'The deadline of '.$this->taskLabel($task).' has been changed.',
   969→   969→                $task->id,
   970→   970→                $task->order_id
   971→   971→            );
   972→   972→        }
   973→   973→
   974→   974→        if ($instructionsChanged && $task->assigned_staff_id !== null) {
   975→   975→            Notifier::users(
   976→   976→                [$task->assigned_staff_id],
   977→   977→                'NEW_INSTRUCTIONS',
   978→   978→                'New production instructions were added to '.$this->taskLabel($task).'.',
   979→   979→                $task->id,
   980→   980→                $task->order_id
   981→   981→            );
   982→   982→        }
   983→   983→
   984→   984→        Audit::log($me, 'TASK_UPDATED', 'TASK', $task->id, $changes);
   985→   985→
   986→   986→        return redirect()
   987→   987→            ->route('tasks.show', $task->id)
   988→   988→            ->with('success', __('td.toast.detailsUpdated'));
   989→   989→    }

  1054→  1054→    /**
  1055→  1055→     * Whether an assignee may submit the task for inspection: manual
  1056→  1056→     * progress reached 100% OR every applicable stage is COMPLETED (view
  1057→  1057→     * contract; supervisors bypass this check like in route.ts).
  1058→  1058→     */
  1059→  1059→    private function readyForInspection(ProductionTask $task): bool
  1060→  1060→    {
  1061→  1061→        if ((int) $task->progress >= 100) {
  1062→  1062→            return true;
  1063→  1063→        }
  1064→  1064→
  1065→  1065→        $stages = ProductionStage::query()
  1066→  1066→            ->where('is_active', true)
  1067→  1067→            ->orderBy('sort_order')
  1068→  1068→            ->get()

  1070→  1070→                static fn (ProductionStage $stage): bool => Atelier::stageAppliesTo(
  1071→  1071→                    (string) $stage->applies_to,

  1075→  1075→

  1084→  1084→
  1085→  1085→        foreach ($stages as $stage) {

  1088→  1088→                : 'NOT_STARTED';

  1091→  1091→                return false;
