     1→"use client";
     2→
     3→// OSSZ ATELIER — user accounts & roles: create accounts, edit profiles,
     4→// reset passwords and activate/deactivate. Admins manage staff accounts only;
     5→// Management manages every account (incl. the last-manager safeguard).
     6→
     7→import { useCallback, useEffect, useState } from "react";
     8→import { api, ApiClientError } from "../client";
     9→import { can, useAppStore } from "../store";
    10→import type { UserRecord } from "../types";
    11→import { useI18n, type TranslateFn } from "../i18n/use-i18n";
    12→import { roleLabel } from "../i18n/core";
    13→import { EmptyState, PageHeader, RoleBadge } from "../ui-bits";
    14→import { useToast } from "@/hooks/use-toast";
    15→import { Button } from "@/components/ui/button";
    16→import { Input } from "@/components/ui/input";
    17→import { Label } from "@/components/ui/label";
    18→import { Skeleton } from "@/components/ui/skeleton";
    19→import { Badge } from "@/components/ui/badge";
    20→import {
    21→  Dialog,
    22→  DialogContent,
    23→  DialogDescription,
    24→  DialogFooter,
    25→  DialogHeader,
    26→  DialogTitle,
    27→} from "@/components/ui/dialog";
    28→import {
    29→  AlertDialog,
    30→  AlertDialogAction,
    31→  AlertDialogCancel,
    32→  AlertDialogContent,
    33→  AlertDialogDescription,
    34→  AlertDialogFooter,
    35→  AlertDialogHeader,
    36→  AlertDialogTitle,
    37→} from "@/components/ui/alert-dialog";
    38→import {
    39→  DropdownMenu,
    40→  DropdownMenuContent,
    41→  DropdownMenuItem,
    42→  DropdownMenuLabel,
    43→  DropdownMenuSeparator,
    44→  DropdownMenuTrigger,
    45→} from "@/components/ui/dropdown-menu";
    46→import {
    47→  Select,
    48→  SelectContent,
    49→  SelectItem,
    50→  SelectTrigger,
    51→  SelectValue,
    52→} from "@/components/ui/select";
    53→import {
    54→  Table,
    55→  TableBody,
    56→  TableCell,
    57→  TableHead,
    58→  TableHeader,
    59→  TableRow,
    60→} from "@/components/ui/table";
    61→import { ROLES } from "@/lib/permissions";
    62→import { formatDate } from "@/lib/format";
    63→import {
    64→  AlertTriangle,
    65→  Eye,
    66→  EyeOff,
    67→  KeyRound,
    68→  MoreHorizontal,
    69→  Pencil,
    70→  Power,
    71→  ShieldCheck,
    72→  UserPlus,
    73→  Users,
    74→} from "lucide-react";
    75→
    76→type CreateForm = {
    77→  name: string;
    78→  email: string;
    79→  password: string;
    80→  role: string;
    81→  phone: string;
    82→  jobTitle: string;
    83→};
    84→
    85→type EditForm = {
    86→  name: string;
    87→  phone: string;
    88→  jobTitle: string;
    89→  role: string;
    90→};
    91→
    92→/** Per-role permission hint shown in the create dialog (bilingual). */
    93→function roleHint(role: string, t: TranslateFn): string {
    94→  switch (role) {
    95→    case "MANAGEMENT":
    96→      return t("users.roleHint.management");
    97→    case "ADMIN":
    98→      return t("users.roleHint.admin");
    99→    case "STAFF":
   100→      return t("users.roleHint.staff");
   101→    default:
   102→      return "";
   103→  }
   104→}
   105→
   106→export function UsersView() {
   107→  const user = useAppStore((s) => s.user);
   108→  const { t, lang } = useI18n();
   109→  const { toast } = useToast();
   110→  const canManageStaff = can(user, "users.manage.staff") || can(user, "users.manage.all");
   111→  const isManagement = user?.role === "MANAGEMENT";
   112→
   113→  const [users, setUsers] = useState<UserRecord[] | null>(null);
   114→  const [error, setError] = useState<string | null>(null);
   115→
   116→  const [createOpen, setCreateOpen] = useState(false);
   117→  const [createForm, setCreateForm] = useState<CreateForm>({
   118→    name: "",
   119→    email: "",
   120→    password: "",
   121→    role: "STAFF",
   122→    phone: "",
   123→    jobTitle: "",
   124→  });
   125→  const [showCreatePw, setShowCreatePw] = useState(false);
   126→  const [formError, setFormError] = useState<string | null>(null);
   127→  const [submitting, setSubmitting] = useState(false);
   128→
   129→  const [editTarget, setEditTarget] = useState<UserRecord | null>(null);
   130→  const [editForm, setEditForm] = useState<EditForm>({ name: "", phone: "", jobTitle: "", role: "STAFF" });
   131→
   132→  const [pwTarget, setPwTarget] = useState<UserRecord | null>(null);
   133→  const [newPassword, setNewPassword] = useState("");
   134→  const [showNewPw, setShowNewPw] = useState(false);
   135→
   136→  const [statusTarget, setStatusTarget] = useState<UserRecord | null>(null);
   137→
   138→  const load = useCallback(async () => {
   139→    try {
   140→      const res = await api.get<{ users: UserRecord[] }>("/api/atelier/users");
   141→      setUsers(res.users);
   142→      setError(null);
   143→    } catch (e) {
   144→      // Empty string marks a generic failure; the translated message is
   145→      // rendered at display time so it follows the active language.
   146→      setError(e instanceof Error ? e.message : "");
   147→    }
   148→  }, []);
   149→
   150→  useEffect(() => {
   151→    load();
   152→  }, [load]);
   153→
   154→  function errMessage(e: unknown): string {
   155→    if (e instanceof ApiClientError) return e.details?.issues?.[0]?.message ?? e.message;
   156→    return t("common.somethingWrong");
   157→  }
   158→
   159→  function openCreate() {
   160→    setCreateForm({ name: "", email: "", password: "", role: "STAFF", phone: "", jobTitle: "" });
   161→    setShowCreatePw(false);
   162→    setFormError(null);
   163→    setCreateOpen(true);
   164→  }
   165→
   166→  async function submitCreate() {
   167→    if (!createForm.name.trim() || !createForm.email.trim()) {
   168→      setFormError(t("users.nameEmailRequired"));
   169→      return;
   170→    }
   171→    if (createForm.password.length < 8) {
   172→      setFormError(t("users.passwordTooShort"));
   173→      return;
   174→    }
   175→    setSubmitting(true);
   176→    setFormError(null);
   177→    try {
   178→      const res = await api.post<{ user: UserRecord }>("/api/atelier/users", {
   179→        name: createForm.name,
   180→        email: createForm.email,
   181→        password: createForm.password,
   182→        role: createForm.role,
   183→        ...(createForm.phone ? { phone: createForm.phone } : {}),
   184→        ...(createForm.jobTitle ? { jobTitle: createForm.jobTitle } : {}),
   185→      });
   186→      toast({
   187→        title: t("users.createdToastTitle"),
   188→        description: t("users.createdToastDesc", {
   189→          name: res.user.name,
   190→          role: roleLabel(res.user.role, lang).toLowerCase(),
   191→        }),
   192→      });
   193→      setCreateOpen(false);
   194→      await load();
   195→    } catch (e) {
   196→      const message = errMessage(e);
   197→      setFormError(message);
   198→      toast({ title: t("users.createFailedToast"), description: message, variant: "destructive" });
   199→    } finally {
   200→      setSubmitting(false);
   201→    }
   202→  }
   203→
   204→  function openEdit(u: UserRecord) {
   205→    setEditTarget(u);
   206→    setEditForm({
   207→      name: u.name,
   208→      phone: u.phone ?? "",
   209→      jobTitle: u.jobTitle ?? "",
   210→      role: u.role,
   211→    });
   212→  }
   213→
   214→  async function submitEdit() {
   215→    if (!editTarget) return;
   216→    if (!editForm.name.trim() || editForm.name.trim().length < 2) {
   217→      toast({
   218→        title: t("users.updateFailedToast"),
   219→        description: t("users.nameTooShort"),
   220→        variant: "destructive",
   221→      });
   222→      return;
   223→    }
   224→    setSubmitting(true);
   225→    try {
   226→      await api.patch(`/api/atelier/users/${editTarget.id}`, {
   227→        name: editForm.name,
   228→        phone: editForm.phone,
   229→        jobTitle: editForm.jobTitle,
   230→        ...(isManagement ? { role: editForm.role } : {}),
   231→      });
   232→      toast({ title: t("users.updatedToastTitle"), description: t("users.savedToast", { name: editForm.name }) });
   233→      setEditTarget(null);
   234→      await load();
   235→    } catch (e) {
   236→      toast({ title: t("users.updateFailedToast"), description: errMessage(e), variant: "destructive" });
   237→    } finally {
   238→      setSubmitting(false);
   239→    }
   240→  }
   241→
   242→  async function submitPasswordReset() {
   243→    if (!pwTarget) return;
   244→    if (newPassword.length < 8) return;
   245→    setSubmitting(true);
   246→    try {
   247→      await api.patch(`/api/atelier/users/${pwTarget.id}`, { password: newPassword });
   248→      toast({ title: t("users.resetToastTitle"), description: t("users.resetToastDesc", { name: pwTarget.name }) });
   249→      setPwTarget(null);
   250→      setNewPassword("");
   251→      setShowNewPw(false);
   252→    } catch (e) {
   253→      toast({ title: t("users.resetFailedToast"), description: errMessage(e), variant: "destructive" });
   254→    } finally {
   255→      setSubmitting(false);
   256→    }
   257→  }
   258→
   259→  async function submitStatusChange() {
   260→    if (!statusTarget) return;
   261→    const next = !statusTarget.isActive;
   262→    setSubmitting(true);
   263→    try {
   264→      await api.patch(`/api/atelier/users/${statusTarget.id}`, { isActive: next });
   265→      toast({
   266→        title: next ? t("users.reactivatedToastTitle") : t("users.deactivatedToastTitle"),
   267→        description: next
   268→          ? t("users.reactivatedToastDesc", { name: statusTarget.name })
   269→          : t("users.deactivatedToastDesc", { name: statusTarget.name }),
   270→      });
   271→      setStatusTarget(null);
   272→      await load();
   273→    } catch (e) {
   274→      toast({ title: t("users.updateFailedToast"), description: errMessage(e), variant: "destructive" });
   275→    } finally {
   276→      setSubmitting(false);
   277→    }
   278→  }
   279→
   280→  function canActOn(u: UserRecord): boolean {
   281→    return isManagement || u.role === "STAFF";
   282→  }
   283→
   284→  return (
   285→    <div>
   286→      <PageHeader
   287→        title={t("users.title")}
   288→        description={t("users.description")}
   289→      >
   290→        {canManageStaff && (
   291→          <Button onClick={openCreate}>
   292→            <UserPlus className="size-4 mr-1.5" />
   293→            {t("users.createAccount")}
   294→          </Button>
   295→        )}
   296→      </PageHeader>
   297→
   298→      {error !== null ? (
   299→        <EmptyState
   300→          icon={AlertTriangle}
   301→          title={t("users.errorTitle")}
   302→          description={error || t("users.loadFailed")}
   303→          action={
   304→            <Button variant="outline" size="sm" onClick={load}>
   305→              Try again
   306→            </Button>
   307→          }
   308→        />
   309→      ) : users === null ? (
   310→        <div className="space-y-3">
   311→          <Skeleton className="h-10 w-full" />
   312→          <Skeleton className="h-16 w-full" />
   313→          <Skeleton className="h-16 w-full" />
   314→          <Skeleton className="h-16 w-full" />
   315→        </div>
   316→      ) : users.length === 0 ? (
   317→        <div className="rounded-lg border">
   318→          <EmptyState icon={Users} title={t("users.emptyTitle")} />
   319→        </div>
   320→      ) : (
   321→        <div className="rounded-lg border overflow-x-auto">
   322→          <Table className="min-w-[980px]">
   323→            <TableHeader>
   324→              <TableRow>
   325→                <TableHead>{t("users.col.userId")}</TableHead>
   326→                <TableHead>{t("users.col.user")}</TableHead>
   327→                <TableHead>{t("users.col.role")}</TableHead>
   328→                <TableHead>{t("users.col.jobTitle")}</TableHead>
   329→                <TableHead>{t("users.col.phone")}</TableHead>
   330→                <TableHead>{t("users.col.status")}</TableHead>
   331→                <TableHead className="text-center">{t("users.col.activeTasks")}</TableHead>
   332→                <TableHead>{t("users.col.created")}</TableHead>
   333→                {canManageStaff && <TableHead className="w-12 text-right sr-only">{t("common.actions")}</TableHead>}
   334→              </TableRow>
   335→            </TableHeader>
   336→            <TableBody>
   337→              {users.map((u) => (
   338→                <TableRow key={u.id}>
   339→                  <TableCell>
   340→                    <span className="inline-flex rounded-md border bg-muted/50 px-2 py-1 font-mono text-xs font-semibold tracking-wide text-foreground">
   341→                      {u.code ?? "—"}
   342→                    </span>
   343→                  </TableCell>
   344→                  <TableCell>
   345→                    <p className="font-medium">{u.name}</p>
   346→                    <p className="text-xs text-muted-foreground truncate">{u.email}</p>
   347→                  </TableCell>
   348→                  <TableCell>
   349→                    <RoleBadge role={u.role} />
   350→                  </TableCell>
   351→                  <TableCell className="text-muted-foreground">{u.jobTitle ?? "—"}</TableCell>
   352→                  <TableCell className="whitespace-nowrap text-muted-foreground">{u.phone ?? "—"}</TableCell>
   353→                  <TableCell>
   354→                    {u.isActive ? (
   355→                      <Badge variant="outline" className="bg-emerald-50 text-emerald-800 border-emerald-300 whitespace-nowrap">
   356→                        {t("users.active")}
   357→                      </Badge>
   358→                    ) : (
   359→                      <Badge variant="outline" className="bg-stone-100 text-stone-500 border-stone-300 whitespace-nowrap">
   360→                        {t("users.inactive")}
   361→                      </Badge>
   362→                    )}
   363→                  </TableCell>
   364→                  <TableCell className="text-center tabular-nums">
   365→                    <span className="inline-flex rounded-full bg-primary/10 text-primary text-xs font-bold px-2 py-0.5">
   366→                      {u.activeTaskCount ?? 0}
   367→                    </span>
   368→                  </TableCell>
   369→                  <TableCell className="text-muted-foreground whitespace-nowrap">{formatDate(u.createdAt)}</TableCell>
   370→                  {canManageStaff && (
   371→                    <TableCell className="text-right">
   372→                      {canActOn(u) ? (
   373→                        <DropdownMenu>
   374→                          <DropdownMenuTrigger asChild>
   375→                            <Button variant="ghost" size="icon" className="size-8" aria-label={t("users.actionsFor", { name: u.name })}>
   376→                              <MoreHorizontal className="size-4" />
   377→                            </Button>
   378→                          </DropdownMenuTrigger>
   379→                          <DropdownMenuContent align="end" className="w-48">
   380→                            <DropdownMenuLabel>{u.name}</DropdownMenuLabel>
   381→                            <DropdownMenuSeparator />
   382→                            <DropdownMenuItem onClick={() => openEdit(u)}>
   383→                              <Pencil className="size-4 mr-2" />
   384→                              {t("users.editDetails")}
   385→                            </DropdownMenuItem>
   386→                            <DropdownMenuItem onClick={() => { setPwTarget(u); setNewPassword(""); setShowNewPw(false); }}>
   387→                              <KeyRound className="size-4 mr-2" />
   388→                              {t("users.resetPassword")}
   389→                            </DropdownMenuItem>
   390→                            {u.id !== user?.id && (
   391→                              <DropdownMenuItem
   392→                                onClick={() => setStatusTarget(u)}
   393→                                className={!u.isActive ? "" : "text-red-700 focus:text-red-700 focus:bg-red-50"}
   394→                              >
   395→                                <Power className="size-4 mr-2" />
   396→                                {u.isActive ? t("users.deactivate") : t("users.reactivate")}
   397→                              </DropdownMenuItem>
   398→                            )}
   399→                          </DropdownMenuContent>
   400→                        </DropdownMenu>
   401→                      ) : (
   402→                        <span className="text-xs text-muted-foreground">{t("users.adminAccessNote")}</span>
   403→                      )}
   404→                    </TableCell>
   405→                  )}
   406→                </TableRow>
   407→              ))}
   408→            </TableBody>
   409→          </Table>
   410→        </div>
   411→      )}
   412→
   413→      {/* ---------------------------------------------------- create account */}
   414→      <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   415→        <DialogContent className="sm:max-w-lg max-h-[85vh] overflow-y-auto">
   416→          <DialogHeader>
   417→            <DialogTitle>{t("users.createAccount")}</DialogTitle>
   418→            <DialogDescription>
   419→              {t("users.createDescription")}
   420→            </DialogDescription>
   421→          </DialogHeader>
   422→          <div className="grid gap-4 sm:grid-cols-2">
   423→            <div className="grid gap-1.5">
   424→              <Label htmlFor="new-name">{t("users.fullName")}</Label>
   425→              <Input
   426→                id="new-name"
   427→                value={createForm.name}
   428→                onChange={(e) => setCreateForm({ ...createForm, name: e.target.value })}
   429→                placeholder={t("users.namePlaceholder")}
   430→                maxLength={120}
   431→              />
   432→            </div>
   433→            <div className="grid gap-1.5">
   434→              <Label htmlFor="new-email">{t("users.email")}</Label>
   435→              <Input
   436→                id="new-email"
   437→                type="email"
   438→                value={createForm.email}
   439→                onChange={(e) => setCreateForm({ ...createForm, email: e.target.value })}
   440→                placeholder="name@atelier.com"
   441→              />
   442→            </div>
   443→            <div className="grid gap-1.5">
   444→              <Label htmlFor="new-password">{t("users.password")}</Label>
   445→              <div className="relative">
   446→                <Input
   447→                  id="new-password"
   448→                  type={showCreatePw ? "text" : "password"}
   449→                  value={createForm.password}
   450→                  onChange={(e) => setCreateForm({ ...createForm, password: e.target.value })}
   451→                  placeholder={t("users.passwordPlaceholder")}
   452→                  className="pr-10"
   453→                  autoComplete="new-password"
   454→                />
   455→                <button
   456→                  type="button"
   457→                  onClick={() => setShowCreatePw(!showCreatePw)}
   458→                  className="absolute right-2 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   459→                  aria-label={showCreatePw ? t("common.hidePassword") : t("common.showPassword")}
   460→                >
   461→                  {showCreatePw ? <EyeOff className="size-4" /> : <Eye className="size-4" />}
   462→                </button>
   463→              </div>
   464→            </div>
   465→            <div className="grid gap-1.5">
   466→              <Label htmlFor="new-role">{t("users.roleRequired")}</Label>
   467→              <Select
   468→                value={createForm.role}
   469→                onValueChange={(v) => setCreateForm({ ...createForm, role: v })}
   470→              >
   471→                <SelectTrigger id="new-role">
   472→                  <SelectValue />
   473→                </SelectTrigger>
   474→                <SelectContent>
   475→                  {(isManagement ? ROLES : (["STAFF"] as const)).map((r) => (
   476→                    <SelectItem key={r} value={r}>
   477→                      {roleLabel(r, lang)}
   478→                    </SelectItem>
   479→                  ))}
   480→                </SelectContent>
   481→              </Select>
   482→            </div>
   483→            <div className="grid gap-1.5">
   484→              <Label htmlFor="new-phone">{t("users.phoneLabel")}</Label>
   485→              <Input
   486→                id="new-phone"
   487→                value={createForm.phone}
   488→                onChange={(e) => setCreateForm({ ...createForm, phone: e.target.value })}
   489→                maxLength={40}
   490→              />
   491→            </div>
   492→            <div className="grid gap-1.5">
   493→              <Label htmlFor="new-job">{t("users.jobTitleLabel")}</Label>
   494→              <Input
   495→                id="new-job"
   496→                value={createForm.jobTitle}
   497→                onChange={(e) => setCreateForm({ ...createForm, jobTitle: e.target.value })}
   498→                placeholder={t("users.jobTitlePlaceholder")}
   499→                maxLength={120}
   500→              />
   501→            </div>
   502→            <div className="sm:col-span-2 rounded-lg border bg-muted/40 p-3 flex gap-2">
   503→              <ShieldCheck className="size-4 shrink-0 mt-0.5 text-primary" />
   504→              <p className="text-xs text-muted-foreground">
   505→                {roleHint(createForm.role, t)}
   506→                {createForm.role === "STAFF" && ` ${t("users.roleHint.staffExtra")}`}
   507→              </p>
   508→            </div>
   509→            {formError && (
   510→              <p className="sm:col-span-2 text-sm font-medium text-red-700">{formError}</p>
   511→            )}
   512→          </div>
   513→          <DialogFooter>
   514→            <Button variant="outline" onClick={() => setCreateOpen(false)} disabled={submitting}>
   515→              {t("common.cancel")}
   516→            </Button>
   517→            <Button onClick={submitCreate} disabled={submitting}>
   518→              {submitting && (
   519→                <span className="size-4 mr-1.5 animate-spin rounded-full border-2 border-current border-t-transparent" />
   520→              )}
   521→              {t("users.createAccount")}
   522→            </Button>
   523→          </DialogFooter>
   524→        </DialogContent>
   525→      </Dialog>
   526→
   527→      {/* ------------------------------------------------------- edit dialog */}
   528→      <Dialog open={!!editTarget} onOpenChange={(open) => !open && setEditTarget(null)}>
   529→        <DialogContent className="sm:max-w-md">
   530→          <DialogHeader>
   531→            <DialogTitle>{t("users.editTitle")}</DialogTitle>
   532→            <DialogDescription>{editTarget?.email}</DialogDescription>
   533→          </DialogHeader>
   534→          <div className="grid gap-4">
   535→            <div className="grid gap-1.5">
   536→              <Label htmlFor="edit-name">{t("users.fullName")}</Label>
   537→              <Input
   538→                id="edit-name"
   539→                value={editForm.name}
   540→                onChange={(e) => setEditForm({ ...editForm, name: e.target.value })}
   541→                maxLength={120}
   542→              />
   543→            </div>
   544→            <div className="grid gap-1.5">
   545→              <Label htmlFor="edit-phone">{t("users.phoneLabel")}</Label>
   546→              <Input
   547→                id="edit-phone"
   548→                value={editForm.phone}
   549→                onChange={(e) => setEditForm({ ...editForm, phone: e.target.value })}
   550→                maxLength={40}
   551→              />
   552→            </div>
   553→            <div className="grid gap-1.5">
   554→              <Label htmlFor="edit-job">{t("users.jobTitleLabel")}</Label>
   555→              <Input
   556→                id="edit-job"
   557→                value={editForm.jobTitle}
   558→                onChange={(e) => setEditForm({ ...editForm, jobTitle: e.target.value })}
   559→                maxLength={120}
   560→              />
   561→            </div>
   562→            {isManagement && (
   563→              <div className="grid gap-1.5">
   564→                <Label htmlFor="edit-role">{t("users.roleLabel")}</Label>
   565→                <Select value={editForm.role} onValueChange={(v) => setEditForm({ ...editForm, role: v })}>
   566→                  <SelectTrigger id="edit-role">
   567→                    <SelectValue />
   568→                  </SelectTrigger>
   569→                  <SelectContent>
   570→                    {ROLES.map((r) => (
   571→                      <SelectItem key={r} value={r}>
   572→                        {roleLabel(r, lang)}
   573→                      </SelectItem>
   574→                    ))}
   575→                  </SelectContent>
   576→                </Select>
   577→              </div>
   578→            )}
   579→          </div>
   580→          <DialogFooter>
   581→            <Button variant="outline" onClick={() => setEditTarget(null)} disabled={submitting}>
   582→              {t("common.cancel")}
   583→            </Button>
   584→            <Button onClick={submitEdit} disabled={submitting}>
   585→              {submitting && (
   586→                <span className="size-4 mr-1.5 animate-spin rounded-full border-2 border-current border-t-transparent" />
   587→              )}
   588→              {t("common.saveChanges")}
   589→            </Button>
   590→          </DialogFooter>
   591→        </DialogContent>
   592→      </Dialog>
   593→
   594→      {/* ----------------------------------------------- reset password alert */}
   595→      <AlertDialog open={!!pwTarget} onOpenChange={(open) => !open && setPwTarget(null)}>
   596→        <AlertDialogContent>
   597→          <AlertDialogHeader>
   598→            <AlertDialogTitle>{t("users.resetPassword")}</AlertDialogTitle>
   599→            <AlertDialogDescription>
   600→              {t("users.resetPwDescription", { name: pwTarget?.name ?? "", email: pwTarget?.email ?? "" })}
   601→            </AlertDialogDescription>
   602→          </AlertDialogHeader>
   603→          <div className="grid gap-1.5">
   604→            <Label htmlFor="reset-password">{t("users.newPassword")}</Label>
   605→            <div className="relative">
   606→              <Input
   607→                id="reset-password"
   608→                type={showNewPw ? "text" : "password"}
   609→                value={newPassword}
   610→                onChange={(e) => setNewPassword(e.target.value)}
   611→                placeholder={t("users.passwordPlaceholder")}
   612→                className="pr-10"
   613→                autoComplete="new-password"
   614→              />
   615→              <button
   616→                type="button"
   617→                onClick={() => setShowNewPw(!showNewPw)}
   618→                className="absolute right-2 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   619→                aria-label={showNewPw ? t("common.hidePassword") : t("common.showPassword")}
   620→              >
   621→                {showNewPw ? <EyeOff className="size-4" /> : <Eye className="size-4" />}
   622→              </button>
   623→            </div>
   624→            {newPassword.length > 0 && newPassword.length < 8 && (
   625→              <p className="text-xs text-red-700">{t("users.passwordTooShort")}</p>
   626→            )}
   627→          </div>
   628→          <AlertDialogFooter>
   629→            <AlertDialogCancel disabled={submitting}>{t("common.cancel")}</AlertDialogCancel>
   630→            <AlertDialogAction
   631→              disabled={newPassword.length < 8 || submitting}
   632→              onClick={(e) => {
   633→                e.preventDefault();
   634→                submitPasswordReset();
   635→              }}
   636→              className="bg-primary text-primary-foreground hover:bg-primary/90"
   637→            >
   638→              {t("users.resetPasswordAction")}
   639→            </AlertDialogAction>
   640→          </AlertDialogFooter>
   641→        </AlertDialogContent>
   642→      </AlertDialog>
   643→
   644→      {/* ------------------------------------- deactivate / reactivate alert */}
   645→      <AlertDialog open={!!statusTarget} onOpenChange={(open) => !open && setStatusTarget(null)}>
   646→        <AlertDialogContent>
   647→          <AlertDialogHeader>
   648→            <AlertDialogTitle>
   649→              {statusTarget?.isActive ? t("users.deactivateTitle") : t("users.reactivateTitle")}
   650→            </AlertDialogTitle>
   651→            <AlertDialogDescription>
   652→              {statusTarget?.isActive
   653→                ? t("users.deactivateDesc", { name: statusTarget?.name ?? "" })
   654→                : t("users.reactivateDesc", { name: statusTarget?.name ?? "" })}
   655→            </AlertDialogDescription>
   656→          </AlertDialogHeader>
   657→          <AlertDialogFooter>
   658→            <AlertDialogCancel disabled={submitting}>{t("common.cancel")}</AlertDialogCancel>
   659→            <AlertDialogAction
   660→              disabled={submitting}
   661→              onClick={(e) => {
   662→                e.preventDefault();
   663→                submitStatusChange();
   664→              }}
   665→              className={
   666→                statusTarget?.isActive
   667→                  ? "bg-red-600 text-white hover:bg-red-700"
   668→                  : "bg-primary text-primary-foreground hover:bg-primary/90"
   669→              }
   670→            >
   671→              {statusTarget?.isActive ? t("users.deactivate") : t("users.reactivate")}
   672→            </AlertDialogAction>
   673→          </AlertDialogFooter>
   674→        </AlertDialogContent>
   675→      </AlertDialog>
   676→    </div>
   677→  );
   678→}
