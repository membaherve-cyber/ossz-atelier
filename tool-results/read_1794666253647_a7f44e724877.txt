   281→   281→    progressHistory?: Array<{ from: number; to: number; by: string; at: Date; note?: string }>;

   305→   305→      return def[3] === "ALL" || def[3].split(",").includes(category);
   306→   306→    };

   318→   318→        data: {
   319→   319→          taskId: task.id,
   320→   320→          stageId: stages[stageCode],
   321→   321→          state,
   322→   322→          updatedById: opts.assignedStaffId ?? null,

   324→   324→          updatedAt: opts.createdAt,
   325→   325→        },

   332→   332→          data: {
   333→   333→            taskId: task.id,
   334→   334→            stageId: stages[u.stageCode],
   335→   335→            stageName: names[u.stageCode],
   336→   336→            userId: u.by === "hamidou" ? hamidou.id : u.by === "lawal" ? lawal.id : aminata.id,
   337→   337→            userName: users[u.by],
   338→   338→            previousState: u.from,
   339→   339→            newState: u.to,

   341→   341→          },

   363→   363→      for (const p of opts.progressHistory) {

   366→   366→            taskId: task.id,

   372→   372→            note: p.note ?? null,
   373→   373→            createdAt: p.at,
   374→   374→          },
   375→   375→        });
   376→   376→      }

   379→   379→      await db.taskAssignment.create({
   380→   380→        data: {
   381→   381→          taskId: task.id,
   382→   382→          staffId: opts.assignedStaffId,
   383→   383→          staffName:
   384→   384→            opts.assignedStaffId === hamidou.id ? hamidou.name : opts.assignedStaffId === lawal.id ? lawal.name : "",

   387→   387→          type: "ASSIGN",
   388→   388→          createdAt: opts.createdAt,
   389→   389→        },
   390→   390→      });
   391→   391→    }
   392→   392→    return task;
