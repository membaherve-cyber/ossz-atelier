    88→    public function run(): void

    90→        // Idempotency guard — never duplicate an already-seeded database.
    91→        if (User::count() > 0) {
    92→            echo "Already seeded — users exist, nothing to do.\n";
    93→
    94→            return;
    95→        }
    96→
    97→        echo "Seeding OSSZ ATELIER standalone database...\n";
    98→
    99→        $this->seedUsers();              // §1 accounts (Management / Admin / Staff)
   100→        $this->seedStages();             // §2 configurable production stages
   101→        $this->seedPerformanceConfig();  // §3 performance config + incentive schemes

   104→        $this->seedHistoryAndExtras();   // §7 comments, §8 notifications, §9 assessments/incentives, §10 audit

   106→        echo "Seed complete.\n";
   107→        echo 'Demo accounts (password: '.self::DEMO_PASSWORD."):\n";
   108→        echo "  MANAGEMENT  management@atelier.com\n";
   109→        echo "  ADMIN       admin@atelier.com\n";

  1106→    /**
  1107→     * Create one production task plus its checklist stage states, stage
  1108→     * update log, status workflow chain, manual progress history and
  1109→     * creation-time assignment row — 1:1 port of the `createTask()`
  1110→     * helper in prisma/seed.ts.
  1111→     *
  1112→     * Option keys:
  1113→     *  - code, order_id, garment_piece, status, progress, created_at  (required)
  1114→     *  - category (default FULL), instructions, priority (default NORMAL),
  1115→     *    assigned_staff_id, deadline                                  (optional)
  1116→     *  - stage_states:      list of [stageCode, state] pairs (appliesTo filtered)
  1117→     *  - stage_updates:     list of ['stage_code', 'from', 'to', 'by', 'at']
  1118→     *  - status_chain:      list of ['to', 'by', 'at', 'reason'?] (from = previous)
  1119→     *  - progress_history:  list of ['from', 'to', 'by', 'at', 'note'?]
  1120→     *
  1121→     * @param array<string, mixed> $opts
  1122→     */
  1123→    private function createTask(array $opts): ProductionTask
  1124→    {
  1125→        $aminata = $this->users['aminata'];
  1126→        $hamidou = $this->users['hamidou'];
  1127→        $lawal = $this->users['lawal'];
  1128→
  1129→        $task = ProductionTask::create([
  1130→            'code' => $opts['code'],
  1131→            'order_id' => $opts['order_id'],
  1132→            'garment_piece' => $opts['garment_piece'],
  1133→            'category' => $opts['category'] ?? 'FULL',
  1134→            'instructions' => $opts['instructions'] ?? null,
  1135→            'deadline' => $opts['deadline'],
  1136→            'priority' => $opts['priority'] ?? 'NORMAL',
  1137→            'status' => $opts['status'],
  1138→            'progress' => $opts['progress'],
  1139→            'assigned_staff_id' => $opts['assigned_staff_id'] ?? null,
  1140→            'created_by_id' => $aminata->id,
  1141→            'created_at' => $opts['created_at'],
  1142→            'updated_at' => Carbon::now(),
  1143→        ]);
  1144→
  1145→        $category = $opts['category'] ?? 'FULL';
  1146→
  1147→        // Current checklist stage states (filtered by appliesTo vs category).
  1148→        foreach ($opts['stage_states'] as [$stageCode, $state]) {
  1149→            if (! $this->stageAppliesTo($stageCode, $category)) {
  1150→                continue;
  1151→            }
  1152→
  1153→            TaskStageState::create([
  1154→                'task_id' => $task->id,
  1155→                'stage_id' => $this->stages[$stageCode],
  1156→                'state' => $state,
  1157→                'updated_by_id' => $opts['assigned_staff_id'] ?? null,
  1158→                'updated_by_name' => ($opts['assigned_staff_id'] ?? null) === $hamidou->id
  1159→                    ? $hamidou->name
  1160→                    : $lawal->name,
  1161→                'updated_at' => $opts['created_at'],
  1162→            ]);
  1163→        }
  1164→
  1165→        // Immutable stage state change log.
  1166→        foreach ($opts['stage_updates'] ?? [] as $update) {
  1167→            if (! $this->stageAppliesTo($update['stage_code'], $category)) {
  1168→                continue;
  1169→            }
  1170→
  1171→            TaskStageUpdate::create([
  1172→                'task_id' => $task->id,
  1173→                'stage_id' => $this->stages[$update['stage_code']],
  1174→                'stage_name' => $this->stageName($update['stage_code']),
  1175→                'user_id' => $update['by'] === 'hamidou'
  1176→                    ? $hamidou->id
  1177→                    : ($update['by'] === 'lawal' ? $lawal->id : $aminata->id),
  1178→                'user_name' => $this->users[$update['by']]->name,
  1179→                'previous_state' => $update['from'],
  1180→                'new_state' => $update['to'],
  1181→                'created_at' => $update['at'],
  1182→            ]);
  1183→        }
  1184→
  1185→        // Status workflow history (each entry chains from the previous status).
  1186→        $previous = 'UNASSIGNED';
  1187→
  1188→        foreach ($opts['status_chain'] ?? [] as $change) {
  1189→            TaskStatusChange::create([
  1190→                'task_id' => $task->id,
  1191→                'user_id' => $this->statusActorId($change['by']),
  1192→                'user_name' => ($this->users[$change['by']] ?? $aminata)->name,
  1193→                'from_status' => $previous,
  1194→                'to_status' => $change['to'],
  1195→                'reason' => $change['reason'] ?? null,
  1196→                'created_at' => $change['at'],
  1197→            ]);
  1198→
  1199→            $previous = $change['to'];
  1200→        }
  1201→
  1202→        // Manual progress history.
  1203→        foreach ($opts['progress_history'] ?? [] as $entry) {
  1204→            TaskProgressHistory::create([
  1205→                'task_id' => $task->id,
  1206→                'user_id' => $entry['by'] === 'hamidou' ? $hamidou->id : $lawal->id,
  1207→                'user_name' => $this->users[$entry['by']]->name,
  1208→                'previous_progress' => $entry['from'],
  1209→                'new_progress' => $entry['to'],
  1210→                'source' => 'MANUAL',
  1211→                'note' => $entry['note'] ?? null,
  1212→                'created_at' => $entry['at'],
  1213→            ]);
  1214→        }
  1215→
  1216→        // Creation-time assignment row (assigned by Management).
  1217→        if (! empty($opts['assigned_staff_id'])) {
  1218→            TaskAssignment::create([
  1219→                'task_id' => $task->id,
  1220→                'staff_id' => $opts['assigned_staff_id'],
  1221→                'staff_name' => $opts['assigned_staff_id'] === $hamidou->id
  1222→                    ? $hamidou->name
  1223→                    : ($opts['assigned_staff_id'] === $lawal->id ? $lawal->name : ''),
  1224→                'assigned_by_id' => $aminata->id,
  1225→                'assigned_by_name' => $aminata->name,
  1226→                'type' => 'ASSIGN',
  1227→                'created_at' => $opts['created_at'],
  1228→            ]);
  1229→        }
  1230→
  1231→        return $task;
  1232→    }

  1240→    /**
  1241→     * Stage states for a fully completed & delivered task of the given
  1242→     * category (doneStates from prisma/seed.ts): every applicable stage
  1243→     * COMPLETED except "blocked" / "other" which are NOT_APPLICABLE.
  1244→     *
  1245→     * @return array<string, string> stage code => state
  1246→     */
  1247→    private function doneStates(string $category): array
  1248→    {
  1249→        $states = [];
  1250→
  1251→        foreach (self::STAGE_DEFS as [$code, , , $appliesTo]) {
  1252→            if (! ($appliesTo === 'ALL' || in_array($category, explode(',', $appliesTo), true))) {
  1253→                continue;
  1254→            }
  1255→
  1256→            $states[$code] = ($code === 'blocked' || $code === 'other') ? 'NOT_APPLICABLE' : 'COMPLETED';
  1257→        }
  1258→
  1259→        return $states;
  1260→    }
  1261→
  1262→    /** True when the stage (by code) applies to the given task category. */
  1263→    private function stageAppliesTo(string $stageCode, string $category): bool
  1264→    {
  1265→        foreach (self::STAGE_DEFS as [$code, , , $appliesTo]) {
  1266→            if ($code === $stageCode) {
  1267→                return $appliesTo === 'ALL'
  1268→                    || in_array($category, explode(',', $appliesTo), true);
  1269→            }
  1270→        }
  1271→
  1272→        return false;
  1273→    }
