   130→   130→/** Convert an ISO timestamp to a `datetime-local` input value (local time). */
   131→   131→function toDatetimeLocal(iso: string | null): string {
   132→   132→  if (!iso) return "";

   134→   134→  if (Number.isNaN(d.getTime())) return "";
   135→   135→  return format(d, "yyyy-MM-dd'T'HH:mm");
   136→   136→}
