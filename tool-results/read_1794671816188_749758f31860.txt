    27→/**
    28→ * Production task workflow mutations (spec §3: /tasks/{id}/…).
    29→ *
    30→ * Ported from the PATCH/POST branches of
    31→ * src/app/api/atelier/tasks/[id]/route.ts (+ progress, stages, comments
    32→ * sub-routes). "Mixed" access: every action requires the acting user to be
    33→ * the assignee or a supervisor (MANAGEMENT/ADMIN); actions gated by a
    34→ * permission in the reference re-check it here because the routes only
    35→ * carry `auth`.
    36→ *
    37→ * Status transitions (EXACT reference rules):
    38→ * - start:    ASSIGNED → ONGOING (assignee or supervisor)
    39→ * - submit:   ASSIGNED|ONGOING|CHANGES_REQUESTED → AWAITING_INSPECTION
    40→ *             (non-supervisors must have reached 100% progress)
    41→ * - inspect:  APPROVED|CHANGES_REQUESTED only from AWAITING_INSPECTION;
    42→ *             REOPENED only from APPROVED|READY_FOR_PICKUP
    43→ * - ready:    APPROVED → READY_FOR_PICKUP (supervisor only)
    44→ * - deliver:  READY_FOR_PICKUP → DELIVERED (supervisor only)
    45→ * - reopen:   APPROVED|READY_FOR_PICKUP → ONGOING (supervisor only)
    46→ * - cancel:   any non-terminal status → CANCELLED (supervisor only, reason)
    47→ * - block:    ASSIGNED|ONGOING|CHANGES_REQUESTED → BLOCKED (reason)
    48→ * - unblock:  BLOCKED → ONGOING (progress > 0) or ASSIGNED
    49→ */
    50→class TaskActionController

    56→    /**
    57→     * POST /tasks/{id}/assign — assign or reassign the task.
    58→     */
    59→    public function assign(Request $request, string $id): RedirectResponse
    60→    {
    61→        $data = $request->validate([
    62→            'staffId' => ['required', 'string'],
    63→            'note' => ['nullable', 'string', 'max:1000'],
    64→        ]);
    65→
    66→        /** @var User $user */
    67→        $user = $request->user();
    68→        $task = $this->taskOrFail($id);
    69→
    70→        if (! $this->canAccess($user, $task)) {
    71→            abort(403);
    72→        }
    73→
    74→        if ($task->isTerminal()) {
    75→            return $this->closedTask($id);
    76→        }
    77→
    78→        $staff = User::query()->find($data['staffId']);
    79→
    80→        if ($staff === null || ! $staff->is_active) {
    81→            return $this->taskError($id, __('flash.invalidStaffAccount'));
    82→        }
    83→
    84→        if ($staff->id === $task->assigned_staff_id) {
    85→            return $this->taskError($id, __('flash.taskAlreadyAssigned'));
    86→        }
    87→
    88→        $reassign = $task->assigned_staff_id !== null;
    89→        $oldAssigneeId = $task->assigned_staff_id;
    90→        $newStatus = $task->status === 'UNASSIGNED' ? 'ASSIGNED' : $task->status;
    91→        $note = $this->text($data, 'note');
    92→
    93→        DB::transaction(function () use ($task, $staff, $user, $reassign, $newStatus, $note): void {
    94→            $task->forceFill([
    95→                'assigned_staff_id' => $staff->id,
    96→                'status' => $newStatus,
    97→            ])->save();
    98→
    99→            TaskAssignment::query()->create([
   100→                'task_id' => $task->id,
   101→                'staff_id' => $staff->id,
   102→                'staff_name' => $staff->name,
   103→                'assigned_by_id' => $user->id,
   104→                'assigned_by_name' => $user->name,
   105→                'type' => $reassign ? 'REASSIGN' : 'ASSIGN',
   106→                'note' => $note,
   107→            ]);
   108→        });
   109→
   110→        if ($newStatus !== $task->status) {
   111→            $this->recordStatusChange($task, $task->status, $newStatus, $note, $user);
   112→        }
   113→
   114→        $label = $this->taskLabel($task);
   115→
   116→        Notifier::users(
   117→            [$staff->id],
   118→            $reassign ? 'TASK_REASSIGNED' : 'TASK_ASSIGNED',
   119→            $label.' has been '.($reassign ? 'reassigned' : 'assigned').' to you.'
   120→            .($note !== null ? ' Note: '.$note : ''),
   121→            $task->id,
   122→            $task->order_id
   123→        );
   124→
   125→        // Spec §5: reassignment notifies BOTH the old and the new assignee.
   126→        if ($reassign && $oldAssigneeId !== null) {
   127→            Notifier::users(
   128→                [$oldAssigneeId],
   129→                'TASK_REASSIGNED',
   130→                $label.' has been reassigned to '.$staff->name.'.',
   131→                $task->id,
   132→                $task->order_id
   133→            );
   134→        }
   135→
   136→        Audit::log($user, $reassign ? 'TASK_REASSIGNED' : 'TASK_ASSIGNED', 'TASK', $task->id, [
   137→            'staff' => $staff->name,
   138→            'note' => $note,
   139→        ]);
   140→
   141→        return $this->taskSuccess(
   142→            $id,
   143→            __($reassign ? 'td.toast.reassignedTo' : 'td.toast.assignedTo', ['name' => $staff->name])
   144→        );
   145→    }

   151→    /**
   152→     * POST /tasks/{id}/status — workflow status transitions.
   153→     */
   154→    public function status(Request $request, string $id): RedirectResponse
   155→    {
   156→        $data = $request->validate([
   157→            'action' => ['required', 'string', 'in:start,submit,ready,deliver,reopen,cancel'],
   158→            'reason' => ['nullable', 'string', 'max:2000'],
   159→            'note' => ['nullable', 'string', 'max:1000'],
   160→            'comments' => ['nullable', 'string', 'max:2000'],
   161→        ]);
   162→
   163→        /** @var User $user */
   164→        $user = $request->user();
   165→        $task = $this->taskOrFail($id);
   166→        $supervisor = $this->supervisor($user);
   167→
   168→        if (! $this->canAccess($user, $task)) {
   169→            abort(403);
   170→        }
   171→
   172→        if ($task->isTerminal()) {
   173→            return $this->closedTask($id);
   174→        }
   175→
   176→        return match ($data['action']) {
   177→            'start' => $this->startTask($task, $user),
   178→            'submit' => $this->submitTask($task, $user, $this->text($data, 'note')),
   179→            'ready' => $this->readyTask($task, $user, $supervisor),
   180→            'deliver' => $this->deliverTask($task, $user, $supervisor, $this->text($data, 'note')),
   181→            'reopen' => $this->reopenTask($task, $user, $supervisor, $this->text($data, 'comments')),
   182→            'cancel' => $this->cancelTask($task, $user, $supervisor, $this->text($data, 'reason')),
   183→            default => $this->taskError($id, __('td.actionFailed')),
   184→        };
   185→    }

   187→    /**
   188→     * action=start — ASSIGNED → ONGOING (assignee or supervisor).
   189→     */
   190→    private function startTask(ProductionTask $task, User $user): RedirectResponse
   191→    {
   192→        if ($task->status !== 'ASSIGNED') {
   193→            return $this->taskError($task->id, __('flash.statusNoStart'));
   194→        }
   195→
   196→        $task->forceFill(['status' => 'ONGOING'])->save();
   197→        $this->recordStatusChange($task, 'ASSIGNED', 'ONGOING', 'Production started', $user);
   198→
   199→        return $this->taskSuccess($task->id, __('flash.taskStarted'));
   200→    }

   202→    /**
   203→     * action=submit — ASSIGNED|ONGOING|CHANGES_REQUESTED → AWAITING_INSPECTION.
   204→     * Non-supervisors must have reached 100% progress.
   205→     */
   206→    private function submitTask(ProductionTask $task, User $user, ?string $note): RedirectResponse
   207→    {
   208→        $supervisor = $this->supervisor($user);
   209→
   210→        if (! in_array($task->status, ['ASSIGNED', 'ONGOING', 'CHANGES_REQUESTED'], true)) {
   211→            return $this->taskError(
   212→                $task->id,
   213→                __('flash.statusNoSubmit', ['status' => __('labels.status.'.$task->status)])
   214→            );
   215→        }
   216→
   217→        if (! $supervisor && $task->progress < 100) {
   218→            return $this->taskError($task->id, __('td.toast.reach100'));
   219→        }
   220→
   221→        $task->forceFill(['status' => 'AWAITING_INSPECTION'])->save();
   222→        $this->recordStatusChange($task, $task->status, 'AWAITING_INSPECTION', $note, $user);
   223→
   224→        if ($note !== null) {
   225→            TaskComment::query()->create([
   226→                'task_id' => $task->id,
   227→                'user_id' => $user->id,
   228→                'user_name' => $user->name,
   229→                'message' => 'Submitted for inspection: '.$note,
   230→            ]);
   231→        }
   232→
   233→        Notifier::management(
   234→            'SUBMITTED_FOR_INSPECTION',
   235→            $user->name.' submitted '.$this->taskLabel($task).' for inspection.',
   236→            $task->id,
   237→            $task->order_id
   238→        );
   239→
   240→        Audit::log($user, 'SUBMITTED_FOR_INSPECTION', 'TASK', $task->id, ['note' => $note]);
   241→
   242→        return $this->taskSuccess($task->id, __('td.toast.submitted'));
   243→    }

   326→     * action=reopen — APPROVED|READY_FOR_PICKUP → ONGOING (production.inspect,

   329→    private function reopenTask(ProductionTask $task, User $user, bool $supervisor, ?string $comments): RedirectResponse

   336→            return $this->taskError($task->id, __('flash.statusNoReopen'));

   338→
   339→        if ($comments === null || mb_strlen($comments) < 3) {
   340→            return $this->taskError($task->id, __('td.toast.reopenCommentsRequired'));

   342→
   343→        $fromStatus = $task->status;
   344→
   345→        $task->forceFill([
   346→            'status' => 'ONGOING',
   347→            'completed_at' => null,
   348→        ])->save();
   349→
   350→        $this->recordStatusChange($task, $fromStatus, 'ONGOING', $comments, $user);
   351→
   352→        if ($task->assigned_staff_id !== null) {
   353→            Notifier::users(
   354→                [$task->assigned_staff_id],
   355→                'TASK_REOPENED',
   356→                $this->taskLabel($task).' has been reopened for production.',
   357→                $task->id,
   358→                $task->order_id
   359→            );
   360→        }
   361→
   362→        InspectionRecord::query()->create([
   363→            'task_id' => $task->id,
   364→            'inspector_id' => $user->id,
   365→            'inspector_name' => $user->name,
   366→            'decision' => 'REOPENED',
   367→            'comments' => $comments,
   368→        ]);
   369→
   370→        Audit::log($user, 'INSPECTION_COMPLETED', 'TASK', $task->id, [
   371→            'decision' => 'REOPENED',
   372→            'comments' => $comments,
   373→        ]);
   374→
   375→        return $this->taskSuccess($task->id, __('td.toast.reopened'));
   376→    }

   420→    /**
   421→     * POST /tasks/{id}/progress — numerical progress update (append-only
   422→     * history; 100% never auto-approves — inspection is explicit).
   423→     */
   424→    public function progress(Request $request, string $id): RedirectResponse
   425→    {
   426→        $data = $request->validate([
   427→            'progress' => ['required', 'integer', 'min:0', 'max:100'],
   428→            'note' => ['nullable', 'string', 'max:500'],
   429→        ]);
   430→
   431→        /** @var User $user */
   432→        $user = $request->user();
   433→        $task = $this->taskOrFail($id);
   434→        $supervisor = $this->supervisor($user);
   435→
   436→        if (! $this->canAccess($user, $task)) {
   437→            abort(403);
   438→        }
   439→
   440→        if ($task->isTerminal()) {
   441→            return $this->closedTask($id);
   442→        }
   443→
   444→        if (! $supervisor && $task->status === 'BLOCKED') {
   445→            return $this->taskError($id, __('flash.resolveBlockerFirst'));
   446→        }
   447→
   448→        if (! $supervisor && ! in_array($task->status, ['ASSIGNED', 'ONGOING', 'CHANGES_REQUESTED'], true)) {
   449→            return $this->taskError(
   450→                $id,
   451→                __('flash.progressLocked', ['status' => __('labels.status.'.$task->status)])
   452→            );
   453→        }
   454→
   455→        $progress = (int) $data['progress'];
   456→
   457→        if ($progress === $task->progress) {
   458→            return $this->taskError($id, __('flash.progressUnchanged', ['value' => $task->progress]));
   459→        }
   460→
   461→        $note = $this->text($data, 'note');
   462→        $newStatus = $task->status === 'ASSIGNED' && $progress > 0 ? 'ONGOING' : $task->status;
   463→
   464→        DB::transaction(function () use ($task, $user, $progress, $note, $newStatus): void {
   465→            $task->forceFill([
   466→                'progress' => $progress,
   467→                'status' => $newStatus,
   468→            ])->save();
   469→
   470→            TaskProgressHistory::query()->create([
   471→                'task_id' => $task->id,
   472→                'user_id' => $user->id,
   473→                'user_name' => $user->name,
   474→                'previous_progress' => $task->getOriginal('progress'),
   475→                'new_progress' => $progress,
   476→                'source' => 'MANUAL',
   477→                'note' => $note,
   478→            ]);
   479→        });
   480→
   481→        if ($newStatus !== $task->getOriginal('status')) {
   482→            $this->recordStatusChange($task, 'ASSIGNED', 'ONGOING', 'Production started', $user);
   483→        }
   484→
   485→        Audit::log($user, 'PROGRESS_UPDATED', 'TASK', $task->id, [
   486→            'from' => $task->getOriginal('progress'),
   487→            'to' => $progress,
   488→            'note' => $note,
   489→            'correction' => $supervisor,
   490→        ]);
   491→
   492→        return $this->taskSuccess($id, __('td.toast.progressUpdated', ['value' => $progress]));
   493→    }

   499→    /**
   500→     * POST /tasks/{id}/stages — update production checklist stages. Every
   501→     * change is recorded immutably; progress is recalculated from the
   502→     * applicable stages via ProgressService.
   503→     */
   504→    public function stages(Request $request, string $id): RedirectResponse
   505→    {
   506→        $data = $request->validate([
   507→            'updates' => ['required', 'array', 'min:1', 'max:30'],
   508→            'updates.*.stageId' => ['required', 'string'],
   509→            'updates.*.state' => ['required', 'string', 'in:'.implode(',', Atelier::STAGE_STATES)],
   510→            'updates.*.comment' => ['nullable', 'string', 'max:500'],
   511→        ]);
   512→
   513→        /** @var User $user */
   514→        $user = $request->user();
   515→        $task = $this->taskOrFail($id);
   516→        $supervisor = $this->supervisor($user);
   517→
   518→        if (! $this->canAccess($user, $task)) {
   519→            abort(403);
   520→        }
   521→
   522→        if ($task->isTerminal()) {
   523→            return $this->closedTask($id);
   524→        }
   525→
   526→        if (! $supervisor && ! in_array($task->status, ['ASSIGNED', 'ONGOING', 'CHANGES_REQUESTED'], true)) {
   527→            return $this->taskError(
   528→                $id,
   529→                __('flash.checklistLocked', ['status' => __('labels.status.'.$task->status)])
   530→            );
   531→        }
   532→
   533→        // Validate the stages against this task's category workflow.
   534→        $applicable = ProductionStage::query()
   535→            ->where('is_active', true)
   536→            ->orderBy('sort_order')
   537→            ->get()
   538→            ->filter(fn (ProductionStage $stage): bool => Atelier::stageAppliesTo(
   539→                (string) $stage->applies_to,
   540→                (string) $task->category
   541→            ));
   542→
   543→        $stageById = $applicable->keyBy('id');
   544→
   545→        foreach ($data['updates'] as $update) {
   546→            if (! $stageById->has($update['stageId'])) {
   547→                return $this->taskError($id, __('flash.stageNotApplicable'));
   548→            }
   549→        }
   550→
   551→        $existingStates = TaskStageState::query()
   552→            ->where('task_id', $task->id)
   553→            ->get()
   554→            ->keyBy('stage_id');
   555→
   556→        $applied = [];
   557→
   558→        DB::transaction(function () use ($data, $task, $user, $stageById, $existingStates, &$applied): void {
   559→            foreach ($data['updates'] as $update) {
   560→                $stage = $stageById->get($update['stageId']);
   561→                $previous = (string) ($existingStates->get($stage->id)->state ?? 'NOT_STARTED');
   562→                $state = (string) $update['state'];
   563→
   564→                if ($previous === $state) {
   565→                    continue;
   566→                }
   567→
   568→                $comment = isset($update['comment']) && $update['comment'] !== ''
   569→                    ? (string) $update['comment'] : null;
   570→
   571→                TaskStageUpdate::query()->create([
   572→                    'task_id' => $task->id,
   573→                    'stage_id' => $stage->id,
   574→                    'stage_name' => $stage->name,
   575→                    'user_id' => $user->id,
   576→                    'user_name' => $user->name,
   577→                    'previous_state' => $previous,
   578→                    'new_state' => $state,
   579→                    'comment' => $comment,
   580→                ]);
   581→
   582→                TaskStageState::query()->updateOrCreate(
   583→                    ['task_id' => $task->id, 'stage_id' => $stage->id],
   584→                    [
   585→                        'state' => $state,
   586→                        'updated_by_id' => $user->id,
   587→                        'updated_by_name' => $user->name,
   588→                    ]
   589→                );
   590→
   591→                $applied[] = [
   592→                    'stageName' => $stage->name,
   593→                    'previousState' => $previous,
   594→                    'newState' => $state,
   595→                ];
   596→            }
   597→        });
   598→
   599→        if ($applied === []) {
   600→            return $this->taskError($id, __('flash.noStageChanges'));
   601→        }
   602→
   603→        // Recalculate numerical progress from the checklist (never decreases).
   604→        $task->refresh();
   605→        ProgressService::recompute($task);
   606→
   607→        // First update moves an assigned task into Ongoing.
   608→        if ($task->status === 'ASSIGNED') {
   609→            $task->forceFill(['status' => 'ONGOING'])->save();
   610→            $this->recordStatusChange($task, 'ASSIGNED', 'ONGOING', 'Production started', $user);
   611→        }
   612→
   613→        Audit::log($user, 'STAGES_UPDATED', 'TASK', $task->id, ['changes' => $applied]);
   614→
   615→        return $this->taskSuccess($id, __('td.toast.checklistUpdated'));
   616→    }

   624→     * (APPROVED | CHANGES_REQUESTED | REOPENED).

   626→    public function inspect(Request $request, string $id): RedirectResponse
   627→    {
   628→        $data = $request->validate([
   629→            'decision' => ['required', 'string', 'in:'.implode(',', Atelier::INSPECTION_DECISIONS)],
   630→            'comments' => ['nullable', 'string', 'max:2000'],

   660→        }
   661→
   662→        return $this->requestChanges($task, $user, $comments);

   687→                    'source' => 'INSPECTION',
   688→                    'note' => 'Set to 100% on approval',
   689→                ]);
   690→            }
   691→        });
   692→
   693→        $this->recordStatusChange($task, 'AWAITING_INSPECTION', 'APPROVED', $comments, $user);

   697→                [$task->assigned_staff_id],

   706→            'task_id' => $task->id,
   707→            'inspector_id' => $user->id,
   708→            'inspector_name' => $user->name,
   709→            'decision' => 'APPROVED',
   710→            'comments' => $comments,
   711→        ]);

   713→        Audit::log($user, 'INSPECTION_COMPLETED', 'TASK', $task->id, [

   918→    /**
   919→     * POST /tasks/{id}/update — edit task details (deadline, priority,
   920→     * instructions, garment piece). The assignee is notified when the
   921→     * deadline or the instructions change.
   922→     */
   923→    public function update(Request $request, string $id): RedirectResponse
   924→    {
   925→        $data = $request->validate([
   926→            'deadline' => ['nullable', 'date'],
   927→            'priority' => ['nullable', 'string', 'in:'.implode(',', Atelier::PRIORITIES)],
   928→            'instructions' => ['nullable', 'string', 'max:2000'],
   929→            'garmentPiece' => ['nullable', 'string', 'max:100'],
   930→        ]);
   931→
   932→        /** @var User $user */
   933→        $user = $request->user();
   934→        $task = $this->taskOrFail($id);
   935→
   936→        if (! $this->canAccess($user, $task)) {
   937→            abort(403);
   938→        }
   939→
   940→        if ($task->isTerminal()) {
   941→            return $this->closedTask($id);
   942→        }
   943→
   944→        $changes = [];
   945→        $force = [];
   946→
   947→        if ($request->has('deadline')) {
   948→            $deadline = $this->text($data, 'deadline') !== null ? Carbon::parse((string) $data['deadline']) : null;
   949→
   950→            $force['deadline'] = $deadline;
   951→            $changes['deadline'] = ['from' => $task->deadline?->toIso8601String(), 'to' => $deadline?->toIso8601String()];
   952→        }
   953→
   954→        if ($request->has('priority') && $this->text($data, 'priority') !== null) {
   955→            $force['priority'] = (string) $data['priority'];
   956→            $changes['priority'] = ['from' => $task->priority, 'to' => (string) $data['priority']];
   957→        }
   958→
   959→        if ($request->has('instructions')) {
   960→            $instructions = $this->text($data, 'instructions');
   961→            $force['instructions'] = $instructions;
   962→            $changes['instructions'] = 'updated';
   963→        }
   964→
   965→        if ($request->has('garmentPiece') && $this->text($data, 'garmentPiece') !== null) {
   966→            $piece = trim((string) $data['garmentPiece']);
   967→
   968→            if ($piece === '') {
   969→                return $this->taskError($id, __('td.toast.pieceRequired'));
   970→            }
   971→
   972→            $clash = ProductionTask::query()
   973→                ->where('order_id', $task->order_id)
   974→                ->where('garment_piece', $piece)
   975→                ->where('status', '!=', 'CANCELLED')
   976→                ->where('id', '!=', $task->id)
   977→                ->exists();
   978→
   979→            if ($clash) {
   980→                return $this->taskError($id, __('flash.pieceExistsOnOrder', [
   981→                    'name' => $piece,
   982→                    'order' => $task->order->order_number,
   983→                ]));
   984→            }
   985→
   986→            $force['garment_piece'] = $piece;
   987→        }
   988→
   989→        if ($force === []) {
   990→            return $this->taskError($id, __('td.toast.nothingToUpdate'));
   991→        }
   992→
   993→        $deadlineChanged = array_key_exists('deadline', $force);
   994→        $instructionsChanged = array_key_exists('instructions', $force);
   995→        $label = $this->taskLabel($task);
   996→
   997→        $task->forceFill($force)->save();
   998→
   999→        if ($deadlineChanged && $task->assigned_staff_id !== null) {
  1000→            Notifier::users(
  1001→                [$task->assigned_staff_id],
  1002→                'DEADLINE_CHANGED',
  1003→                'The deadline of '.$label.' has been changed.',
  1004→                $task->id,
  1005→                $task->order_id
  1006→            );
  1007→        }
  1008→
  1009→        if ($instructionsChanged && $task->assigned_staff_id !== null) {
  1010→            Notifier::users(
  1011→                [$task->assigned_staff_id],
  1012→                'NEW_INSTRUCTIONS',
  1013→                'New production instructions were added to '.$label.'.',
  1014→                $task->id,
  1015→                $task->order_id
  1016→            );
  1017→        }
  1018→
  1019→        Audit::log($user, 'TASK_UPDATED', 'TASK', $task->id, $changes);
  1020→
  1021→        return $this->taskSuccess($id, __('td.toast.detailsUpdated'));
  1022→    }
