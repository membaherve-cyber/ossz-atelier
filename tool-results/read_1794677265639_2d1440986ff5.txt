    88→function useDebounced<T>(value: T, ms = 300) {
    89→  const [debounced, setDebounced] = useState(value);
    90→  useEffect(() => {
    91→    const t = setTimeout(() => setDebounced(value), ms);
    92→    return () => clearTimeout(t);
    93→  }, [value, ms]);
    94→  return debounced;
    95→}

    97→function toInputDate(iso: string | null): string {
    98→  return iso ? iso.slice(0, 10) : "";
    99→}
   100→
   101→function isPast(iso: string | null): boolean {
   102→  return !!iso && new Date(iso).getTime() < Date.now();
   103→}

   110→  // Hoisted because the task list below shadows `t` with its map parameter.
   111→  const unassignedLabel = t("orders.unassigned");

   120→  const debouncedSearch = useDebounced(search);
   121→
   122→  const [customers, setCustomers] = useState<Customer[]>([]);
   123→  const [formOpen, setFormOpen] = useState(false);
   124→  const [editing, setEditing] = useState<OrderSummary | null>(null);
   125→  const [form, setForm] = useState<OrderForm>(EMPTY_FORM);
   126→  const [formError, setFormError] = useState<string | null>(null);
   127→  const [submitting, setSubmitting] = useState(false);
   128→

   130→  const [detailLoading, setDetailLoading] = useState(false);

   136→        `/api/atelier/orders${qs({ search: debouncedSearch, priority, from, to })}`
   137→      );
   138→      setOrders(res.orders);
   139→      setError(null);
   140→    } catch (e) {
   141→      setError(e instanceof Error ? e.message : t("orders.loadFail"));
   142→    }
   143→  }, [debouncedSearch, priority, from, to, t]);
   144→
   145→  const loadCustomers = useCallback(async () => {
   146→    try {
   147→      const res = await api.get<{ customers: Customer[] }>("/api/atelier/customers");
   148→      setCustomers(res.customers);
   149→    } catch {
   150→      // Non-blocking: the new-customer mode remains available.
   151→    }
   152→  }, []);
   153→
   154→  useEffect(() => {
   155→    load();
   156→  }, [load]);
   157→
   158→  useEffect(() => {
   159→    loadCustomers();

   161→
   162→  const openDetail = useCallback(async (id: string) => {
   163→    setDetailId(id);
   164→    setDetail(null);

   166→    try {
   167→      const res = await api.get<OrderDetail>(`/api/atelier/orders/${id}`);
   168→      setDetail(res);
   169→    } catch (e) {
   170→      toast({
   171→        title: t("orders.loadOrderFailed"),
   172→        description: e instanceof Error ? e.message : undefined,
   173→        variant: "destructive",
   174→      });
   175→      setDetailId(null);
   176→    } finally {
   177→      setDetailLoading(false);
   178→    }
   179→  }, [toast, t]);
   180→
   181→  function openCreate() {
   182→    setEditing(null);
   183→    setForm({
   184→      ...EMPTY_FORM,
   185→      customerMode: customers.length > 0 ? "existing" : "new",
   186→      customerId: "",
   187→    });
   188→    setFormError(null);
   189→    setFormOpen(true);
   190→  }
   191→
   192→  function openEdit(o: OrderSummary) {
   193→    setEditing(o);
   194→    setForm({
   195→      orderNumber: o.orderNumber,
   196→      customerMode: "existing",
   197→      customerId: o.customerId,
   198→      ncName: "",

   201→      ncAddress: "",
   202→      ncNotes: "",
   203→      orderDate: toInputDate(o.orderDate),
   204→      expectedDeliveryDate: toInputDate(o.expectedDeliveryDate),
   205→      description: o.description,
   206→      garmentCategory: o.garmentCategory,
   207→      numberOfPieces: String(o.numberOfPieces),
   208→      priority: o.priority,
   209→      instructions: o.instructions ?? "",
   210→      notes: o.notes ?? "",
   211→    });
   212→    setFormError(null);
   213→    setDetail(null);
   214→    setFormOpen(true);
   215→  }
   216→
   217→  function buildPayload(): Record<string, unknown> | string {
   218→    if (!form.orderNumber.trim()) return t("orders.errOrderNumber");
   219→    if (!form.orderDate) return t("orders.errOrderDate");
   220→    if (!form.description.trim()) return t("orders.errDescription");
   221→    if (!form.garmentCategory.trim()) return t("orders.errCategory");
   222→    const pieces = Number.parseInt(form.numberOfPieces, 10);
   223→    if (!Number.isFinite(pieces) || pieces < 1 || pieces > 200) {
   224→      return t("orders.errPieces");
   225→    }
   226→    let customer: Record<string, unknown> | undefined;
   227→    if (!editing) {
   228→      if (form.customerMode === "existing") {
   229→        if (!form.customerId) return t("orders.errCustomerSelect");
   230→      } else {
   231→        if (!form.ncName.trim() || !form.ncPhone.trim()) {
   232→          return t("orders.errNewCustomer");
   233→        }
   234→        customer = {
   235→          name: form.ncName,
   236→          phone: form.ncPhone,
   237→          ...(form.ncEmail ? { email: form.ncEmail } : {}),
   238→          ...(form.ncAddress ? { address: form.ncAddress } : {}),
   239→          ...(form.ncNotes ? { notes: form.ncNotes } : {}),
   240→        };
   241→      }
   242→    }
   243→    return {
   244→      orderNumber: form.orderNumber,
   245→      ...(editing ? {} : form.customerMode === "existing" ? { customerId: form.customerId } : { newCustomer: customer }),
   246→      orderDate: form.orderDate,
   247→      expectedDeliveryDate: form.expectedDeliveryDate || null,
   248→      description: form.description,
   249→      garmentCategory: form.garmentCategory,
   250→      numberOfPieces: pieces,
   251→      priority: form.priority,
   252→      ...(form.instructions ? { instructions: form.instructions } : {}),
   253→      ...(form.notes ? { notes: form.notes } : {}),
   254→    };
   255→  }

   257→  async function submit() {
   258→    const payload = buildPayload();
   259→    if (typeof payload === "string") {
   260→      setFormError(payload);
   261→      return;
   262→    }
   263→    setSubmitting(true);
   264→    setFormError(null);
   265→    try {
   266→      if (editing) {
   267→        await api.patch(`/api/atelier/orders/${editing.id}`, payload);
   268→        toast({
   269→          title: t("orders.updatedToast"),
   270→          description: t("orders.updatedToastDesc", { number: form.orderNumber }),
   271→        });
   272→      } else {
   273→        await api.post("/api/atelier/orders", payload);
   274→        toast({
   275→          title: t("orders.registeredToast"),
   276→          description: t("orders.registeredToastDesc", { number: form.orderNumber }),
   277→        });
   278→      }
   279→      setFormOpen(false);
   280→      await load();

   284→        e instanceof ApiClientError
   285→          ? e.details?.issues?.[0]?.message ?? e.message
   286→          : t("common.somethingWrong");
   287→      setFormError(message);
   288→      toast({
   289→        title: editing ? t("orders.updateFailed") : t("orders.registerFailed"),
   290→        description: message,
   291→        variant: "destructive",
   292→      });
   293→    } finally {
   294→      setSubmitting(false);
   295→    }
   296→  }
   297→
   298→  const hasFilters = !!search || !!priority || !!from || !!to;
   299→
   300→  return (
   301→    <div>

   303→        title={t("orders.title")}
   304→        description={t("orders.subtitle")}

   307→          <Button onClick={openCreate}>
   308→            <Plus className="size-4 mr-1.5" />
   309→            {t("orders.register")}
   310→          </Button>
   311→        )}
   312→      </PageHeader>
   313→
   314→      <div className="mb-4 flex flex-col sm:flex-row gap-2 sm:items-center">

   316→          <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   317→          <Input
   318→            value={search}
   319→            onChange={(e) => setSearch(e.target.value)}
   320→            placeholder={t("orders.searchPlaceholder")}
   321→            className="pl-8"
   322→            aria-label={t("orders.searchLabel")}
   323→          />
   324→        </div>
   325→        <div className="flex flex-wrap gap-2 items-center">
   326→          <Select value={priority || "all"} onValueChange={(v) => setPriority(v === "all" ? "" : v)}>
   327→            <SelectTrigger className="w-[150px]" aria-label={t("orders.filterPriorityLabel")}>
   328→              <SelectValue placeholder={t("orders.priorityPlaceholder")} />
   329→            </SelectTrigger>
   330→            <SelectContent>
   331→              <SelectItem value="all">{t("orders.allPriorities")}</SelectItem>
   332→              {PRIORITIES.map((p) => (
   333→                <SelectItem key={p} value={p}>
   334→                  {priorityLabel(p, lang)}
   335→                </SelectItem>
   336→              ))}
   337→            </SelectContent>
   338→          </Select>
   339→          <div className="flex items-center gap-1.5">
   340→            <Input
   341→              type="date"
   342→              value={from}
   343→              onChange={(e) => setFrom(e.target.value)}
   344→              className="w-[145px]"
   345→              aria-label={t("orders.fromDateLabel")}
   346→            />
   347→            <span className="text-muted-foreground text-sm">{t("orders.toLabel")}</span>
   348→            <Input
   349→              type="date"
   350→              value={to}
   351→              onChange={(e) => setTo(e.target.value)}
   352→              className="w-[145px]"
   353→              aria-label={t("orders.toDateLabel")}
   354→            />
   355→          </div>
   356→          {hasFilters && (
   357→            <Button
   358→              variant="ghost"
   359→              size="sm"
   360→              onClick={() => {
   361→                setSearch("");
   362→                setPriority("");
   363→                setFrom("");
   364→                setTo("");
   365→              }}
   366→            >
   367→              {t("orders.clear")}
   368→            </Button>
   369→          )}
   370→        </div>
   371→      </div>
   372→
   373→      {error ? (
   374→        <EmptyState
   375→          icon={AlertTriangle}
   376→          title={t("orders.loadFailedTitle")}
   377→          description={error}
   378→          action={
   379→            <Button variant="outline" size="sm" onClick={load}>
   380→              {t("common.tryAgain")}
   381→            </Button>
   382→          }
   383→        />
   384→      ) : orders === null ? (
   385→        <div className="space-y-3">
   386→          <Skeleton className="h-10 w-full" />
   387→          <Skeleton className="h-16 w-full" />
   388→          <Skeleton className="h-16 w-full" />
   389→          <Skeleton className="h-16 w-full" />
   390→        </div>
   391→      ) : orders.length === 0 ? (
   392→        <div className="rounded-lg border">
   393→          <EmptyState
   394→            icon={Package}
   395→            title={hasFilters ? t("orders.emptyFiltersTitle") : t("orders.emptyTitle")}
   396→            description={
   397→              hasFilters
   398→                ? t("orders.emptyFiltersDesc")
   399→                : t("orders.emptyDesc")
   400→            }
   401→            action={
   402→              canManage && !hasFilters ? (
   403→                <Button size="sm" onClick={openCreate}>
   404→                  <Plus className="size-4 mr-1.5" />
   405→                  {t("orders.register")}
   406→                </Button>
   407→              ) : undefined
   408→            }
   409→          />
   410→        </div>
   411→      ) : (
   412→        <div className="rounded-lg border overflow-x-auto">
   413→          <Table className="min-w-[980px]">
   414→            <TableHeader>
   415→              <TableRow>
   416→                <TableHead>{t("orders.colOrder")}</TableHead>
   417→                <TableHead>{t("orders.colCustomer")}</TableHead>
   418→                <TableHead>{t("orders.colGarment")}</TableHead>
   419→                <TableHead className="text-center">{t("orders.colPieces")}</TableHead>
   420→                <TableHead>{t("orders.colPriority")}</TableHead>
   421→                <TableHead>{t("orders.colDelivery")}</TableHead>
   422→                <TableHead className="min-w-52">{t("orders.colProduction")}</TableHead>
   423→                <TableHead>{t("orders.colRegistered")}</TableHead>
   424→              </TableRow>
   425→            </TableHeader>
   426→            <TableBody>
   427→              {orders.map((o) => (
   428→                <TableRow
   429→                  key={o.id}
   430→                  className="cursor-pointer"
   431→                  onClick={() => openDetail(o.id)}
   432→                >
   433→                  <TableCell className="font-medium whitespace-nowrap">{o.orderNumber}</TableCell>
   434→                  <TableCell>
   435→                    <p className="font-medium truncate">{o.customerName}</p>
   436→                    <p className="text-xs text-muted-foreground truncate">{o.customerPhone}</p>
   437→                  </TableCell>
   438→                  <TableCell className="max-w-56">
   439→                    <p className="truncate">{o.description}</p>
   440→                    <p className="text-xs text-muted-foreground truncate">{o.garmentCategory}</p>
   441→                  </TableCell>
   442→                  <TableCell className="text-center tabular-nums">{o.numberOfPieces}</TableCell>
   443→                  <TableCell>
   444→                    <PriorityBadge priority={o.priority} />
   445→                  </TableCell>
   446→                  <TableCell>
   447→                    <DueDate deadline={o.expectedDeliveryDate} overdue={isPast(o.expectedDeliveryDate)} />
   448→                  </TableCell>
   449→                  <TableCell>
   450→                    <ProgressCell value={o.avgProgress} />
   451→                    <p className="text-xs text-muted-foreground mt-1">

   456→                            total: o.taskCount,
   457→                          })}
   458→                    </p>
   459→                  </TableCell>
   460→                  <TableCell className="text-muted-foreground whitespace-nowrap">
   461→                    {formatDate(o.createdAt)}
   462→                  </TableCell>
   463→                </TableRow>
   464→              ))}
   465→            </TableBody>
   466→          </Table>
   467→        </div>
   468→      )}
   469→
   470→      {/* ------------------------------------------------ order detail dialog */}
   471→      <Dialog
   472→        open={!!detailId}
   473→        onOpenChange={(open) => {
   474→          if (!open) setDetailId(null);
   475→        }}
   476→      >
   477→        <DialogContent className="sm:max-w-2xl max-h-[85vh] overflow-y-auto">
   478→          {detailLoading || !detail ? (
   479→            <div className="space-y-3 py-2">
   480→              <Skeleton className="h-7 w-56" />
   481→              <Skeleton className="h-24 w-full" />
   482→              <Skeleton className="h-32 w-full" />
   483→            </div>
   484→          ) : (
   485→            <>
   486→              <DialogHeader>
   487→                <DialogTitle className="flex items-center gap-2 flex-wrap">
   488→                  <span>{detail.order.orderNumber}</span>
   489→                  <PriorityBadge priority={detail.order.priority} />

   491→                <DialogDescription>
   492→                  {t("orders.detailMeta", {

   495→                  })}
   496→                </DialogDescription>
   497→              </DialogHeader>
   498→
   499→              <div className="grid grid-cols-2 sm:grid-cols-4 gap-3 text-sm">
   500→                <div>
   501→                  <p className="text-xs text-muted-foreground uppercase tracking-wide">{t("orders.statOrderDate")}</p>
   502→                  <p className="font-medium mt-0.5">{formatDate(detail.order.orderDate)}</p>
   503→                </div>
   504→                <div>
   505→                  <p className="text-xs text-muted-foreground uppercase tracking-wide">{t("orders.statDelivery")}</p>
   506→                  <div className="mt-0.5">
   507→                    <DueDate
   508→                      deadline={detail.order.expectedDeliveryDate}
   509→                      overdue={isPast(detail.order.expectedDeliveryDate)}
   510→                    />
   511→                  </div>
   512→                </div>
   513→                <div>
   514→                  <p className="text-xs text-muted-foreground uppercase tracking-wide">{t("orders.statPieces")}</p>
   515→                  <p className="font-medium mt-0.5 tabular-nums">{detail.order.numberOfPieces}</p>
   516→                </div>
   517→                <div>
   518→                  <p className="text-xs text-muted-foreground uppercase tracking-wide">{t("orders.statCategory")}</p>
   519→                  <p className="font-medium mt-0.5">{detail.order.garmentCategory}</p>
   520→                </div>
   521→              </div>
   522→
   523→              <div>
   524→                <p className="text-xs text-muted-foreground uppercase tracking-wide mb-1">{t("orders.descriptionLabel")}</p>
   525→                <p className="text-sm">{detail.order.description}</p>
   526→              </div>
   527→              {detail.order.instructions && (
   528→                <div>
   529→                  <p className="text-xs text-muted-foreground uppercase tracking-wide mb-1">{t("orders.instructionsLabel")}</p>
   530→                  <p className="text-sm whitespace-pre-line">{detail.order.instructions}</p>
   531→                </div>
   532→              )}
   533→              {detail.order.notes && (
   534→                <div>
   535→                  <p className="text-xs text-muted-foreground uppercase tracking-wide mb-1">{t("orders.notesLabel")}</p>
   536→                  <p className="text-sm whitespace-pre-line text-muted-foreground">{detail.order.notes}</p>
   537→                </div>
   538→              )}
   539→
   540→              <Separator />
   541→
   542→              <div>
   543→                <div className="flex items-center justify-between gap-2 mb-2">
   544→                  <p className="text-sm font-semibold">{t("orders.productionTasks")}</p>
   545→                  <span className="text-xs text-muted-foreground">
   546→                    {t(
   547→                      detail.order.taskCount > 1
   548→                        ? "orders.summaryDonePlural"
   549→                        : "orders.summaryDone",
   550→                      {
   551→                        done: detail.order.completedTaskCount,
   552→                        total: detail.order.taskCount,
   553→                        progress: detail.order.avgProgress,
   554→                      }
   555→                    )}
   556→                  </span>
   557→                </div>
   558→                {detail.tasks.length === 0 ? (
   559→                  <EmptyState
   560→                    icon={ClipboardList}
   561→                    title={t("orders.noTasksTitle")}
   562→                    description={t("orders.noTasksDesc")}
   563→                  />
   564→                ) : (
   565→                  <div className="rounded-lg border divide-y overflow-x-auto">
   566→                    {detail.tasks.map((t) => (
   567→                      <div
   568→                        key={t.id}
   569→                        className="flex flex-col sm:flex-row sm:items-center gap-2 sm:gap-4 px-4 py-2.5 min-w-[520px]"
   570→                      >
   571→                        <div className="min-w-0 flex-1">
   572→                          <p className="text-sm font-medium truncate">
