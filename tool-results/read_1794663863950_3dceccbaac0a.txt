    54→function StatusBadge({ status }: { status: string }) {
    55→  const styles: Record<string, string> = {
    56→    PENDING: "bg-amber-50 text-amber-800 border-amber-300",
    57→    APPROVED: "bg-emerald-50 text-emerald-800 border-emerald-300",
    58→    REJECTED: "bg-red-50 text-red-800 border-red-300",
    59→  };
    60→  const labels: Record<string, string> = {
    61→    PENDING: "Pending",
    62→    APPROVED: "Approved",
    63→    REJECTED: "Rejected",
    64→  };
    65→  return (
    66→    <Badge variant="outline" className={cn("whitespace-nowrap", styles[status] ?? "")}>
    67→      {labels[status] ?? status}
    68→    </Badge>
    69→  );
    70→}
    71→
    72→function PaymentBadge({ record }: { record: IncentiveRecordRow }) {
    73→  if (record.paymentStatus === "PAID") {
    74→    return (
    75→      <div className="flex flex-col gap-0.5">
    76→        <Badge variant="outline" className="bg-emerald-50 text-emerald-800 border-emerald-300 whitespace-nowrap">
    77→          Paid
    78→        </Badge>
    79→        <span className="text-[11px] text-muted-foreground whitespace-nowrap">
    80→          {record.paymentDate ? formatDate(record.paymentDate) : ""}
    81→        </span>
    82→      </div>
    83→    );
    84→  }
    85→  return (
    86→    <Badge variant="outline" className="bg-stone-100 text-stone-600 border-stone-300 whitespace-nowrap">
    87→      Unpaid
    88→    </Badge>
    89→  );
    90→}

   109→  description: "",

   119→
   120→export function IncentivesView() {
   121→  const user = useAppStore((s) => s.user);
   122→  const { toast } = useToast();
   123→  const canManageSchemes = can(user, "incentives.manage.schemes");
   124→  const canPrepare = can(user, "incentives.prepare");
   125→  const canApprove = can(user, "incentives.approve");
   126→  const canPay = can(user, "incentives.payment");
   127→
   128→  const [schemes, setSchemes] = useState<IncentiveSchemeRecord[] | null>(null);
   129→  const [records, setRecords] = useState<IncentiveRecordRow[] | null>(null);
   130→  const [tab, setTab] = useState("schemes");
   131→
   132→  // --- scheme dialog ---
   133→  const [schemeDialogOpen, setSchemeDialogOpen] = useState(false);
   134→  const [editingScheme, setEditingScheme] = useState<IncentiveSchemeRecord | null>(null);
   135→  const [schemeForm, setSchemeForm] = useState<SchemeForm>(EMPTY_SCHEME);
   136→  const [savingScheme, setSavingScheme] = useState(false);
   137→  const [togglingId, setTogglingId] = useState<string | null>(null);
   138→
   139→  // --- generation toolbar ---
   140→  const [selectedSchemeId, setSelectedSchemeId] = useState("");
   141→  const [periodInput, setPeriodInput] = useState(
   142→    `${new Date().getUTCFullYear()}-${String(new Date().getUTCMonth() + 1).padStart(2, "0")}`
   143→  );
   144→  const [generating, setGenerating] = useState(false);
   145→
   146→  // --- review dialog ---
   147→  const [reviewing, setReviewing] = useState<IncentiveRecordRow | null>(null);
   148→  const [approveAmount, setApproveAmount] = useState("");
   149→  const [reviewComments, setReviewComments] = useState("");
   150→  const [submittingReview, setSubmittingReview] = useState(false);
   151→  const [payingId, setPayingId] = useState<string | null>(null);
   152→
   153→  const loadSchemes = useCallback(async () => {
   154→    try {
   155→      const res = await api.get<{ schemes: IncentiveSchemeRecord[] }>("/api/atelier/incentives/schemes");
   156→      setSchemes(res.schemes);
   157→      setSelectedSchemeId((prev) => prev || res.schemes.find((s) => s.isActive)?.id || "");
   158→    } catch {
   159→      setSchemes([]);
   160→    }
   161→  }, []);
   162→
   163→  const loadRecords = useCallback(async () => {
   164→    try {
   165→      const res = await api.get<{ records: IncentiveRecordRow[] }>("/api/atelier/incentives/records");
   166→      setRecords(res.records);
   167→    } catch {
   168→      setRecords([]);
   169→    }
   170→  }, []);
   171→
   172→  useEffect(() => {
   173→    void loadSchemes();
   174→  }, [loadSchemes]);
   175→
   176→  useEffect(() => {
   177→    void loadRecords();
   178→  }, [loadRecords]);
   179→
   180→  const selectedScheme = schemes?.find((s) => s.id === selectedSchemeId) ?? null;
   181→  const isQuarterly = selectedScheme?.periodType === "QUARTERLY";
   182→  const isAnnual = selectedScheme?.periodType === "ANNUAL";
   183→  const periodPlaceholder = isQuarterly ? "2026-Q4" : isAnnual ? "2026" : "2026-11";
   184→  const periodHint = isQuarterly
   185→    ? "Enter a quarter (YYYY-QN)"
   186→    : isAnnual
   187→      ? "Enter a year (YYYY)"
   188→      : "Enter a month (YYYY-MM)";
   189→
   190→  // ------------------------------------------------------------- schemes tab
   191→
   192→  function openCreateScheme() {
   193→    setEditingScheme(null);
   194→    setSchemeForm(EMPTY_SCHEME);
   195→    setSchemeDialogOpen(true);
   196→  }
   197→
   198→  function openEditScheme(scheme: IncentiveSchemeRecord) {
   199→    setEditingScheme(scheme);
   200→    setSchemeForm({
   201→      name: scheme.name,
   202→      periodType: (scheme.periodType as IncentivePeriodType) ?? "MONTHLY",
   203→      minScore: scheme.minScore == null ? "" : String(scheme.minScore),
   204→      amount: scheme.amount == null ? "" : String(scheme.amount),
   205→      description: scheme.description ?? "",
   206→    });
   207→    setSchemeDialogOpen(true);
   208→  }
   209→
   210→  async function saveScheme() {
   211→    const name = schemeForm.name.trim();
   212→    if (name.length < 2 || name.length > 120) {
   213→      toast({
   214→        title: "Invalid name",
   215→        description: "Scheme name must be between 2 and 120 characters.",
   216→        variant: "destructive",
   217→      });
   218→      return;
   219→    }
   220→    const minScore = schemeForm.minScore.trim() === "" ? null : Number(schemeForm.minScore);
   221→    if (minScore != null && (!Number.isFinite(minScore) || minScore < 0 || minScore > 100)) {
   222→      toast({
   223→        title: "Invalid minimum score",
   224→        description: "Minimum score must be between 0 and 100.",
   225→        variant: "destructive",
   226→      });
   227→      return;
   228→    }
   229→    const amount = schemeForm.amount.trim() === "" ? null : Number(schemeForm.amount);
   230→    if (amount != null && (!Number.isFinite(amount) || amount < 0)) {
   231→      toast({
   232→        title: "Invalid amount",
   233→        description: "Amount must be zero or more (FCFA).",
   234→        variant: "destructive",
   235→      });
   236→      return;
   237→    }
   238→    const description = schemeForm.description.trim();
   239→
   240→    setSavingScheme(true);
   241→    try {
   242→      if (editingScheme) {
   243→        await api.patch(`/api/atelier/incentives/schemes/${editingScheme.id}`, {
   244→          name,
   245→          periodType: schemeForm.periodType,
   246→          minScore,
   247→          amount,
   248→          description: description || null,
   249→        });
   250→        toast({ title: "Scheme updated", description: `${name} was saved.` });
   251→      } else {
   252→        await api.post("/api/atelier/incentives/schemes", {
   253→          name,
   254→          periodType: schemeForm.periodType,
   255→          ...(minScore != null ? { minScore } : {}),
   256→          ...(amount != null ? { amount } : {}),
   257→          ...(description ? { description } : {}),
   258→        });
   259→        toast({ title: "Scheme created", description: `${name} was added.` });
   260→      }
   261→      setSchemeDialogOpen(false);
   262→      await loadSchemes();
   263→    } catch (e) {
   264→      toast({
   265→        title: "Could not save scheme",
   266→        description: e instanceof Error ? e.message : "Unexpected error.",
   267→        variant: "destructive",
   268→      });
   269→    } finally {
   270→      setSavingScheme(false);
   271→    }
   272→  }
   273→
   274→  async function toggleSchemeActive(scheme: IncentiveSchemeRecord, checked: boolean) {
   275→    setTogglingId(scheme.id);
   276→    try {
   277→      await api.patch(`/api/atelier/incentives/schemes/${scheme.id}`, { isActive: checked });
   278→      setSchemes((prev) =>
   279→        prev ? prev.map((s) => (s.id === scheme.id ? { ...s, isActive: checked } : s)) : prev
   280→      );
   281→      toast({
   282→        title: checked ? "Scheme activated" : "Scheme deactivated",
   283→        description: scheme.name,
   284→      });
   285→    } catch (e) {
   286→      toast({
   287→        title: "Could not update scheme",
   288→        description: e instanceof Error ? e.message : "Unexpected error.",
   289→        variant: "destructive",
   290→      });
   291→    } finally {
   292→      setTogglingId(null);
   293→    }
   294→  }
   295→
   296→  // --------------------------------------------------------- evaluations tab
   297→
   298→  async function generateRecommendations() {
   299→    if (!selectedSchemeId) {
   300→      toast({
   301→        title: "Select a scheme",
   302→        description: "Choose the incentive scheme to evaluate.",
   303→        variant: "destructive",
   304→      });
   305→      return;
   306→    }
   307→    const period = periodInput.trim();
   308→    if (!period) {
   309→      toast({
   310→        title: "Enter a period",
   311→        description: isQuarterly
   312→          ? "Expected format: YYYY-QN (e.g. 2026-Q4)."
   313→          : isAnnual
   314→            ? "Expected format: YYYY (e.g. 2026)."
   315→            : "Expected format: YYYY-MM (e.g. 2026-11).",
   316→        variant: "destructive",
   317→      });
   318→      return;
   319→    }
   320→    setGenerating(true);
   321→    try {
   322→      const res = await api.post<{ records: IncentiveRecordRow[] }>("/api/atelier/incentives/records", {
   323→        schemeId: selectedSchemeId,
   324→        period,
   325→      });
   326→      const eligibleCount = res.records.filter((r) => r.eligible).length;
   327→      toast({
   328→        title: "Recommendations generated",
   329→        description: `${res.records.length} evaluation(s) prepared — ${eligibleCount} eligible.`,
   330→      });
   331→      await loadRecords();
   332→    } catch (e) {
   333→      toast({
   334→        title: "Generation failed",
   335→        description: e instanceof Error ? e.message : "Unexpected error.",
   336→        variant: "destructive",
   337→      });
   338→    } finally {
   339→      setGenerating(false);
   340→    }
   341→  }
   342→
   343→  function openReview(record: IncentiveRecordRow) {
   344→    setReviewing(record);
   345→    setApproveAmount(String(record.proposedAmount));
   346→    setReviewComments("");
   347→  }
   348→
   349→  async function submitReview(status: "APPROVED" | "REJECTED") {
   350→    if (!reviewing) return;
   351→    let approvedAmount: number | undefined;
   352→    if (status === "APPROVED") {
   353→      const n = Number(approveAmount);
   354→      if (!Number.isFinite(n) || n < 0) {
   355→        toast({
   356→          title: "Invalid amount",
   357→          description: "Approved amount must be zero or more (FCFA).",
   358→          variant: "destructive",
   359→        });
   360→        return;
   361→      }
   362→      approvedAmount = n;
   363→    }
   364→    setSubmittingReview(true);
   365→    try {
   366→      const res = await api.patch<{ record: IncentiveRecordRow }>(
   367→        `/api/atelier/incentives/records/${reviewing.id}`,
   368→        {
   369→          action: "REVIEW",
   370→          status,
   371→          ...(approvedAmount !== undefined ? { approvedAmount } : {}),
   372→          ...(reviewComments.trim() ? { approvalComments: reviewComments.trim() } : {}),
   373→        }
   374→      );
   375→      setRecords((prev) =>
   376→        prev ? prev.map((r) => (r.id === res.record.id ? res.record : r)) : prev
   377→      );
   378→      toast({
   379→        title: status === "APPROVED" ? "Incentive approved" : "Incentive rejected",
   380→        description: `${res.record.staffName} — ${res.record.schemeName} (${res.record.period}).`,
   381→      });
   382→      setReviewing(null);
   383→    } catch (e) {
   384→      toast({
   385→        title: "Review failed",
   386→        description: e instanceof Error ? e.message : "Unexpected error.",
   387→        variant: "destructive",
   388→      });
   389→    } finally {
   390→      setSubmittingReview(false);
   391→    }
   392→  }
   393→
   394→  async function markPaid(record: IncentiveRecordRow) {
   395→    setPayingId(record.id);
   396→    try {
   397→      const res = await api.patch<{ record: IncentiveRecordRow }>(
   398→        `/api/atelier/incentives/records/${record.id}`,
   399→        { action: "PAYMENT", paymentStatus: "PAID" }
   400→      );
   401→      setRecords((prev) =>
   402→        prev ? prev.map((r) => (r.id === res.record.id ? res.record : r)) : prev
   403→      );
   404→      toast({
   405→        title: "Payment recorded",
   406→        description: `${record.staffName} — ${formatFCFA(res.record.approvedAmount)}.`,
   407→      });
   408→    } catch (e) {
   409→      toast({
   410→        title: "Could not record payment",
   411→        description: e instanceof Error ? e.message : "Unexpected error.",
   412→        variant: "destructive",
   413→      });
   414→    } finally {
   415→      setPayingId(null);
   416→    }
   417→  }
   418→
   419→  // ----------------------------------------------------------------- render
   420→
   421→  return (
   422→    <div>
   423→      <PageHeader
   424→        title="Staff Incentives"
   425→        description="Define bonus schemes, generate evaluation recommendations from performance data, approve payments and record payouts."
   426→      />
   427→
   428→      <Tabs value={tab} onValueChange={setTab}>
   429→        <TabsList className="mb-4">
   430→          <TabsTrigger value="schemes">
   431→            <Gift className="size-4 mr-1.5" />
   432→            Incentive Schemes
   433→          </TabsTrigger>
   434→          <TabsTrigger value="evaluations">
   435→            <Award className="size-4 mr-1.5" />
   436→            Evaluations & Approvals
   437→          </TabsTrigger>
   438→        </TabsList>
   439→
   440→        {/* ------------------------------------------------------ schemes tab */}
   441→        <TabsContent value="schemes">
   442→          <SectionCard
   443→            title="Incentive Schemes"
   444→            action={
   445→              canManageSchemes ? (
   446→                <Button size="sm" onClick={openCreateScheme}>
   447→                  <Plus className="size-4 mr-1.5" />
   448→                  New Scheme
   449→                </Button>
   450→              ) : undefined
   451→            }
   452→            contentClassName="p-0"
   453→          >
   454→            {schemes === null ? (
   455→              <div className="p-4 space-y-3">
   456→                {Array.from({ length: 3 }).map((_, i) => (
   457→                  <Skeleton key={i} className="h-12 w-full" />
   458→                ))}
   459→              </div>
   460→            ) : schemes.length === 0 ? (
   461→              <EmptyState
   462→                icon={Gift}
   463→                title="No incentive schemes yet"
   464→                description={
   465→                  canManageSchemes
   466→                    ? "Create your first scheme to start rewarding staff performance."
   467→                    : "Management has not defined any incentive scheme yet."
   468→                }
   469→                action={
   470→                  canManageSchemes ? (
   471→                    <Button size="sm" onClick={openCreateScheme}>
   472→                      <Plus className="size-4 mr-1.5" />
   473→                      New Scheme
   474→                    </Button>
   475→                  ) : undefined
   476→                }
   477→              />
   478→            ) : (
   479→              <div className="overflow-x-auto">
   480→                <Table className="min-w-[860px]">
   481→                  <TableHeader>
   482→                    <TableRow>
   483→                      <TableHead>Name</TableHead>
   484→                      <TableHead>Period Type</TableHead>
   485→                      <TableHead className="text-center">Min Score</TableHead>
   486→                      <TableHead className="text-right">Amount</TableHead>
   487→                      <TableHead className="text-center">Active</TableHead>
   488→                      <TableHead>Created By</TableHead>
   489→                      {canManageSchemes && <TableHead className="text-right">Actions</TableHead>}
   490→                    </TableRow>
   491→                  </TableHeader>
   492→                  <TableBody>
   493→                    {schemes.map((s) => (
   494→                      <TableRow key={s.id} className={cn(!s.isActive && "opacity-60")}>
   495→                        <TableCell>
   496→                          <p className="font-medium">{s.name}</p>
   497→                          {s.description && (
   498→                            <p className="text-xs text-muted-foreground max-w-md truncate">
   499→                              {s.description}
   500→                            </p>
   501→                          )}
   502→                        </TableCell>
   503→                        <TableCell>
   504→                          <Badge variant="secondary" className="whitespace-nowrap">
   505→                            {periodLabel(s.periodType)}
   506→                          </Badge>
   507→                        </TableCell>
   508→                        <TableCell className="text-center tabular-nums">
   509→                          {s.minScore == null ? <span className="text-muted-foreground">Any</span> : `≥ ${s.minScore}`}
   510→                        </TableCell>
   511→                        <TableCell className="text-right tabular-nums font-medium">
   512→                          {s.amount == null ? <span className="text-muted-foreground">—</span> : formatFCFA(s.amount)}
   513→                        </TableCell>
   514→                        <TableCell className="text-center">
   515→                          {canManageSchemes ? (
   516→                            <Switch
   517→                              checked={s.isActive}
   518→                              disabled={togglingId === s.id}
   519→                              onCheckedChange={(checked) => void toggleSchemeActive(s, checked)}
   520→                              aria-label={`Toggle ${s.name}`}
   521→                            />
   522→                          ) : (
   523→                            <Badge
   524→                              variant="outline"
   525→                              className={cn(
   526→                                "whitespace-nowrap",
   527→                                s.isActive
   528→                                  ? "bg-emerald-50 text-emerald-800 border-emerald-300"
   529→                                  : "bg-stone-100 text-stone-500 border-stone-300"
   530→                              )}
   531→                            >
   532→                              {s.isActive ? "Active" : "Inactive"}
   533→                            </Badge>
   534→                          )}
   535→                        </TableCell>
   536→                        <TableCell className="text-muted-foreground">{s.createdByName ?? "—"}</TableCell>
   537→                        {canManageSchemes && (
   538→                          <TableCell className="text-right">
   539→                            <Button
   540→                              variant="ghost"
   541→                              size="sm"
   542→                              onClick={() => openEditScheme(s)}
   543→                              aria-label={`Edit ${s.name}`}
   544→                            >
   545→                              <Pencil className="size-3.5 mr-1" />
   546→                              Edit
   547→                            </Button>
   548→                          </TableCell>
   549→                        )}
   550→                      </TableRow>
   551→                    ))}
   552→                  </TableBody>
   553→                </Table>
   554→              </div>
   555→            )}
   556→          </SectionCard>
   557→        </TabsContent>
   558→
   559→        {/* -------------------------------------------------- evaluations tab */}
   560→        <TabsContent value="evaluations">
   561→          <SectionCard title="Generate Recommendations" className="mb-4">
   562→            <div className="flex flex-col sm:flex-row sm:items-end gap-3">
   563→              <div className="space-y-1.5 flex-1 min-w-[200px]">
   564→                <Label htmlFor="incentive-scheme">Scheme</Label>
   565→                <Select value={selectedSchemeId} onValueChange={setSelectedSchemeId}>
   566→                  <SelectTrigger id="incentive-scheme" className="w-full">
   567→                    <SelectValue placeholder="Select a scheme" />
   568→                  </SelectTrigger>
   569→                  <SelectContent>
   570→                    {(schemes ?? []).filter((s) => s.isActive).map((s) => (
   571→                      <SelectItem key={s.id} value={s.id}>
   572→                        {s.name}
   573→                      </SelectItem>
   574→                    ))}
   575→                  </SelectContent>
   576→                </Select>
   577→              </div>
   578→              <div className="space-y-1.5">
   579→                <Label htmlFor="incentive-period">Period</Label>
   580→                <Input
   581→                  id="incentive-period"
   582→                  className="sm:w-[170px]"
   583→                  placeholder={periodPlaceholder}
   584→                  value={periodInput}
   585→                  onChange={(e) => setPeriodInput(e.target.value)}
   586→                />
   587→              </div>
   588→              {canPrepare && (
   589→                <Button onClick={() => void generateRecommendations()} disabled={generating || !selectedSchemeId}>
   590→                  <Sparkles className="size-4 mr-1.5" />
   591→                  {generating ? "Generating..." : "Generate Recommendations"}
   592→                </Button>
   593→              )}
   594→            </div>
   595→            <p className="text-xs text-muted-foreground mt-3">
   596→              {periodHint}
   597→              {selectedScheme ? ` — ${periodLabel(selectedScheme.periodType)}.` : "."} Scores are read
   598→              from the performance engine; approved or rejected records are never overwritten by regeneration.
   599→            </p>
   600→          </SectionCard>
   601→
   602→          <SectionCard title="Evaluation Records" contentClassName="p-0">
   603→            {records === null ? (
   604→              <div className="p-4 space-y-3">
   605→                {Array.from({ length: 3 }).map((_, i) => (
   606→                  <Skeleton key={i} className="h-12 w-full" />
   607→                ))}
   608→              </div>
   609→            ) : records.length === 0 ? (
   610→              <EmptyState
   611→                icon={Award}
   612→                title="No evaluations yet"
   613→                description="Generate recommendations for a scheme and period to create evaluation records."
   614→              />
   615→            ) : (
   616→              <div className="overflow-x-auto max-h-[520px] overflow-y-auto">
   617→                <Table className="min-w-[1100px]">
   618→                  <TableHeader>
   619→                    <TableRow>
   620→                      <TableHead>Staff</TableHead>
   621→                      <TableHead>Scheme</TableHead>
   622→                      <TableHead>Period</TableHead>
   623→                      <TableHead className="text-center">Score</TableHead>
   624→                      <TableHead className="text-center">Eligible</TableHead>
   625→                      <TableHead className="text-right">Proposed</TableHead>
   626→                      <TableHead>Status</TableHead>
   627→                      <TableHead className="text-right">Approved</TableHead>
   628→                      <TableHead>Approved By</TableHead>
   629→                      <TableHead>Payment</TableHead>
   630→                      {(canApprove || canPay) && <TableHead className="text-right">Actions</TableHead>}
   631→                    </TableRow>
   632→                  </TableHeader>
   633→                  <TableBody>
   634→                    {records.map((r) => (
   635→                      <TableRow key={r.id}>
   636→                        <TableCell className="font-medium">{r.staffName}</TableCell>
   637→                        <TableCell className="text-muted-foreground">{r.schemeName}</TableCell>
   638→                        <TableCell className="tabular-nums">{r.period}</TableCell>
   639→                        <TableCell className="text-center">
   640→                          <span className="font-semibold tabular-nums">{r.performanceScore}</span>
   641→                        </TableCell>
   642→                        <TableCell className="text-center">
   643→                          {r.eligible ? (
   644→                            <CheckCircle2 className="size-4 text-emerald-600 mx-auto" aria-label="Eligible" />
   645→                          ) : (
   646→                            <XCircle className="size-4 text-stone-400 mx-auto" aria-label="Not eligible" />
   647→                          )}
   648→                        </TableCell>
   649→                        <TableCell className="text-right tabular-nums">{formatFCFA(r.proposedAmount)}</TableCell>
   650→                        <TableCell>
   651→                          <StatusBadge status={r.status} />
   652→                        </TableCell>
   653→                        <TableCell className="text-right tabular-nums">
   654→                          {r.approvedAmount == null ? (
   655→                            <span className="text-muted-foreground">—</span>
   656→                          ) : (
   657→                            formatFCFA(r.approvedAmount)
   658→                          )}
   659→                        </TableCell>
   660→                        <TableCell className="text-muted-foreground">{r.approvedByName ?? "—"}</TableCell>
   661→                        <TableCell>
   662→                          <PaymentBadge record={r} />
   663→                        </TableCell>
   664→                        {(canApprove || canPay) && (
   665→                          <TableCell className="text-right">
   666→                            <div className="flex items-center justify-end gap-1.5">
   667→                              {canApprove && r.status === "PENDING" && (
   668→                                <Button variant="outline" size="sm" onClick={() => openReview(r)}>
   669→                                  Review
   670→                                </Button>
   671→                              )}
   672→                              {canPay && r.status === "APPROVED" && r.paymentStatus === "UNPAID" && (
   673→                                <Button
   674→                                  variant="outline"
   675→                                  size="sm"
   676→                                  disabled={payingId === r.id}
   677→                                  onClick={() => void markPaid(r)}
   678→                                >
   679→                                  <CheckCircle2 className="size-3.5 mr-1" />
   680→                                  Mark as Paid
   681→                                </Button>
   682→                              )}
   683→                              {!(canApprove && r.status === "PENDING") &&
   684→                                !(canPay && r.status === "APPROVED" && r.paymentStatus === "UNPAID") && (
   685→                                  <span className="text-muted-foreground text-sm">—</span>
   686→                                )}
   687→                            </div>
   688→                          </TableCell>
   689→                        )}
   690→                      </TableRow>
   691→                    ))}
   692→                  </TableBody>
   693→                </Table>
   694→              </div>
   695→            )}
   696→          </SectionCard>
   697→        </TabsContent>
   698→      </Tabs>
   699→
   700→      {/* ------------------------------------------------------ scheme dialog */}
   701→      <Dialog open={schemeDialogOpen} onOpenChange={setSchemeDialogOpen}>
   702→        <DialogContent>
   703→          <DialogHeader>
   704→            <DialogTitle>{editingScheme ? "Edit Scheme" : "New Incentive Scheme"}</DialogTitle>
   705→            <DialogDescription>
   706→              Define the evaluation rule. Amounts are in FCFA; minimum score is compared against the
   707→              staff performance score for the period.
   708→            </DialogDescription>
   709→          </DialogHeader>
   710→          <div className="space-y-4">
   711→            <div className="space-y-1.5">
   712→              <Label htmlFor="scheme-name">
   713→                Name <span className="text-red-600">*</span>
   714→              </Label>
   715→              <Input
   716→                id="scheme-name"
   717→                maxLength={120}
   718→                value={schemeForm.name}
   719→                onChange={(e) => setSchemeForm((f) => ({ ...f, name: e.target.value }))}
   720→                placeholder="e.g. Monthly Excellence Bonus"
   721→              />
   722→            </div>
   723→            <div className="space-y-1.5">
   724→              <Label>Period type</Label>
   725→              <Select
   726→                value={schemeForm.periodType}
   727→                onValueChange={(v) => setSchemeForm((f) => ({ ...f, periodType: v as IncentivePeriodType }))}
   728→              >
   729→                <SelectTrigger className="w-full">
   730→                  <SelectValue />
   731→                </SelectTrigger>
   732→                <SelectContent>
   733→                  {INCENTIVE_PERIOD_TYPES.map((t) => (
   734→                    <SelectItem key={t} value={t}>
   735→                      {INCENTIVE_PERIOD_LABELS[t]}
   736→                    </SelectItem>
   737→                  ))}
   738→                </SelectContent>
   739→              </Select>
   740→            </div>
   741→            <div className="grid grid-cols-2 gap-3">
   742→              <div className="space-y-1.5">
   743→                <Label htmlFor="scheme-min">Min score (0-100)</Label>
   744→                <Input
   745→                  id="scheme-min"
   746→                  type="number"
   747→                  min={0}
   748→                  max={100}
   749→                  value={schemeForm.minScore}
   750→                  onChange={(e) => setSchemeForm((f) => ({ ...f, minScore: e.target.value }))}
   751→                  placeholder="Optional"
   752→                />
   753→              </div>
   754→              <div className="space-y-1.5">
   755→                <Label htmlFor="scheme-amount">Amount (FCFA)</Label>
   756→                <Input
   757→                  id="scheme-amount"
   758→                  type="number"
   759→                  min={0}
   760→                  value={schemeForm.amount}
   761→                  onChange={(e) => setSchemeForm((f) => ({ ...f, amount: e.target.value }))}
   762→                  placeholder="Optional"
   763→                />
   764→              </div>
   765→            </div>
   766→            <div className="space-y-1.5">
   767→              <Label htmlFor="scheme-desc">Description</Label>
   768→              <Textarea
   769→                id="scheme-desc"
   770→                rows={3}
   771→                value={schemeForm.description}
   772→                onChange={(e) => setSchemeForm((f) => ({ ...f, description: e.target.value }))}
   773→                placeholder="What this scheme rewards."
   774→              />
   775→            </div>
   776→          </div>
   777→          <DialogFooter>
   778→            <Button variant="outline" onClick={() => setSchemeDialogOpen(false)} disabled={savingScheme}>
   779→              Cancel
   780→            </Button>
   781→            <Button onClick={() => void saveScheme()} disabled={savingScheme}>
   782→              <Save className="size-4 mr-1.5" />
   783→              {savingScheme ? "Saving..." : editingScheme ? "Save Changes" : "Create Scheme"}
   784→            </Button>
   785→          </DialogFooter>
   786→        </DialogContent>
   787→      </Dialog>
   788→
   789→      {/* ------------------------------------------------------ review dialog */}
   790→      <Dialog open={reviewing !== null} onOpenChange={(o) => !o && setReviewing(null)}>
   791→        <DialogContent>
   792→          <DialogHeader>
   793→            <DialogTitle>Review Incentive</DialogTitle>
   794→            <DialogDescription>
   795→              Approve or reject this incentive recommendation. The decision is final — regeneration
   796→              will not alter reviewed records.
   797→            </DialogDescription>
   798→          </DialogHeader>
   799→          {reviewing && (
   800→            <div className="space-y-4">
   801→              <div className="rounded-lg border bg-muted/40 p-3 text-sm space-y-1">
   802→                <p className="font-medium">
   803→                  {reviewing.staffName} — {reviewing.schemeName}
   804→                </p>
   805→                <p className="text-muted-foreground">
   806→                  Period {reviewing.period} · Score {reviewing.performanceScore} ·{" "}
   807→                  {reviewing.eligible ? "eligible" : "not eligible"}
   808→                </p>
   809→                <p>
   810→                  Proposed amount:{" "}
   811→                  <span className="font-semibold">{formatFCFA(reviewing.proposedAmount)}</span>
   812→                </p>
   813→              </div>
   814→              <div className="space-y-1.5">
   815→                <Label htmlFor="approve-amount">Approved amount (FCFA)</Label>
   816→                <Input
   817→                  id="approve-amount"
   818→                  type="number"
   819→                  min={0}
   820→                  value={approveAmount}
   821→                  onChange={(e) => setApproveAmount(e.target.value)}
   822→                />
   823→              </div>
   824→              <div className="space-y-1.5">
   825→                <Label htmlFor="review-comments">Comments</Label>
   826→                <Textarea
   827→                  id="review-comments"
   828→                  rows={3}
   829→                  value={reviewComments}
   830→                  onChange={(e) => setReviewComments(e.target.value)}
   831→                  placeholder="Optional note recorded with the decision."
   832→                />
   833→              </div>
   834→            </div>
   835→          )}
   836→          <DialogFooter className="gap-2 sm:gap-2">
   837→            <Button
   838→              variant="destructive"
   839→              disabled={submittingReview}
   840→              onClick={() => void submitReview("REJECTED")}
   841→            >
   842→              <XCircle className="size-4 mr-1.5" />
   843→              Reject
   844→            </Button>
   845→            <Button disabled={submittingReview} onClick={() => void submitReview("APPROVED")}>
   846→              <CheckCircle2 className="size-4 mr-1.5" />
   847→              Approve
   848→            </Button>
   849→          </DialogFooter>
   850→        </DialogContent>
   851→      </Dialog>
   852→    </div>
   853→  );
   854→}
