    24→/**
    25→ * Staff incentives (spec §3: GET /incentives + scheme management,
    26→ * recommendation generation, review and payment recording).
    27→ *
    28→ * 1:1 port of the Next.js reference implementation:
    29→ * - GET/POST src/app/api/atelier/incentives/schemes (+ [id]) — scheme
    30→ *   definitions. The schema has a SINGLE nullable `amount` column (FCFA);
    31→ *   schemes are never hard-deleted, only deactivated.
    32→ * - POST    src/app/api/atelier/incentives/records — recommendation
    33→ *   generation from the live performance engine (App\Services\
    34→ *   PerformanceService, parallel-built A1/R1 contract).
    35→ * - PATCH   src/app/api/atelier/incentives/records/[id] — review
    36→ *   (approve / reject, MANAGEMENT only via `perm:incentives.approve`)
    37→ *   and payment recording (`perm:incentives.payment`).
    38→ *
    39→ * Snapshot models: IncentiveRecord / IncentiveScheme keep denormalized
    40→ * scheme_name / staff_name / approved_by_name columns and define NO
    41→ * Eloquent relations (1:1 Prisma port), so this controller attaches the
    42→ * related models IN MEMORY — both access styles work in the views:
    43→ *   $record->user?->name / $record->scheme?->name /
    44→ *   $record->approvedBy?->name (null when the account / scheme was
    45→ *   deleted), or the never-null snapshots $record->staff_name /
    46→ *   $record->scheme_name / $record->approved_by_name; schemes expose
    47→ *   $scheme->recordsCount (also ->records_count).
    48→ *
    49→ * NO notifications are emitted for incentive decisions (report decision:
    50→ * incentives are reviewed inside the Management UI — toasts only).
    51→ *
    52→ * Deliberately does not extend a base Controller class, mirroring
    53→ * LoginController / LocaleController.
    54→ */

    60→    /**
    61→     * Scheme period type => performance-engine mapping (SCHEME_PERIOD_MAP
    62→     * port from the Next.js generate route): the range type handed to
    63→     * PerformanceService::forRange(), the accepted periodValue pattern and
    64→     * the toast key describing the expected format.
    65→     *
    66→     * MONTHLY / QUALITY / ON_TIME / CUSTOM are monthly formats, QUARTERLY
    67→     * expects YYYY-QN, ANNUAL expects YYYY.
    68→     *
    69→     * @var array<string, array{range: string, pattern: string, format: string}>
    70→     */
    71→    private const SCHEME_PERIOD_MAP = [
    72→        'MONTHLY' => ['range' => 'MONTH', 'pattern' => '/^\d{4}-(0[1-9]|1[0-2])$/', 'format' => 'formatMonth'],
    73→        'QUARTERLY' => ['range' => 'QUARTER', 'pattern' => '/^\d{4}-Q[1-4]$/', 'format' => 'formatQuarter'],
    74→        'ANNUAL' => ['range' => 'YEAR', 'pattern' => '/^\d{4}$/', 'format' => 'formatYear'],
    75→        'QUALITY' => ['range' => 'MONTH', 'pattern' => '/^\d{4}-(0[1-9]|1[0-2])$/', 'format' => 'formatMonth'],
    76→        'ON_TIME' => ['range' => 'MONTH', 'pattern' => '/^\d{4}-(0[1-9]|1[0-2])$/', 'format' => 'formatMonth'],
    77→        'CUSTOM' => ['range' => 'MONTH', 'pattern' => '/^\d{4}-(0[1-9]|1[0-2])$/', 'format' => 'formatMonth'],
    78→    ];

   104→        if ($filters['period'] !== '') {

   112→        $records = $recordsQuery
   113→            ->paginate(self::RECORDS_PER_PAGE)
   114→            ->withQueryString();

   116→        $this->hydrateRecords($records);

   118→        return view('incentives.index', [

   120→            'records' => $records,
   121→            'filters' => $filters,
   122→            'staffList' => User::query()
   123→                ->where('is_active', true)
   124→                ->where('role', 'STAFF')

   142→    public function storeScheme(Request $request): RedirectResponse
   143→    {
   144→        /** @var User $me */
   145→        $me = $request->user();
   146→
   147→        $data = $request->validate([
   148→            'name' => ['required', 'string', 'min:2', 'max:120'],
   149→            'periodType' => ['required', 'string', Rule::in(Atelier::INCENTIVE_PERIOD_TYPES)],
   150→            'minScore' => ['nullable', 'numeric', 'min:0', 'max:100'],
   151→            'amount' => ['nullable', 'numeric', 'min:0'],
   152→            'description' => ['nullable', 'string', 'max:2000'],
   153→            'isActive' => ['nullable', 'boolean'],
   154→        ]);
   155→
   156→        $description = trim((string) ($data['description'] ?? ''));
   157→
   158→        $scheme = IncentiveScheme::query()->create([
   159→            'name' => trim((string) $data['name']),
   160→            'period_type' => (string) $data['periodType'],
   161→            'min_score' => isset($data['minScore']) ? (float) $data['minScore'] : null,
   162→            'amount' => isset($data['amount']) ? (float) $data['amount'] : null,
   163→            'description' => $description !== '' ? $description : null,
   164→            'is_active' => array_key_exists('isActive', $request->all())
   165→                ? $request->boolean('isActive')
   166→                : true,
   167→            'created_by_id' => $me->id,
   168→            'created_by_name' => $me->name,
   169→        ]);
   170→
   171→        Audit::log($me, 'INCENTIVE_SCHEME_CREATED', 'INCENTIVE_SCHEME', $scheme->id, [
   172→            'name' => $scheme->name,
   173→            'periodType' => $scheme->period_type,
   174→            'minScore' => $scheme->min_score,
   175→            'amount' => $scheme->amount,
   176→        ]);
   177→
   178→        return redirect()
   179→            ->back()
   180→            ->with('success', __('inc.toast.schemeCreated'));

   183→    /**
   184→     * POST /incentives/schemes/{id} — update a scheme (same fields as
   185→     * storeScheme, all optional: only keys actually present in the request
   186→     * are applied — PATCH semantics; present-but-empty values clear the
   187→     * nullable columns, mirroring the reference route).
   188→     */
   189→    public function updateScheme(Request $request, string $id): RedirectResponse
   190→    {
   191→        /** @var User $me */
   192→        $me = $request->user();
   193→
   194→        $data = $request->validate([
   195→            'name' => ['sometimes', 'string', 'min:2', 'max:120'],
   196→            'periodType' => ['sometimes', 'string', Rule::in(Atelier::INCENTIVE_PERIOD_TYPES)],
   197→            'minScore' => ['sometimes', 'nullable', 'numeric', 'min:0', 'max:100'],
   198→            'amount' => ['sometimes', 'nullable', 'numeric', 'min:0'],
   199→            'description' => ['sometimes', 'nullable', 'string', 'max:2000'],
   200→            'isActive' => ['sometimes', 'boolean'],
   201→        ]);
   202→
   203→        $scheme = IncentiveScheme::query()->findOrFail($id);
   204→
   205→        $input = $request->all();
   206→        $has = static fn (string $key): bool => array_key_exists($key, $input);
   207→
   208→        $update = [];
   209→        $changes = [];
   210→
   211→        if ($has('name')) {
   212→            $update['name'] = trim((string) ($data['name'] ?? ''));
   213→            $changes['name'] = $update['name'];
   214→        }
   215→
   216→        if ($has('periodType')) {
   217→            $update['period_type'] = (string) $data['periodType'];
   218→            $changes['periodType'] = $update['period_type'];
   219→        }
   220→
   221→        if ($has('minScore')) {
   222→            $update['min_score'] = $data['minScore'] !== null ? (float) $data['minScore'] : null;
   223→            $changes['minScore'] = $update['min_score'];
   224→        }
   225→
   226→        if ($has('amount')) {
   227→            $update['amount'] = $data['amount'] !== null ? (float) $data['amount'] : null;
   228→            $changes['amount'] = $update['amount'];
   229→        }
   230→
   231→        if ($has('description')) {
   232→            $description = trim((string) ($data['description'] ?? ''));
   233→            $update['description'] = $description !== '' ? $description : null;
   234→            $changes['description'] = $update['description'];
   235→        }
   236→
   237→        if ($has('isActive')) {
   238→            $update['is_active'] = $request->boolean('isActive');
   239→            $changes['isActive'] = $update['is_active'];
   240→        }
   241→
   242→        if ($update !== []) {
   243→            $scheme->forceFill($update)->save();
   244→        }
   245→
   246→        Audit::log($me, 'INCENTIVE_SCHEME_UPDATED', 'INCENTIVE_SCHEME', $scheme->id, [
   247→            'name' => $scheme->name,
   248→            'updated' => $changes,
   249→        ]);
   250→
   251→        return redirect()
   252→            ->back()
   253→            ->with('success', __('inc.toast.schemeUpdated'));
   254→    }

   256→    /**
   257→     * POST /incentives/generate — create PENDING recommendations from the
   258→     * performance engine (route perm incentives.prepare).
   259→     *
   260→     * Input contract: periodType (INCENTIVE_PERIOD_TYPES) + periodValue
   261→     * (required, format depends on the period type: MONTHLY / QUALITY /
   262→     * ON_TIME / CUSTOM = YYYY-MM, QUARTERLY = YYYY-QN, ANNUAL = YYYY);
   263→     * OPTIONAL schemeId narrows the run to a single scheme, otherwise
   264→     * every ACTIVE scheme of the posted period type is evaluated against
   265→     * every active STAFF account.
   266→     *
   267→     * Below-minScore staff are skipped; existing (scheme, staff, period)
   268→     * records are skipped silently (decisions are final — the reference
   269→     * route never touches APPROVED / REJECTED rows, and this port keeps
   270→     * PENDING rows immutable too). Created rows: eligible = true,
   271→     * proposed_amount = scheme amount (all created rows are eligible),
   272→     * status PENDING, payment_status UNPAID.
   273→     */
   274→    public function generate(Request $request): RedirectResponse
   275→    {
   276→        /** @var User $me */
   277→        $me = $request->user();
   278→
   279→        $data = $request->validate([
   280→            'periodType' => ['required', 'string', Rule::in(Atelier::INCENTIVE_PERIOD_TYPES)],
   281→            'periodValue' => ['required', 'string', 'max:20'],
   282→            'schemeId' => ['nullable', 'string'],
   283→        ]);
   284→
   285→        $periodType = (string) $data['periodType'];
   286→        $periodValue = strtoupper(trim((string) $data['periodValue']));
   287→        $format = self::SCHEME_PERIOD_MAP[$periodType];
   288→
   289→        if ($periodValue === '' || preg_match($format['pattern'], $periodValue) !== 1) {
   290→            return redirect()
   291→                ->back()
   292→                ->with('error', __('inc.toast.'.$format['format']));
   293→        }
   294→
   295→        // Target schemes: every ACTIVE scheme of the posted period type,
   296→        // or the single scheme named by the optional schemeId input.
   297→        $schemeId = trim((string) ($data['schemeId'] ?? ''));
   298→
   299→        $schemes = IncentiveScheme::query()
   300→            ->where('is_active', true)
   301→            ->when($schemeId !== '', fn ($query) => $query->where('id', $schemeId))
   302→            ->when($schemeId === '', fn ($query) => $query->where('period_type', $periodType))
   303→            ->orderBy('name')
   304→            ->get();
   305→
   306→        if ($schemes->isEmpty()) {
   307→            return redirect()
   308→                ->back()
   309→                ->with('error', __('inc.toast.selectSchemeDesc'));
   310→        }
   311→
   312→        $activeStaff = User::query()
   313→            ->where('is_active', true)
   314→            ->where('role', 'STAFF')
   315→            ->orderBy('name')
   316→            ->get();
   317→
   318→        // Existing (scheme, staff, period) rows — skipped silently.
   319→        $existing = IncentiveRecord::query()
   320→            ->where('period', $periodValue)
   321→            ->whereIn('scheme_id', $schemes->pluck('id'))
   322→            ->get(['scheme_id', 'staff_id']);
   323→
   324→        $existingKeys = [];
   325→
   326→        foreach ($existing as $row) {
   327→            $existingKeys[$row->scheme_id.'|'.$row->staff_id] = true;
   328→        }
   329→
   330→        $created = 0;
   331→
   332→        try {
   333→            // Performance results cached per range type (MONTH | QUARTER |
   334→            // YEAR — SCHEME_PERIOD_MAP port); the service is owned by the
   335→            // parallel R1 agent and called through its documented API.
   336→            $performance = [];
   337→
   338→            foreach ($schemes as $scheme) {
   339→                $rangeType = self::SCHEME_PERIOD_MAP[$scheme->period_type]['range'] ?? 'MONTH';
   340→
   341→                if (! array_key_exists($rangeType, $performance)) {
   342→                    $performance[$rangeType] = $this->performanceScores(
   343→                        PerformanceService::forRange($rangeType, $periodValue)
   344→                    );
   345→                }
   346→
   347→                $scores = $performance[$rangeType];
   348→                $field = $this->schemeScoreField($scheme->period_type);
   349→
   350→                foreach ($activeStaff as $staff) {
   351→                    $score = $scores[$staff->id][$field] ?? null;
   352→
   353→                    if ($score === null
   354→                        || ($scheme->min_score !== null && $score < $scheme->min_score)) {
   355→                        continue; // no score, or below the scheme threshold
   356→                    }
   357→
   358→                    if (isset($existingKeys[$scheme->id.'|'.$staff->id])) {
   359→                        continue; // already generated for (scheme, staff, period)
   360→                    }
   361→
   362→                    IncentiveRecord::query()->create([
   363→                        'scheme_id' => $scheme->id,
   364→                        'scheme_name' => $scheme->name,
   365→                        'staff_id' => $staff->id,
   366→                        'staff_name' => $staff->name,
   367→                        'period' => $periodValue,
   368→                        'performance_score' => $score,
   369→                        'eligible' => true,
   370→                        'proposed_amount' => $scheme->amount ?? 0,
   371→                        'status' => 'PENDING',
   372→                        'payment_status' => 'UNPAID',
   373→                    ]);
   374→
   375→                    $existingKeys[$scheme->id.'|'.$staff->id] = true;
   376→                    $created++;
   377→                }
   378→            }
   379→        } catch (Throwable) {
   380→            return redirect()
   381→                ->back()
   382→                ->with('error', __('inc.toast.generationFailed'));
   383→        }
   384→
   385→        Audit::log(
   386→            $me,
   387→            'INCENTIVE_RECOMMENDATIONS_GENERATED',
   388→            'INCENTIVE_SCHEME',
   389→            $schemes->count() === 1 ? $schemes->first()->id : null,
   390→            [
   391→                'created' => $created,
   392→                'period' => $periodValue,
   393→                'periodType' => $periodType,
   394→                'schemes' => $schemes->pluck('name')->all(),
   395→            ]
   396→        );
   397→
   398→        return redirect()
   399→            ->back()
   400→            ->with('success', Trans::fill('inc.toast.generatedDesc', [
   401→                'total' => $created,
   402→                'eligible' => $created,
   403→            ]));
   404→    }

   406→    /**
   407→     * POST /incentives/{id}/review — approve or reject a recommendation
   408→     * (route perm incentives.approve — MANAGEMENT only).
   409→     *
   410→     * Input contract: decision (approve|reject). approve requires
   411→     * approvedAmount (numeric > 0); reject requires rejectionReason
   412→     * (min 3, stored in management_comments — rejection is a Management
   413→     * decision). Optional comments / approvalComments are stored in
   414→     * approval_comments. Only PENDING records are reviewable; the
   415→     * decision is final (regeneration never touches reviewed rows).
   416→     */
   417→    public function review(Request $request, string $id): RedirectResponse
   418→    {
   419→        /** @var User $me */
   420→        $me = $request->user();
   421→
   422→        $base = $request->validate([
   423→            'decision' => ['required', 'string', Rule::in(['approve', 'reject'])],
   424→        ]);
   425→
   426→        $decision = (string) $base['decision'];
   427→
   428→        // Decision-specific requirements (validation errors, not flashes).
   429→        $data = $request->validate($decision === 'approve' ? [
   430→            'approvedAmount' => ['required', 'numeric', 'gt:0'],
   431→            'comments' => ['nullable', 'string', 'max:2000'],
   432→            'approvalComments' => ['nullable', 'string', 'max:2000'],
   433→        ] : [
   434→            'rejectionReason' => ['required', 'string', 'min:3', 'max:2000'],
   435→            'comments' => ['nullable', 'string', 'max:2000'],
   436→            'approvalComments' => ['nullable', 'string', 'max:2000'],
   437→        ]);
   438→
   439→        $record = IncentiveRecord::query()->findOrFail($id);
   440→
   441→        if ($record->status !== 'PENDING') {
   442→            return redirect()
   443→                ->back()
   444→                ->with('error', __('inc.toast.reviewFailed'));
   445→        }
   446→
   447→        $note = trim((string) ($data['approvalComments'] ?? $data['comments'] ?? ''));
   448→
   449→        if ($decision === 'approve') {
   450→            $amount = (float) $data['approvedAmount'];
   451→
   452→            $record->forceFill([
   453→                'status' => 'APPROVED',
   454→                'approved_amount' => $amount,
   455→                'approved_by_id' => $me->id,
   456→                'approved_by_name' => $me->name,
   457→                'approval_comments' => $note !== '' ? $note : null,
   458→            ])->save();
   459→
   460→            Audit::log($me, 'INCENTIVE_APPROVED', 'INCENTIVE_RECORD', $record->id, [
   461→                'recordId' => $record->id,
   462→                'amount' => $amount,
   463→                'staffName' => $record->staff_name,
   464→                'period' => $record->period,
   465→            ]);
   466→
   467→            return redirect()
   468→                ->back()
   469→                ->with('success', __('inc.toast.approvedTitle'));
   470→        }
   471→
   472→        $record->forceFill([
   473→            'status' => 'REJECTED',
   474→            'approved_by_id' => $me->id,
   475→            'approved_by_name' => $me->name,
   476→            'management_comments' => trim((string) $data['rejectionReason']),
   477→            'approval_comments' => $note !== '' ? $note : null,
   478→        ])->save();
   479→
   480→        Audit::log($me, 'INCENTIVE_REJECTED', 'INCENTIVE_RECORD', $record->id, [
   481→            'recordId' => $record->id,
   482→            'amount' => $record->proposed_amount,
   483→            'staffName' => $record->staff_name,
   484→            'period' => $record->period,
   485→        ]);
   486→
   487→        return redirect()
   488→            ->back()
   489→            ->with('success', __('inc.toast.rejectedTitle'));
   490→    }

   492→    /**
   493→     * POST /incentives/{id}/payment — record a payout (route perm
   494→     * incentives.payment — MANAGEMENT only).
   495→     *
   496→     * Input contract: paidAt (optional date, defaults to now) sets
   497→     * payment_status PAID + the payment_date column. Only APPROVED and
   498→     * still-unpaid records can be paid.
   499→     */
   500→    public function payment(Request $request, string $id): RedirectResponse
   501→    {
   502→        /** @var User $me */
   503→        $me = $request->user();
   504→
   505→        $data = $request->validate([
   506→            'paidAt' => ['nullable', 'date'],
   507→        ]);
   508→
   509→        $record = IncentiveRecord::query()->findOrFail($id);
   510→
   511→        if ($record->status !== 'APPROVED' || $record->payment_status === 'PAID') {
   512→            return redirect()
   513→                ->back()
   514→                ->with('error', __('inc.toast.paymentFailed'));
   515→        }
   516→
   517→        $paidAt = isset($data['paidAt']) && $data['paidAt'] !== null
   518→            ? Carbon::parse((string) $data['paidAt'])
   519→            : now();
   520→
   521→        $record->forceFill([
   522→            'payment_status' => 'PAID',
   523→            'payment_date' => $paidAt,
   524→        ])->save();
   525→
   526→        Audit::log($me, 'INCENTIVE_PAYMENT_RECORDED', 'INCENTIVE_RECORD', $record->id, [
   527→            'recordId' => $record->id,
   528→            'staffName' => $record->staff_name,
   529→            'amount' => $record->approved_amount,
   530→            'paymentDate' => $paidAt->format(DATE_ATOM),
   531→        ]);
   532→
   533→        return redirect()
   534→            ->back()
   535→            ->with('success', __('inc.toast.paymentRecorded'));
   536→    }

   542→    /**
   543→     * All schemes, newest first, each carrying its evaluation count as an
   544→     * in-memory recordsCount / records_count attribute (IncentiveScheme
   545→     * has no Eloquent relations — the count cannot be eager-loaded).
   546→     */
   547→    private function schemesWithRecordCounts(): Collection
   548→    {
   549→        $schemes = IncentiveScheme::query()->orderByDesc('created_at')->get();
   550→
   551→        $counts = IncentiveRecord::query()
   552→            ->whereIn('scheme_id', $schemes->pluck('id'))
   553→            ->groupBy('scheme_id')
   554→            ->selectRaw('scheme_id, count(*) as aggregate')
   555→            ->pluck('aggregate', 'scheme_id');
   556→
   557→        return $schemes->each(static function (IncentiveScheme $scheme) use ($counts): void {
   558→            $count = (int) ($counts[$scheme->id] ?? 0);
   559→
   560→            $scheme->setAttribute('recordsCount', $count);
   561→            $scheme->setAttribute('records_count', $count);
   562→        });
   563→    }
   564→
   565→    /**
   566→     * Attach the related User / IncentiveScheme models to each record row
   567→     * IN MEMORY (IncentiveRecord has no Eloquent relations): ->user,
   568→     * ->scheme and ->approvedBy, null when the account / scheme was
   569→     * deleted — the snapshots (staff_name, scheme_name,
   570→     * approved_by_name) always exist.
   571→     */
   572→    private function hydrateRecords(LengthAwarePaginator $records): void
   573→    {
   574→        $rows = $records->getCollection();
   575→
   576→        $staffIds = $rows->pluck('staff_id')
   577→            ->filter(static fn ($id) => (string) $id !== '')
   578→            ->unique()
   579→            ->values();
   580→
   581→        $approverIds = $rows->pluck('approved_by_id')
   582→            ->filter(static fn ($id) => (string) $id !== '')
   583→            ->unique()
   584→            ->values();
   585→
   586→        $schemeIds = $rows->pluck('scheme_id')
   587→            ->filter(static fn ($id) => (string) $id !== '')
   588→            ->unique()
   589→            ->values();
   590→
   591→        $userIds = $staffIds->merge($approverIds)->unique()->values();
   592→
   593→        $users = $userIds->isEmpty()
   594→            ? Collection::make()
   595→            : User::query()->whereIn('id', $userIds)->get()->keyBy('id');
   596→
   597→        $schemes = $schemeIds->isEmpty()
   598→            ? Collection::make()
   599→            : IncentiveScheme::query()->whereIn('id', $schemeIds)->get()->keyBy('id');
   600→
   601→        $rows->each(static function (IncentiveRecord $record) use ($users, $schemes): void {
   602→            $record->setAttribute('user', $users->get($record->staff_id));
   603→            $record->setAttribute('scheme', $schemes->get($record->scheme_id));
   604→            $record->setAttribute('approvedBy', $users->get($record->approved_by_id));
   605→        });
   606→    }

   608→    // ------------------------------------------------------------------
   609→    // PerformanceService row parsing (defensive)
   610→    // ------------------------------------------------------------------
   611→
   612→    /**
   613→     * Normalize a PerformanceService result into a plain list of row
   614→     * arrays. Defensive per the A3 contract: accepts a plain list of rows
   615→     * OR a ['rows' => …] / ['staff' => …] envelope, arrays or Traversable.
   616→     *
   617→     * @return list<array<string, mixed>>
   618→     */
   619→    private function performanceRows(mixed $result): array
   620→    {
   621→        if ($result instanceof Traversable) {
   622→            $result = iterator_to_array($result);
   623→        }
   624→
   625→        if (! is_array($result)) {
   626→            return [];
   627→        }
   628→
   629→        foreach (['rows', 'staff'] as $key) {
   630→            if (isset($result[$key]) && is_array($result[$key])) {
   631→                return array_values($result[$key]);
   632→            }
   633→        }
   634→
   635→        return array_values($result);
   636→    }
   637→
   638→    /**
   639→     * Staff id of a performance row — read from the explicit staffId /
   640→     * staff_id / userId / user_id keys (Next.js StaffPerformance shape)
   641→     * or from the nested User model ('user' => User, the PerformanceService
   642→     * row shape).
   643→     *
   644→     * @param  array<string, mixed>  $row
   645→     */
   646→    private function rowStaffId(array $row): ?string
   647→    {
   648→        foreach (['staffId', 'staff_id', 'userId', 'user_id'] as $key) {
   649→            if (isset($row[$key]) && (string) $row[$key] !== '') {
   650→                return (string) $row[$key];
   651→            }
   652→        }
   653→
   654→        $user = $row['user'] ?? null;
   655→
   656→        return $user instanceof User ? (string) $user->id : null;
   657→    }
   658→
   659→    /**
   660→     * One numeric score of a performance row, read defensively: nested
   661→     * scores.<field> first (Next.js shape), then the flat <field> key
   662→     * (PerformanceService shape), then the overallScore / overall_score
   663→     * fallbacks for the overall score.
   664→     *
   665→     * @param  array<string, mixed>  $row
   666→     */
   667→    private function numericField(array $row, string $field): ?float
   668→    {
   669→        $scores = $row['scores'] ?? null;
   670→
   671→        if (is_array($scores) && isset($scores[$field]) && is_numeric($scores[$field])) {
   672→            return (float) $scores[$field];
   673→        }
   674→
   675→        if (isset($row[$field]) && is_numeric($row[$field])) {
   676→            return (float) $row[$field];
   677→        }
   678→
   679→        if ($field === 'overall') {
   680→            foreach (['overallScore', 'overall_score'] as $key) {
   681→                if (isset($row[$key]) && is_numeric($row[$key])) {
   682→                    return (float) $row[$key];
   683→                }
   684→            }
   685→        }
   686→
   687→        return null;
   688→    }
   689→
   690→    /**
   691→     * staffId => per-field performance scores (quality / timeliness /
   692→     * overall) for one PerformanceService result.
   693→     *
   694→     * @return array<string, array{quality: ?float, timeliness: ?float, overall: ?float}>
   695→     */
   696→    private function performanceScores(mixed $result): array
   697→    {
   698→        $scores = [];
   699→
   700→        foreach ($this->performanceRows($result) as $row) {
   701→            $staffId = $this->rowStaffId($row);
   702→
   703→            if ($staffId === null) {
   704→                continue;
   705→            }
   706→
   707→            $scores[$staffId] = [
   708→                'quality' => $this->numericField($row, 'quality'),
   709→                'timeliness' => $this->numericField($row, 'timeliness'),
   710→                'overall' => $this->numericField($row, 'overall'),
   711→            ];
   712→        }
   713→
   714→        return $scores;
   715→    }
   716→
   717→    /**
   718→     * Performance field a scheme scores against (Next.js parity): QUALITY
   719→     * schemes evaluate the quality score, ON_TIME the timeliness score,
   720→     * everything else the overall score.
   721→     */
   722→    private function schemeScoreField(string $periodType): string
   723→    {
   724→        if ($periodType === 'QUALITY') {
   725→            return 'quality';
   726→        }
   727→
   728→        if ($periodType === 'ON_TIME') {
   729→            return 'timeliness';
   730→        }
   731→
   732→        return 'overall';
   733→    }
   734→}
