    30→    30→export function monthNow(): string {
    31→    31→  const now = new Date();
    32→    32→  return `${now.getUTCFullYear()}-${String(now.getUTCMonth() + 1).padStart(2, "0")}`;
    33→    33→}
    34→    34→
    35→    35→export function yearNow(): string {
    36→    36→  return String(new Date().getUTCFullYear());
    37→    37→}
    38→    38→
    39→    39→function quarterNow(): string {
    40→    40→  const now = new Date();
    41→    41→  return `${now.getUTCFullYear()}-Q${Math.floor(now.getUTCMonth() / 3) + 1}`;
    42→    42→}
    43→    43→
    44→    44→function shiftMonth(value: string, delta: number): string {
    45→    45→  const m = /^(\d{4})-(\d{2})$/.exec(value);
    46→    46→  if (!m) return value;
    47→    47→  const d = new Date(Date.UTC(Number(m[1]), Number(m[2]) - 1 + delta, 1));
    48→    48→  return `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, "0")}`;
    49→    49→}
    50→    50→
    51→    51→function shiftQuarter(value: string, delta: number): string {
    52→    52→  const m = /^(\d{4})-Q([1-4])$/.exec(value);
    53→    53→  if (!m) return value;
    54→    54→  const total = Number(m[1]) * 4 + (Number(m[2]) - 1) + delta;
    55→    55→  const year = Math.floor(total / 4);
    56→    56→  return `${year}-Q${total - year * 4 + 1}`;
    57→    57→}
    58→    58→
    59→    59→function shiftYear(value: string, delta: number): string {
    60→    60→  const m = /^\d{4}$/.exec(value);
    61→    61→  if (!m) return value;
    62→    62→  return String(Number(m[1]) + delta);
    63→    63→}

    75→    75→function csvCell(value: string | number | null | undefined): string {
    76→    76→  const s = value == null ? "" : String(value);

    78→    78→}

    80→    80→export function downloadCsv(
    81→    81→  filename: string,
    82→    82→  rows: Array<Array<string | number | null | undefined>>
    83→    83→): void {
    84→    84→  const csv = rows.map((r) => r.map(csvCell).join(",")).join("\n");
    85→    85→  const blob = new Blob([`\uFEFF${csv}`], { type: "text/csv;charset=utf-8" });
    86→    86→  const url = URL.createObjectURL(blob);
    87→    87→  const link = document.createElement("a");
    88→    88→  link.href = url;
    89→    89→  link.download = filename;
    90→    90→  document.body.appendChild(link);
    91→    91→  link.click();
    92→    92→  link.remove();
    93→    93→  URL.revokeObjectURL(url);
    94→    94→}
