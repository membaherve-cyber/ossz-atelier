    88→    88→function useDebounced<T>(value: T, ms = 300) {
    89→    89→  const [debounced, setDebounced] = useState(value);
    90→    90→  useEffect(() => {
    91→    91→    const t = setTimeout(() => setDebounced(value), ms);
    92→    92→    return () => clearTimeout(t);
    93→    93→  }, [value, ms]);
    94→    94→  return debounced;
    95→    95→}

    97→    97→function toInputDate(iso: string | null): string {

   101→   101→function isPast(iso: string | null): boolean {
   102→   102→  return !!iso && new Date(iso).getTime() < Date.now();
   103→   103→}

   227→   227→    if (!editing) {

   229→   229→        if (!form.customerId) return t("orders.errCustomerSelect");

   231→   231→        if (!form.ncName.trim() || !form.ncPhone.trim()) {
   232→   232→          return t("orders.errNewCustomer");

   234→   234→        customer = {
   235→   235→          name: form.ncName,
   236→   236→          phone: form.ncPhone,
   237→   237→          ...(form.ncEmail ? { email: form.ncEmail } : {}),
   238→   238→          ...(form.ncAddress ? { address: form.ncAddress } : {}),
   239→   239→          ...(form.ncNotes ? { notes: form.ncNotes } : {}),
   240→   240→        };
   241→   241→      }
   242→   242→    }
   243→   243→    return {
   244→   244→      orderNumber: form.orderNumber,
   245→   245→      ...(editing ? {} : form.customerMode === "existing" ? { customerId: form.customerId } : { newCustomer: customer }),
   246→   246→      orderDate: form.orderDate,
   247→   247→      expectedDeliveryDate: form.expectedDeliveryDate || null,
   248→   248→      description: form.description,
   249→   249→      garmentCategory: form.garmentCategory,
   250→   250→      numberOfPieces: pieces,
   251→   251→      priority: form.priority,
   252→   252→      ...(form.instructions ? { instructions: form.instructions } : {}),
   253→   253→      ...(form.notes ? { notes: form.notes } : {}),
   254→   254→    };
   255→   255→  }
