    86→    86→function useDebounced<T>(value: T, ms = 300) {
    87→    87→  const [debounced, setDebounced] = useState(value);
    88→    88→  useEffect(() => {
    89→    89→    const t = setTimeout(() => setDebounced(value), ms);
    90→    90→    return () => clearTimeout(t);
    91→    91→  }, [value, ms]);
    92→    92→  return debounced;
    93→    93→}

    95→    95→function toInputDate(iso: string | null): string {
    96→    96→  return iso ? iso.slice(0, 10) : "";

    99→    99→function isPast(iso: string | null): boolean {
   100→   100→  return !!iso && new Date(iso).getTime() < Date.now();
   101→   101→}

   239→   239→      orderNumber: form.orderNumber,
   240→   240→      ...(editing ? {} : form.customerMode === "existing" ? { customerId: form.customerId } : { newCustomer: customer }),
   241→   241→      orderDate: form.orderDate,

   243→   243→      description: form.description,

   246→   246→      priority: form.priority,

   248→   248→      ...(form.notes ? { notes: form.notes } : {}),
