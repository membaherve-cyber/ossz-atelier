    55→function StatusBadge({ status }: { status: string }) {
    56→  const { lang } = useI18n();
    57→  const styles: Record<string, string> = {
    58→    PENDING: "bg-amber-50 text-amber-800 border-amber-300",
    59→    APPROVED: "bg-emerald-50 text-emerald-800 border-emerald-300",
    60→    REJECTED: "bg-red-50 text-red-800 border-red-300",
    61→  };
    62→  return (
    63→    <Badge variant="outline" className={cn("whitespace-nowrap", styles[status] ?? "")}>
    64→      {incentiveStatusLabel(status, lang)}
    65→    </Badge>
    66→  );
    67→}
    68→
    69→function PaymentBadge({ record }: { record: IncentiveRecordRow }) {
    70→  const { lang } = useI18n();
    71→  if (record.paymentStatus === "PAID") {
    72→    return (
    73→      <div className="flex flex-col gap-0.5">
    74→        <Badge variant="outline" className="bg-emerald-50 text-emerald-800 border-emerald-300 whitespace-nowrap">
    75→          {paymentStatusLabel(record.paymentStatus, lang)}
    76→        </Badge>
    77→        <span className="text-[11px] text-muted-foreground whitespace-nowrap">
    78→          {record.paymentDate ? formatDate(record.paymentDate) : ""}
    79→        </span>
    80→      </div>
    81→    );
    82→  }
    83→  return (
    84→    <Badge variant="outline" className="bg-stone-100 text-stone-600 border-stone-300 whitespace-nowrap">
    85→      {paymentStatusLabel(record.paymentStatus, lang)}
    86→    </Badge>
    87→  );
    88→}

   114→export function IncentivesView() {
   115→  const user = useAppStore((s) => s.user);
   116→  const { toast } = useToast();
   117→  const { t, lang } = useI18n();
   118→  const canManageSchemes = can(user, "incentives.manage.schemes");
   119→  const canPrepare = can(user, "incentives.prepare");
   120→  const canApprove = can(user, "incentives.approve");
   121→  const canPay = can(user, "incentives.payment");
   122→
   123→  const [schemes, setSchemes] = useState<IncentiveSchemeRecord[] | null>(null);
   124→  const [records, setRecords] = useState<IncentiveRecordRow[] | null>(null);
   125→  const [tab, setTab] = useState("schemes");
   126→
   127→  // --- scheme dialog ---
   128→  const [schemeDialogOpen, setSchemeDialogOpen] = useState(false);
   129→  const [editingScheme, setEditingScheme] = useState<IncentiveSchemeRecord | null>(null);
   130→  const [schemeForm, setSchemeForm] = useState<SchemeForm>(EMPTY_SCHEME);
   131→  const [savingScheme, setSavingScheme] = useState(false);
   132→  const [togglingId, setTogglingId] = useState<string | null>(null);
   133→
   134→  // --- generation toolbar ---
   135→  const [selectedSchemeId, setSelectedSchemeId] = useState("");
   136→  const [periodInput, setPeriodInput] = useState(
   137→    `${new Date().getUTCFullYear()}-${String(new Date().getUTCMonth() + 1).padStart(2, "0")}`
   138→  );
   139→  const [generating, setGenerating] = useState(false);
   140→
   141→  // --- review dialog ---
   142→  const [reviewing, setReviewing] = useState<IncentiveRecordRow | null>(null);
   143→  const [approveAmount, setApproveAmount] = useState("");
   144→  const [reviewComments, setReviewComments] = useState("");
   145→  const [submittingReview, setSubmittingReview] = useState(false);
   146→  const [payingId, setPayingId] = useState<string | null>(null);
   147→
   148→  const loadSchemes = useCallback(async () => {
   149→    try {
   150→      const res = await api.get<{ schemes: IncentiveSchemeRecord[] }>("/api/atelier/incentives/schemes");
   151→      setSchemes(res.schemes);
   152→      setSelectedSchemeId((prev) => prev || res.schemes.find((s) => s.isActive)?.id || "");
   153→    } catch {
   154→      setSchemes([]);
   155→    }
   156→  }, []);
   157→
   158→  const loadRecords = useCallback(async () => {
   159→    try {
   160→      const res = await api.get<{ records: IncentiveRecordRow[] }>("/api/atelier/incentives/records");
   161→      setRecords(res.records);
   162→    } catch {
   163→      setRecords([]);
   164→    }
   165→  }, []);
   166→
   167→  useEffect(() => {
   168→    void loadSchemes();
   169→  }, [loadSchemes]);
   170→
   171→  useEffect(() => {
   172→    void loadRecords();
   173→  }, [loadRecords]);
   174→
   175→  const selectedScheme = schemes?.find((s) => s.id === selectedSchemeId) ?? null;
   176→  const isQuarterly = selectedScheme?.periodType === "QUARTERLY";
   177→  const isAnnual = selectedScheme?.periodType === "ANNUAL";
   178→  const periodPlaceholder = isQuarterly ? "2026-Q4" : isAnnual ? "2026" : "2026-11";
   179→  const periodHint = isQuarterly
   180→    ? t("inc.gen.hintQuarter")
   181→    : isAnnual
   182→      ? t("inc.gen.hintYear")
   183→      : t("inc.gen.hintMonth");
   184→
   185→  // ------------------------------------------------------------- schemes tab
   186→
   187→  function openCreateScheme() {
   188→    setEditingScheme(null);
   189→    setSchemeForm(EMPTY_SCHEME);
   190→    setSchemeDialogOpen(true);
   191→  }
   192→
   193→  function openEditScheme(scheme: IncentiveSchemeRecord) {
   194→    setEditingScheme(scheme);
   195→    setSchemeForm({
   196→      name: scheme.name,
   197→      periodType: (scheme.periodType as IncentivePeriodType) ?? "MONTHLY",
   198→      minScore: scheme.minScore == null ? "" : String(scheme.minScore),
   199→      amount: scheme.amount == null ? "" : String(scheme.amount),
   200→      description: scheme.description ?? "",
   201→    });
   202→    setSchemeDialogOpen(true);
   203→  }
   204→
   205→  async function saveScheme() {
   206→    const name = schemeForm.name.trim();
   207→    if (name.length < 2 || name.length > 120) {
   208→      toast({
   209→        title: t("inc.toast.invalidNameTitle"),
   210→        description: t("inc.toast.invalidNameDesc"),
   211→        variant: "destructive",
   212→      });
   213→      return;
   214→    }
   215→    const minScore = schemeForm.minScore.trim() === "" ? null : Number(schemeForm.minScore);
   216→    if (minScore != null && (!Number.isFinite(minScore) || minScore < 0 || minScore > 100)) {
   217→      toast({
   218→        title: t("inc.toast.invalidScoreTitle"),
   219→        description: t("inc.toast.invalidScoreDesc"),
   220→        variant: "destructive",
   221→      });
   222→      return;
   223→    }
   224→    const amount = schemeForm.amount.trim() === "" ? null : Number(schemeForm.amount);
   225→    if (amount != null && (!Number.isFinite(amount) || amount < 0)) {
   226→      toast({
   227→        title: t("inc.toast.invalidAmountTitle"),
   228→        description: t("inc.toast.invalidAmountDesc"),
   229→        variant: "destructive",
   230→      });
   231→      return;
   232→    }
   233→    const description = schemeForm.description.trim();
   234→
   235→    setSavingScheme(true);
   236→    try {
   237→      if (editingScheme) {
   238→        await api.patch(`/api/atelier/incentives/schemes/${editingScheme.id}`, {
   239→          name,
   240→          periodType: schemeForm.periodType,
   241→          minScore,
   242→          amount,
   243→          description: description || null,
   244→        });
   245→        toast({ title: t("inc.toast.schemeUpdated"), description: t("inc.toast.schemeSavedDesc", { name }) });
   246→      } else {
   247→        await api.post("/api/atelier/incentives/schemes", {
   248→          name,
   249→          periodType: schemeForm.periodType,
   250→          ...(minScore != null ? { minScore } : {}),
   251→          ...(amount != null ? { amount } : {}),
   252→          ...(description ? { description } : {}),
   253→        });
   254→        toast({ title: t("inc.toast.schemeCreated"), description: t("inc.toast.schemeCreatedDesc", { name }) });
   255→      }
   256→      setSchemeDialogOpen(false);
   257→      await loadSchemes();
   258→    } catch (e) {
   259→      toast({
   260→        title: t("inc.toast.saveFailed"),
   261→        description: e instanceof Error ? e.message : t("inc.toast.unexpected"),
   262→        variant: "destructive",
   263→      });
   264→    } finally {
   265→      setSavingScheme(false);
   266→    }
   267→  }
   268→
   269→  async function toggleSchemeActive(scheme: IncentiveSchemeRecord, checked: boolean) {
   270→    setTogglingId(scheme.id);
   271→    try {
   272→      await api.patch(`/api/atelier/incentives/schemes/${scheme.id}`, { isActive: checked });
   273→      setSchemes((prev) =>
   274→        prev ? prev.map((s) => (s.id === scheme.id ? { ...s, isActive: checked } : s)) : prev
   275→      );
   276→      toast({
   277→        title: checked ? t("inc.toast.schemeActivated") : t("inc.toast.schemeDeactivated"),
   278→        description: scheme.name,
   279→      });
   280→    } catch (e) {
   281→      toast({
   282→        title: t("inc.toast.updateFailed"),
   283→        description: e instanceof Error ? e.message : t("inc.toast.unexpected"),
   284→        variant: "destructive",
   285→      });
   286→    } finally {
   287→      setTogglingId(null);
   288→    }
   289→  }
   290→
   291→  // --------------------------------------------------------- evaluations tab
   292→
   293→  async function generateRecommendations() {
   294→    if (!selectedSchemeId) {
   295→      toast({
   296→        title: t("inc.gen.selectScheme"),
   297→        description: t("inc.toast.selectSchemeDesc"),
   298→        variant: "destructive",
   299→      });
   300→      return;
   301→    }
   302→    const period = periodInput.trim();
   303→    if (!period) {
   304→      toast({
   305→        title: t("inc.toast.enterPeriod"),
   306→        description: isQuarterly
   307→          ? t("inc.toast.formatQuarter")
   308→          : isAnnual
   309→            ? t("inc.toast.formatYear")
   310→            : t("inc.toast.formatMonth"),
   311→        variant: "destructive",
   312→      });
   313→      return;
   314→    }
   315→    setGenerating(true);
   316→    try {
   317→      const res = await api.post<{ records: IncentiveRecordRow[] }>("/api/atelier/incentives/records", {
   318→        schemeId: selectedSchemeId,
   319→        period,
   320→      });
   321→      const eligibleCount = res.records.filter((r) => r.eligible).length;
   322→      toast({
   323→        title: t("inc.toast.generatedTitle"),
   324→        description: t("inc.toast.generatedDesc", { total: res.records.length, eligible: eligibleCount }),
   325→      });
   326→      await loadRecords();
   327→    } catch (e) {
   328→      toast({
   329→        title: t("inc.toast.generationFailed"),
   330→        description: e instanceof Error ? e.message : t("inc.toast.unexpected"),
   331→        variant: "destructive",
   332→      });
   333→    } finally {
   334→      setGenerating(false);
   335→    }
   336→  }
   337→
   338→  function openReview(record: IncentiveRecordRow) {
   339→    setReviewing(record);
   340→    setApproveAmount(String(record.proposedAmount));
   341→    setReviewComments("");
   342→  }
   343→
   344→  async function submitReview(status: "APPROVED" | "REJECTED") {
   345→    if (!reviewing) return;
   346→    let approvedAmount: number | undefined;
   347→    if (status === "APPROVED") {
   348→      const n = Number(approveAmount);
   349→      if (!Number.isFinite(n) || n < 0) {
   350→        toast({
   351→          title: t("inc.toast.invalidAmountTitle"),
   352→          description: t("inc.toast.invalidApprovedAmountDesc"),
   353→          variant: "destructive",
   354→        });
   355→        return;
   356→      }
   357→      approvedAmount = n;
   358→    }
   359→    setSubmittingReview(true);
   360→    try {
   361→      const res = await api.patch<{ record: IncentiveRecordRow }>(
   362→        `/api/atelier/incentives/records/${reviewing.id}`,
   363→        {
   364→          action: "REVIEW",
   365→          status,
   366→          ...(approvedAmount !== undefined ? { approvedAmount } : {}),
   367→          ...(reviewComments.trim() ? { approvalComments: reviewComments.trim() } : {}),
   368→        }
   369→      );
   370→      setRecords((prev) =>
   371→        prev ? prev.map((r) => (r.id === res.record.id ? res.record : r)) : prev
   372→      );
   373→      toast({
   374→        title: status === "APPROVED" ? t("inc.toast.approvedTitle") : t("inc.toast.rejectedTitle"),
   375→        description: t("inc.toast.reviewedDesc", {
   376→          staff: res.record.staffName,
   377→          scheme: res.record.schemeName,
   378→          period: res.record.period,
   379→        }),
   380→      });
   381→      setReviewing(null);
   382→    } catch (e) {
   383→      toast({
   384→        title: t("inc.toast.reviewFailed"),
   385→        description: e instanceof Error ? e.message : t("inc.toast.unexpected"),
   386→        variant: "destructive",
   387→      });
   388→    } finally {
   389→      setSubmittingReview(false);
   390→    }
   391→  }
   392→
   393→  async function markPaid(record: IncentiveRecordRow) {
   394→    setPayingId(record.id);
   395→    try {
   396→      const res = await api.patch<{ record: IncentiveRecordRow }>(
   397→        `/api/atelier/incentives/records/${record.id}`,
   398→        { action: "PAYMENT", paymentStatus: "PAID" }
   399→      );
   400→      setRecords((prev) =>
   401→        prev ? prev.map((r) => (r.id === res.record.id ? res.record : r)) : prev
   402→      );
   403→      toast({
   404→        title: t("inc.toast.paymentRecorded"),
   405→        description: t("inc.toast.paymentRecordedDesc", {
   406→          staff: record.staffName,
   407→          amount: formatFCFA(res.record.approvedAmount),
   408→        }),
   409→      });
   410→    } catch (e) {
   411→      toast({
   412→        title: t("inc.toast.paymentFailed"),
   413→        description: e instanceof Error ? e.message : t("inc.toast.unexpected"),
   414→        variant: "destructive",
   415→      });
   416→    } finally {
   417→      setPayingId(null);
   418→    }
   419→  }
   420→
   421→  // ----------------------------------------------------------------- render
   422→
   423→  return (
   424→    <div>
   425→      <PageHeader title={t("inc.title")} description={t("inc.subtitle")} />
   426→
   427→      <Tabs value={tab} onValueChange={setTab}>
   428→        <TabsList className="mb-4">
   429→          <TabsTrigger value="schemes">
   430→            <Gift className="size-4 mr-1.5" />
   431→            {t("inc.tab.schemes")}
   432→          </TabsTrigger>
   433→          <TabsTrigger value="evaluations">
   434→            <Award className="size-4 mr-1.5" />
   435→            {t("inc.tab.evaluations")}
   436→          </TabsTrigger>
   437→        </TabsList>
   438→
   439→        {/* ------------------------------------------------------ schemes tab */}
   440→        <TabsContent value="schemes">
   441→          <SectionCard
   442→            title={t("inc.schemes.title")}
   443→            action={
   444→              canManageSchemes ? (
   445→                <Button size="sm" onClick={openCreateScheme}>
   446→                  <Plus className="size-4 mr-1.5" />
   447→                  {t("inc.schemes.new")}
   448→                </Button>
   449→              ) : undefined
   450→            }
   451→            contentClassName="p-0"
   452→          >
   453→            {schemes === null ? (
   454→              <div className="p-4 space-y-3">
   455→                {Array.from({ length: 3 }).map((_, i) => (
   456→                  <Skeleton key={i} className="h-12 w-full" />
   457→                ))}
   458→              </div>
   459→            ) : schemes.length === 0 ? (
   460→              <EmptyState
   461→                icon={Gift}
   462→                title={t("inc.schemes.emptyTitle")}
   463→                description={
   464→                  canManageSchemes
   465→                    ? t("inc.schemes.emptyDescManage")
   466→                    : t("inc.schemes.emptyDescView")
   467→                }
   468→                action={
   469→                  canManageSchemes ? (
   470→                    <Button size="sm" onClick={openCreateScheme}>
   471→                      <Plus className="size-4 mr-1.5" />
   472→                      {t("inc.schemes.new")}
   473→                    </Button>
   474→                  ) : undefined
   475→                }
   476→              />
   477→            ) : (
   478→              <div className="overflow-x-auto">
   479→                <Table className="min-w-[860px]">
   480→                  <TableHeader>
   481→                    <TableRow>
   482→                      <TableHead>{t("inc.schemes.col.name")}</TableHead>
   483→                      <TableHead>{t("inc.schemes.col.periodType")}</TableHead>
   484→                      <TableHead className="text-center">{t("inc.schemes.col.minScore")}</TableHead>
   485→                      <TableHead className="text-right">{t("inc.schemes.col.amount")}</TableHead>
   486→                      <TableHead className="text-center">{t("inc.schemes.col.active")}</TableHead>
   487→                      <TableHead>{t("inc.schemes.col.createdBy")}</TableHead>
   488→                      {canManageSchemes && <TableHead className="text-right">{t("common.actions")}</TableHead>}
   489→                    </TableRow>
   490→                  </TableHeader>
   491→                  <TableBody>
   492→                    {schemes.map((s) => (
   493→                      <TableRow key={s.id} className={cn(!s.isActive && "opacity-60")}>
   494→                        <TableCell>
   495→                          <p className="font-medium">{s.name}</p>
   496→                          {s.description && (
   497→                            <p className="text-xs text-muted-foreground max-w-md truncate">
   498→                              {s.description}
   499→                            </p>
   500→                          )}
   501→                        </TableCell>
   502→                        <TableCell>
   503→                          <Badge variant="secondary" className="whitespace-nowrap">
   504→                            {incentivePeriodLabel(s.periodType, lang)}
   505→                          </Badge>
   506→                        </TableCell>
   507→                        <TableCell className="text-center tabular-nums">
   508→                          {s.minScore == null ? (
   509→                            <span className="text-muted-foreground">{t("inc.schemes.anyScore")}</span>
   510→                          ) : (
   511→                            `≥ ${s.minScore}`
   512→                          )}
   513→                        </TableCell>
   514→                        <TableCell className="text-right tabular-nums font-medium">
   515→                          {s.amount == null ? <span className="text-muted-foreground">—</span> : formatFCFA(s.amount)}
   516→                        </TableCell>
   517→                        <TableCell className="text-center">
   518→                          {canManageSchemes ? (
   519→                            <Switch
   520→                              checked={s.isActive}
   521→                              disabled={togglingId === s.id}
   522→                              onCheckedChange={(checked) => void toggleSchemeActive(s, checked)}
   523→                              aria-label={t("inc.schemes.toggleAria", { name: s.name })}
   524→                            />
   525→                          ) : (
   526→                            <Badge
   527→                              variant="outline"
   528→                              className={cn(
   529→                                "whitespace-nowrap",
   530→                                s.isActive
   531→                                  ? "bg-emerald-50 text-emerald-800 border-emerald-300"
   532→                                  : "bg-stone-100 text-stone-500 border-stone-300"
   533→                              )}
   534→                            >
   535→                              {s.isActive ? t("inc.schemes.active") : t("inc.schemes.inactive")}
   536→                            </Badge>
   537→                          )}
   538→                        </TableCell>
   539→                        <TableCell className="text-muted-foreground">{s.createdByName ?? "—"}</TableCell>
   540→                        {canManageSchemes && (
   541→                          <TableCell className="text-right">
   542→                            <Button
   543→                              variant="ghost"
   544→                              size="sm"
   545→                              onClick={() => openEditScheme(s)}
   546→                              aria-label={t("inc.schemes.editAria", { name: s.name })}
   547→                            >
   548→                              <Pencil className="size-3.5 mr-1" />
   549→                              {t("common.edit")}
   550→                            </Button>
   551→                          </TableCell>
   552→                        )}
   553→                      </TableRow>
   554→                    ))}
   555→                  </TableBody>
   556→                </Table>
   557→              </div>
   558→            )}
   559→          </SectionCard>
   560→        </TabsContent>
   561→
   562→        {/* -------------------------------------------------- evaluations tab */}
   563→        <TabsContent value="evaluations">
   564→          <SectionCard title={t("inc.gen.title")} className="mb-4">
