    45→ * (MANAGEMENT / ADMIN); STAFF users reaching another user's task get 403.
    46→ * Terminal tasks (DELIVERED / CANCELLED) reject every action, like the
    47→ * outer gate of the Next.js PATCH handler.

    60→    /**
    61→     * POST /tasks/{id}/assign — assign or reassign the task (route perm
    62→     * production.assign: MANAGEMENT / ADMIN only).
    63→     *
    64→     * Ported from the ASSIGN case of tasks/[id]/route.ts: the new status
    65→     * only moves UNASSIGNED -> ASSIGNED; the assignment history row is
    66→     * typed ASSIGN or REASSIGN; the new assignee is notified (message
    67→     * mirrors the reference implementation exactly — the old assignee is
    68→     * NOT notified, matching route.ts).
    69→     */
    70→    public function assign(Request $request, string $id): RedirectResponse
    71→    {
    72→        $me = $request->user();
    73→        $task = $this->loadTask($id);
    74→        $this->authorize($me, $task);
    75→
    76→        if ($task->isTerminal()) {
    77→            return $this->closedTaskError($task);
    78→        }
    79→
    80→        $data = $request->validate([
    81→            'staffId' => [
    82→                'required',
    83→                'string',
    84→                Rule::exists('users', 'id')->where('is_active', true),
    85→            ],
    86→            'note' => ['nullable', 'string', 'max:1000'],
    87→        ]);
    88→
    89→        $staff = User::query()->findOrFail($data['staffId']);
    90→
    91→        if (! in_array($staff->role, Permissions::assignableRoles(), true)) {
    92→            return $this->actionError($task, __('td.toast.selectStaff'));
    93→        }
    94→
    95→        if ($staff->id === $task->assigned_staff_id) {
    96→            return $this->actionError($task, __('td.toast.selectStaff'));
    97→        }
    98→
    99→        $reassign = $task->assigned_staff_id !== null;
   100→        $fromStatus = $task->status;
   101→        $newStatus = $fromStatus === 'UNASSIGNED' ? 'ASSIGNED' : $fromStatus;
   102→        $note = trim((string) ($data['note'] ?? ''));
   103→
   104→        DB::transaction(function () use ($task, $staff, $me, $reassign, $newStatus, $note): void {
   105→            $task->forceFill([
   106→                'assigned_staff_id' => $staff->id,
   107→                'status' => $newStatus,
   108→            ])->save();
   109→
   110→            TaskAssignment::query()->create([
   111→                'task_id' => $task->id,
   112→                'staff_id' => $staff->id,
   113→                'staff_name' => $staff->name,
   114→                'assigned_by_id' => $me->id,
   115→                'assigned_by_name' => $me->name,
   116→                'type' => $reassign ? 'REASSIGN' : 'ASSIGN',
   117→                'note' => $note !== '' ? $note : null,
   118→            ]);
   119→        });
   120→
   121→        if ($newStatus !== $fromStatus) {
   122→            $this->recordStatusChange($task, $fromStatus, $newStatus, $me, $note !== '' ? $note : null);
   123→        }
   124→
   125→        $message = $this->taskLabel($task).' has been '.($reassign ? 'reassigned' : 'assigned')
   126→            .' to you.'.($note !== '' ? ' Note: '.$note : '');
   127→
   128→        Notifier::users(
   129→            [$staff->id],
   130→            $reassign ? 'TASK_REASSIGNED' : 'TASK_ASSIGNED',
   131→            $message,
   132→            $task->id,
   133→            $task->order_id
   134→        );
   135→
   136→        Audit::log($me, $reassign ? 'TASK_REASSIGNED' : 'TASK_ASSIGNED', 'TASK', $task->id, [
   137→            'staff' => $staff->name,
   138→            'note' => $note !== '' ? $note : null,
   139→        ]);
   140→
   141→        return redirect()
   142→            ->route('tasks.show', $task->id)
   143→            ->with('success', Trans::fill($reassign ? 'td.toast.reassignedTo' : 'td.toast.assignedTo', ['name' => $staff->name]));
   144→    }

   146→    /**
   147→     * POST /tasks/{id}/status — workflow transitions driven by the hidden
   148→     * `action` field: start|submit|ready|deliver|reopen|cancel (view
   149→     * contract). Gates, side effects, notifications and audit codes are
   150→     * ported from the SUBMIT / READY / DELIVER / CANCEL cases of
   151→     * tasks/[id]/route.ts; "start" and "reopen" implement the contract
   152→     * additions (reopen mirrors the REOPENED inspection decision).
   153→     */
   154→    public function status(Request $request, string $id): RedirectResponse
   155→    {
   156→        $me = $request->user();
   157→        $task = $this->loadTask($id);
   158→        $this->authorize($me, $task);
   159→
   160→        if ($task->isTerminal()) {
   161→            return $this->closedTaskError($task);
   162→        }
   163→
   164→        $data = $request->validate([
   165→            'action' => ['required', 'string', Rule::in(['start', 'submit', 'ready', 'deliver', 'reopen', 'cancel'])],
   166→            'note' => ['nullable', 'string', 'max:1000'],
   167→        ]);
   168→
   169→        $action = (string) $data['action'];
   170→        $note = trim((string) ($data['note'] ?? ''));
   171→        $supervisor = $this->isSupervisor($me);
   172→
   173→        switch ($action) {
   174→            case 'start': {
   175→                // Contract: assignee or supervisor, from ASSIGNED or
   176→                // CHANGES_REQUESTED -> ONGOING.
   177→                if (! in_array($task->status, ['ASSIGNED', 'CHANGES_REQUESTED'], true)) {
   178→                    return $this->actionError($task, __('td.actionFailed'));
   179→                }
   180→
   181→                $fromStatus = $task->status;
   182→                $task->forceFill(['status' => 'ONGOING'])->save();
   183→                $this->recordStatusChange($task, $fromStatus, 'ONGOING', $me, $note !== '' ? $note : null);
   184→
   185→                return redirect()
   186→                    ->route('tasks.show', $task->id)
   187→                    ->with('success', Trans::fill('td.hist.statusChange', [
   188→                        'from' => __('labels.status.'.$fromStatus),
   189→                        'to' => __('labels.status.ONGOING'),
   190→                    ]));
   191→            }
   192→
   193→            case 'submit': {
   194→                if (! in_array($task->status, self::IN_WORK_STATUSES, true)) {
   195→                    return $this->actionError($task, __('td.actionFailed'));
   196→                }
   197→
   198→                // Assignees must reach 100% (or complete every applicable
   199→                // stage) before submitting; supervisors may force-submit.
   200→                if (! $supervisor && ! $this->readyForInspection($task)) {
   201→                    return $this->actionError($task, __('td.reach100'));
   202→                }
   203→
   204→                $fromStatus = $task->status;
   205→                $task->forceFill(['status' => 'AWAITING_INSPECTION'])->save();
   206→                $this->recordStatusChange($task, $fromStatus, 'AWAITING_INSPECTION', $me, $note !== '' ? $note : null);
   207→
   208→                if ($note !== '') {
   209→                    TaskComment::query()->create([
   210→                        'task_id' => $task->id,
   211→                        'user_id' => $me->id,
   212→                        'user_name' => $me->name,
   213→                        'message' => 'Submitted for inspection: '.$note,
   214→                    ]);
   215→                }
   216→
   217→                Notifier::management(
   218→                    'SUBMITTED_FOR_INSPECTION',
   219→                    $me->name.' submitted '.$this->taskLabel($task).' for inspection.',
   220→                    $task->id,
   221→                    $task->order_id
   222→                );
   223→
   224→                Audit::log($me, 'SUBMITTED_FOR_INSPECTION', 'TASK', $task->id, [
   225→                    'note' => $note !== '' ? $note : null,
   226→                ]);
   227→
   228→                return redirect()
   229→                    ->route('tasks.show', $task->id)
   230→                    ->with('success', __('td.toast.submitted'));
   231→            }
   232→
   233→            case 'ready': {
   234→                if (! $supervisor || ! Permissions::allows($me, 'production.inspect')) {
   235→                    abort(403);
   236→                }
   237→
   238→                if ($task->status !== 'APPROVED') {
   239→                    return $this->actionError($task, __('td.actionFailed'));
   240→                }
   241→
   242→                $fromStatus = $task->status;
   243→                $task->forceFill(['status' => 'READY_FOR_PICKUP'])->save();
   244→                $this->recordStatusChange($task, $fromStatus, 'READY_FOR_PICKUP', $me);
   245→
   246→                Notifier::management(
   247→                    'READY_FOR_PICKUP',
   248→                    $this->taskLabel($task).' is approved and ready for customer pickup.',
   249→                    $task->id,
   250→                    $task->order_id
   251→                );
   252→
   253→                if ($task->assigned_staff_id !== null) {
   254→                    Notifier::users(
   255→                        [$task->assigned_staff_id],
   256→                        'READY_FOR_PICKUP',
   257→                        $this->taskLabel($task).' is ready for customer pickup.',
   258→                        $task->id,
   259→                        $task->order_id
   260→                    );
   261→                }
   262→
   263→                Audit::log($me, 'TASK_READY_FOR_PICKUP', 'TASK', $task->id);
   264→
   265→                return redirect()
   266→                    ->route('tasks.show', $task->id)
   267→                    ->with('success', __('td.toast.ready'));
   268→            }
   269→
   270→            case 'deliver': {
   271→                if (! $supervisor || ! Permissions::allows($me, 'production.inspect')) {
   272→                    abort(403);
   273→                }
   274→
   275→                if ($task->status !== 'READY_FOR_PICKUP') {
   276→                    return $this->actionError($task, __('td.actionFailed'));
   277→                }
   278→
   279→                $fromStatus = $task->status;
   280→                $task->forceFill([
   281→                    'status' => 'DELIVERED',
   282→                    'delivered_at' => now(),
   283→                ])->save();
   284→                $this->recordStatusChange($task, $fromStatus, 'DELIVERED', $me, $note !== '' ? $note : null);
   285→
   286→                $message = $this->taskLabel($task).' has been delivered/collected by the customer.';
   287→
   288→                Notifier::management('TASK_DELIVERED', $message, $task->id, $task->order_id);
   289→
   290→                if ($task->assigned_staff_id !== null) {
   291→                    Notifier::users([$task->assigned_staff_id], 'TASK_DELIVERED', $message, $task->id, $task->order_id);
   292→                }
   293→
   294→                Audit::log($me, 'TASK_DELIVERED', 'TASK', $task->id, [
   295→                    'note' => $note !== '' ? $note : null,
   296→                ]);
   297→
   298→                return redirect()
   299→                    ->route('tasks.show', $task->id)
   300→                    ->with('success', __('td.toast.delivered'));
   301→            }
   302→
   303→            case 'reopen': {
   304→                if (! $supervisor) {
   305→                    abort(403);
   306→                }
   307→
   308→                if (! in_array($task->status, ['APPROVED', 'READY_FOR_PICKUP', 'AWAITING_INSPECTION', 'CHANGES_REQUESTED'], true)) {
   309→                    return $this->actionError($task, __('td.actionFailed'));
   310→                }
   311→
   312→                $reason = trim((string) $request->input('reason', ''));
   313→
   314→                if ($reason === '' || mb_strlen($reason) < 3) {
   315→                    throw ValidationException::withMessages([
   316→                        'reason' => Trans::fill('td.toast.reasonRequiredDesc', [
   317→                            'field' => __('td.dialog.reopenPlaceholder'),
   318→                            'min' => 3,
   319→                        ]),
   320→                    ]);
   321→                }
   322→
   323→                $fromStatus = $task->status;
   324→                $task->forceFill([
   325→                    'status' => 'ONGOING',
   326→                    'completed_at' => null,
   327→                ])->save();
   328→                $this->recordStatusChange($task, $fromStatus, 'ONGOING', $me, $reason);
   329→
   330→                if ($task->assigned_staff_id !== null) {
   331→                    Notifier::users(
   332→                        [$task->assigned_staff_id],
   333→                        'TASK_REOPENED',
   334→                        $this->taskLabel($task).' has been reopened for production.',
   335→                        $task->id,
   336→                        $task->order_id
   337→                    );
   338→                }
   339→
   340→                return redirect()
   341→                    ->route('tasks.show', $task->id)
   342→                    ->with('success', __('td.toast.reopened'));
   343→            }
   344→
   345→            case 'cancel': {
   346→                if (! $supervisor || ! Permissions::allows($me, 'production.cancel')) {
   347→                    abort(403);
   348→                }
   349→
   350→                $reason = trim((string) $request->input('reason', ''));
   351→
   352→                if ($reason === '' || mb_strlen($reason) < 3) {
   353→                    throw ValidationException::withMessages([
   354→                        'reason' => Trans::fill('td.toast.reasonRequiredDesc', [
   355→                            'field' => __('td.reason.cancel'),
   356→                            'min' => 3,
   357→                        ]),
   358→                    ]);
   359→                }
   360→
   361→                $fromStatus = $task->status;
   362→                $task->forceFill([
   363→                    'status' => 'CANCELLED',
   364→                    'cancel_reason' => $reason,
   365→                ])->save();
   366→                $this->recordStatusChange($task, $fromStatus, 'CANCELLED', $me, $reason);
   367→
   368→                if ($task->assigned_staff_id !== null) {
   369→                    Notifier::users(
   370→                        [$task->assigned_staff_id],
   371→                        'TASK_CANCELLED',
   372→                        $this->taskLabel($task).' has been cancelled. Reason: '.$reason,
   373→                        $task->id,
   374→                        $task->order_id
   375→                    );
   376→                }
   377→
   378→                Audit::log($me, 'TASK_CANCELLED', 'TASK', $task->id, ['reason' => $reason]);
   379→
   380→                return redirect()
   381→                    ->route('tasks.show', $task->id)
   382→                    ->with('success', __('td.toast.cancelled'));
   383→            }
   384→        }
   385→
   386→        return $this->actionError($task, __('td.actionFailed'));
   387→    }

   389→    /**
   390→     * POST /tasks/{id}/progress — manual numerical progress update
   391→     * (ported from tasks/[id]/progress/route.ts). 100% never auto-approves:
   392→     * inspection submission stays an explicit action. The first positive
   393→     * value on an ASSIGNED task moves it to ONGOING.
   394→     */
   395→    public function progress(Request $request, string $id): RedirectResponse
   396→    {
   397→        $me = $request->user();
   398→        $task = $this->loadTask($id);
   399→        $this->authorize($me, $task);
   400→
   401→        $supervisor = $this->isSupervisor($me);
   402→
   403→        if ($task->isTerminal()) {
   404→            return $this->closedTaskError($task);
   405→        }
   406→
   407→        if (! $supervisor && ! in_array($task->status, self::IN_WORK_STATUSES, true)) {
   408→            return $this->actionError($task, __('td.toast.progressFailed'));
   409→        }
   410→
   411→        // View contract names the field "value"; fall back to the
   412→        // reference implementation's "progress" name just in case.
   413→        if ($request->missing('value') && $request->has('progress')) {
   414→            $request->merge(['value' => $request->input('progress')]);
   415→        }
   416→
   417→        $data = $request->validate([
   418→            'value' => ['required', 'integer', 'min:0', 'max:100'],
   419→            'note' => ['nullable', 'string', 'max:500'],
   420→        ]);
   421→
   422→        $value = (int) $data['value'];
   423→        $previous = (int) $task->progress;
   424→
   425→        if ($value === $previous) {
   426→            return $this->actionError($task, __('td.toast.progressFailed'));
   427→        }
   428→
   429→        $note = trim((string) ($data['note'] ?? ''));
   430→        $fromStatus = $task->status;
   431→        $newStatus = $fromStatus === 'ASSIGNED' && $value > 0 ? 'ONGOING' : $fromStatus;
   432→
   433→        DB::transaction(function () use ($task, $me, $value, $previous, $newStatus, $note): void {
   434→            $task->forceFill([
   435→                'progress' => $value,
   436→                'status' => $newStatus,
   437→            ])->save();
   438→
   439→            TaskProgressHistory::query()->create([
   440→                'task_id' => $task->id,
   441→                'user_id' => $me->id,
   442→                'user_name' => $me->name,
   443→                'previous_progress' => $previous,
   444→                'new_progress' => $value,
   445→                'source' => 'MANUAL',
   446→                'note' => $note !== '' ? $note : null,
   447→            ]);
   448→        });
   449→
   450→        if ($newStatus !== $fromStatus) {
   451→            $this->recordStatusChange($task, $fromStatus, $newStatus, $me, 'Production started');
   452→        }
   453→
   454→        Audit::log($me, 'PROGRESS_UPDATED', 'TASK', $task->id, [
   455→            'from' => $previous,
   456→            'to' => $value,
   457→            'note' => $note !== '' ? $note : null,
   458→            'correction' => $supervisor,
   459→        ]);
   460→
   461→        return redirect()
   462→            ->route('tasks.show', $task->id)
   463→            ->with('success', Trans::fill('td.toast.progressUpdated', ['value' => $value]));
   464→    }

   466→    /**
   467→     * POST /tasks/{id}/stages — production checklist update (ported from
   468→     * tasks/[id]/stages/route.ts). Every change is recorded immutably;
   469→     * progress is recalculated via ProgressService (never decreases); the
   470→     * first change on an ASSIGNED task moves it to ONGOING.
   471→     */
   472→    public function stages(Request $request, string $id): RedirectResponse
   473→    {
   474→        $me = $request->user();
   475→        $task = $this->loadTask($id);
   476→        $this->authorize($me, $task);
   477→
   478→        $supervisor = $this->isSupervisor($me);
   479→
   480→        if ($task->isTerminal()) {
   481→            return $this->closedTaskError($task);
   482→        }
   483→
   484→        if (! $supervisor && ! in_array($task->status, self::IN_WORK_STATUSES, true)) {
   485→            return $this->actionError($task, __('td.toast.checklistFailed'));
   486→        }
   487→
   488→        $data = $request->validate([
   489→            'updates' => ['required', 'array', 'min:1', 'max:30'],
   490→            'updates.*.stageId' => ['required', 'string'],
   491→            'updates.*.state' => ['required', 'string', Rule::in(Atelier::STAGE_STATES)],
   492→            'updates.*.comment' => ['nullable', 'string', 'max:500'],
   493→        ]);
   494→
   495→        // Only stages applicable to this task's category may be updated.
   496→        $applicable = ProductionStage::query()
   497→            ->where('is_active', true)
   498→            ->orderBy('sort_order')
   499→            ->get()
   500→            ->filter(
   501→                static fn (ProductionStage $stage): bool => Atelier::stageAppliesTo(
   502→                    (string) $stage->applies_to,
   503→                    (string) $task->category
   504→                )
   505→            )
   506→            ->keyBy('id');
   507→
   508→        foreach ($data['updates'] as $update) {
   509→            if (! $applicable->has((string) $update['stageId'])) {
   510→                return $this->actionError($task, __('td.toast.checklistFailed'));
   511→            }
   512→        }
   513→
   514→        $existing = TaskStageState::query()
   515→            ->where('task_id', $task->id)
   516→            ->get()
   517→            ->keyBy('stage_id');
   518→
   519→        $applied = [];
   520→
   521→        DB::transaction(function () use ($data, $task, $me, $applicable, $existing, &$applied): void {
   522→            foreach ($data['updates'] as $update) {
   523→                /** @var ProductionStage $stage */
   524→                $stage = $applicable->get((string) $update['stageId']);
   525→                $newState = (string) $update['state'];
   526→                $previous = $existing->has($stage->id)
   527→                    ? (string) $existing->get($stage->id)->state
   528→                    : 'NOT_STARTED';
   529→
   530→                if ($previous === $newState) {
   531→                    continue;
   532→                }
   533→
   534→                $comment = trim((string) ($update['comment'] ?? ''));
   535→
   536→                TaskStageUpdate::query()->create([
   537→                    'task_id' => $task->id,
   538→                    'stage_id' => $stage->id,
   539→                    'stage_name' => $stage->name,
   540→                    'user_id' => $me->id,
   541→                    'user_name' => $me->name,
   542→                    'previous_state' => $previous,
   543→                    'new_state' => $newState,
   544→                    'comment' => $comment !== '' ? $comment : null,
   545→                ]);
   546→
   547→                if ($existing->has($stage->id)) {
   548→                    $existing->get($stage->id)->forceFill([
   549→                        'state' => $newState,
   550→                        'updated_by_id' => $me->id,
   551→                        'updated_by_name' => $me->name,
   552→                    ])->save();
   553→                } else {
   554→                    TaskStageState::query()->create([
   555→                        'task_id' => $task->id,
   556→                        'stage_id' => $stage->id,
   557→                        'state' => $newState,
   558→                        'updated_by_id' => $me->id,
   559→                        'updated_by_name' => $me->name,
   560→                    ]);
   561→                }
   562→
   563→                $applied[] = [
   564→                    'stageName' => $stage->name,
   565→                    'previousState' => $previous,
   566→                    'newState' => $newState,
   567→                ];
   568→            }
   569→        });
   570→
   571→        if ($applied === []) {
   572→            return $this->actionError($task, __('td.toast.checklistFailed'));
   573→        }
   574→
   575→        // Recalculate numerical progress from the checklist.
   576→        ProgressService::recompute($task);
   577→
   578→        // First update moves an assigned task into Ongoing.
   579→        if ($task->status === 'ASSIGNED') {
   580→            $task->forceFill(['status' => 'ONGOING'])->save();
   581→            $this->recordStatusChange($task, 'ASSIGNED', 'ONGOING', $me, 'Production started');
   582→        }
   583→
   584→        Audit::log($me, 'STAGES_UPDATED', 'TASK', $task->id, ['changes' => $applied]);
   585→
   586→        return redirect()
   587→            ->route('tasks.show', $task->id)
   588→            ->with('success', __('td.toast.checklistUpdated'));
   589→    }
   590→
   591→    /**
   592→     * POST /tasks/{id}/inspect — quality inspection (route perm
   593→     * production.inspect: MANAGEMENT / ADMIN only). Ported from the INSPECT
   594→     * case of tasks/[id]/route.ts.
   595→     */
   596→    public function inspect(Request $request, string $id): RedirectResponse
   597→    {
   598→        $me = $request->user();
   599→        $task = $this->loadTask($id);
   600→        $this->authorize($me, $task);
   601→
   602→        if ($task->isTerminal()) {
   603→            return $this->closedTaskError($task);
   604→        }
   605→
   606→        $data = $request->validate([
