    32→export function monthNow(): string {
    33→  const now = new Date();
    34→  return `${now.getUTCFullYear()}-${String(now.getUTCMonth() + 1).padStart(2, "0")}`;
    35→}
    36→
    37→export function yearNow(): string {
    38→  return String(new Date().getUTCFullYear());
    39→}
    40→
    41→function quarterNow(): string {
    42→  const now = new Date();
    43→  return `${now.getUTCFullYear()}-Q${Math.floor(now.getUTCMonth() / 3) + 1}`;
    44→}
    45→
    46→function shiftMonth(value: string, delta: number): string {
    47→  const m = /^(\d{4})-(\d{2})$/.exec(value);
    48→  if (!m) return value;
    49→  const d = new Date(Date.UTC(Number(m[1]), Number(m[2]) - 1 + delta, 1));
    50→  return `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, "0")}`;
    51→}
    52→
    53→function shiftQuarter(value: string, delta: number): string {
    54→  const m = /^(\d{4})-Q([1-4])$/.exec(value);
    55→  if (!m) return value;
    56→  const total = Number(m[1]) * 4 + (Number(m[2]) - 1) + delta;
    57→  const year = Math.floor(total / 4);
    58→  return `${year}-Q${total - year * 4 + 1}`;
    59→}
    60→
    61→function shiftYear(value: string, delta: number): string {
    62→  const m = /^\d{4}$/.exec(value);
    63→  if (!m) return value;
    64→  return String(Number(m[1]) + delta);
    65→}

    71→export function scoreTone(value: number): string {
    72→  if (value >= 75) return "text-emerald-700";
    73→  if (value >= 50) return "text-amber-700";
    74→  return "text-red-700";
    75→}

    77→function csvCell(value: string | number | null | undefined): string {
    78→  const s = value == null ? "" : String(value);
    79→  return /[",\n]/.test(s) ? `"${s.replaceAll('"', '""')}"` : s;
    80→}

    82→export function downloadCsv(
    83→  filename: string,
    84→  rows: Array<Array<string | number | null | undefined>>
    85→): void {
    86→  const csv = rows.map((r) => r.map(csvCell).join(",")).join("\n");
    87→  const blob = new Blob([`\uFEFF${csv}`], { type: "text/csv;charset=utf-8" });
    88→  const url = URL.createObjectURL(blob);
    89→  const link = document.createElement("a");
    90→  link.href = url;
    91→  link.download = filename;
    92→  document.body.appendChild(link);
    93→  link.click();
    94→  link.remove();
    95→  URL.revokeObjectURL(url);
    96→}

    98→/** Localized display label for a server period range (server labels are English-only). */
    99→export function periodLabel(
   100→  t: TranslateFn,
   101→  lang: Lang,
   102→  range: { type: string; value: string; label: string }
   103→): string {
   104→  if (range.type === "MONTH") {
   105→    const m = /^(\d{4})-(\d{2})$/.exec(range.value);
   106→    if (m) {
   107→      const month = new Intl.DateTimeFormat(lang === "fr" ? "fr-FR" : "en-US", {
   108→        month: "long",
   109→        timeZone: "UTC",
   110→      }).format(new Date(Date.UTC(Number(m[1]), Number(m[2]) - 1, 1)));
   111→      return t("perf.rangeMonth", { month, year: m[1] });
   112→    }
   113→  }
   114→  if (range.type === "QUARTER") {
   115→    const q = /^(\d{4})-Q([1-4])$/.exec(range.value.toUpperCase());
   116→    if (q) return t("perf.rangeQuarter", { quarter: q[2], year: q[1] });
   117→  }
   118→  if (range.type === "YEAR" && /^\d{4}$/.test(range.value)) {
   119→    return t("perf.rangeYear", { year: range.value });
   120→  }
   121→  return range.label;
   122→}

   535→                <Label htmlFor="perf-qyear">{t("perf.year")}</Label>
   536→                <Input
   537→                  id="perf-qyear"

   561→                </Select>

   565→
   566→          {pType === "YEAR" && (

   568→              <Label htmlFor="perf-year">{t("perf.year")}</Label>

   570→                id="perf-year"

   578→
   579→          <div className="flex items-center gap-1.5 pb-0.5">

   581→              variant="outline"
   582→              size="icon"
   583→              onClick={() => stepPeriod(-1)}

   589→              variant="outline"
   590→              size="icon"

   593→            >
   594→              <ChevronRight className="size-4" />
   595→            </Button>
   596→          </div>
   597→
   598→          {data && (

   603→        </div>
   604→      </SectionCard>
   605→
   606→      {/* ------------------------------------------- weights (Management only) */}
   607→      {canConfigure && weightsForm && (

   610→          className="mb-4"
   611→          action={

   617→              <Save className="size-4 mr-1.5" />

   652→            </div>
   653→          </div>
   654→          <div className="flex flex-wrap items-center justify-between gap-2 mt-4">

   665→
   666→      {/* -------------------------------------------------------- staff table */}

   678→            <Button variant="outline" size="sm" onClick={() => void loadPerformance()}>

   682→        />

   690→        <SectionCard
   691→          title={t("perf.staffTableTitle", { period: periodLabel(t, lang, data.range) })}
   692→          contentClassName="p-0"
   693→          className="mb-4"
   694→        >
   695→          <StaffPerformanceTable staff={data.staff} assessmentCell={assessmentCell} />
   696→          <p className="text-xs text-muted-foreground px-4 py-3 border-t">
   697→            {t("perf.formulaNote")}
   698→          </p>
   699→        </SectionCard>
   700→      )}
   701→
   702→      {!loading && data && data.staff.length > 0 && (
   703→        <p className="text-xs text-muted-foreground flex items-center gap-1.5">
   704→          <TrendingUp className="size-3.5" />
   705→          {t("perf.colorNote")}
   706→        </p>
   707→      )}
   708→    </div>
   709→  );
