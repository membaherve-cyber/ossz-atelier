   107→type HistoryKind = "assignment" | "status" | "stage" | "progress" | "inspection" | "comment";

   125→const STAGE_STATE_OPTIONS = STAGE_STATES.map((s) => ({
   126→  value: s,
   127→  label: s === "NOT_APPLICABLE" ? "N/A" : STAGE_STATE_LABELS[s],

   129→
   130→/** Convert an ISO timestamp to a `datetime-local` input value (local time). */
   131→function toDatetimeLocal(iso: string | null): string {
   132→  if (!iso) return "";
   133→  const d = new Date(iso);
   134→  if (Number.isNaN(d.getTime())) return "";
   135→  return format(d, "yyyy-MM-dd'T'HH:mm");
   136→}
   137→
   138→function statusLabel(status: string): string {
   139→  return STATUS_LABELS[status as TaskStatus] ?? status;
   140→}
   141→
   142→function stageLabel(state: string): string {
   143→  return STAGE_STATE_LABELS[state as StageState] ?? state;
   144→}

   612→  const doReopen = () => {
