    30→    30→    30→/**

    32→    32→    32→ * stages|inspect|comment|block|unblock|update).
    33→    33→    33→ *
    34→    34→    34→ * Every action is a 1:1 port of the Next.js reference implementation:
    35→    35→    35→ * - PATCH src/app/api/atelier/tasks/[id]/route.ts — assign, update, cancel,
    36→    36→    36→ *   block, unblock, submit, inspect, ready, deliver (status-transition
    37→    37→    37→ *   gates, notifications, audit codes ported verbatim);
    38→    38→    38→ * - POST src/app/api/atelier/tasks/[id]/progress/route.ts — progress;

    40→    40→    40→ * - POST src/app/api/atelier/tasks/[id]/comments/route.ts — notes/issues.
    41→    41→    41→ * The `status` endpoint additionally offers "start" and "reopen" verbs per
    42→    42→    42→ * the controller/view contract.

    46→    46→    46→ * Terminal tasks (DELIVERED / CANCELLED) reject every action, like the
    47→    47→    47→ * outer gate of the Next.js PATCH handler.
    48→    48→    48→ *
    49→    49→    49→ * Deliberately does not extend a base Controller class, mirroring

    60→    60→    60→    /**
    61→    61→    61→     * POST /tasks/{id}/assign — assign or reassign the task (route perm
    62→    62→    62→     * production.assign: MANAGEMENT / ADMIN only).
    63→    63→    63→     *
    64→    64→    64→     * Ported from the ASSIGN case of tasks/[id]/route.ts: the new status
    65→    65→    65→     * only moves UNASSIGNED -> ASSIGNED; the assignment history row is
    66→    66→    66→     * typed ASSIGN or REASSIGN; the new assignee is notified (message
    67→    67→    67→     * mirrors the reference implementation exactly — the old assignee is
    68→    68→    68→     * NOT notified, matching route.ts).
    69→    69→    69→     */
    70→    70→    70→    public function assign(Request $request, string $id): RedirectResponse
    71→    71→    71→    {
    72→    72→    72→        $me = $request->user();
    73→    73→    73→        $task = $this->loadTask($id);
    74→    74→    74→        $this->authorize($me, $task);
    75→    75→    75→
    76→    76→    76→        if ($task->isTerminal()) {
    77→    77→    77→            return $this->closedTaskError($task);
    78→    78→    78→        }
    79→    79→    79→
    80→    80→    80→        $data = $request->validate([
    81→    81→    81→            'staffId' => [
    82→    82→    82→                'required',
    83→    83→    83→                'string',
    84→    84→    84→                Rule::exists('users', 'id')->where('is_active', true),
    85→    85→    85→            ],
    86→    86→    86→            'note' => ['nullable', 'string', 'max:1000'],
    87→    87→    87→        ]);
    88→    88→    88→
    89→    89→    89→        $staff = User::query()->findOrFail($data['staffId']);
    90→    90→    90→
    91→    91→    91→        if (! in_array($staff->role, Permissions::assignableRoles(), true)) {
    92→    92→    92→            return $this->actionError($task, __('td.toast.selectStaff'));
    93→    93→    93→        }

    95→    95→    95→        if ($staff->id === $task->assigned_staff_id) {
    96→    96→    96→            return $this->actionError($task, __('td.toast.selectStaff'));
    97→    97→    97→        }
    98→    98→    98→
    99→    99→    99→        $reassign = $task->assigned_staff_id !== null;
   100→   100→   100→        $fromStatus = $task->status;
   101→   101→   101→        $newStatus = $fromStatus === 'UNASSIGNED' ? 'ASSIGNED' : $fromStatus;
   102→   102→   102→        $note = trim((string) ($data['note'] ?? ''));
   103→   103→   103→
   104→   104→   104→        DB::transaction(function () use ($task, $staff, $me, $reassign, $newStatus, $note): void {
   105→   105→   105→            $task->forceFill([
   106→   106→   106→                'assigned_staff_id' => $staff->id,
   107→   107→   107→                'status' => $newStatus,
   108→   108→   108→            ])->save();
   109→   109→   109→
   110→   110→   110→            TaskAssignment::query()->create([
   111→   111→   111→                'task_id' => $task->id,
   112→   112→   112→                'staff_id' => $staff->id,
   113→   113→   113→                'staff_name' => $staff->name,
   114→   114→   114→                'assigned_by_id' => $me->id,
   115→   115→   115→                'assigned_by_name' => $me->name,
   116→   116→   116→                'type' => $reassign ? 'REASSIGN' : 'ASSIGN',
   117→   117→   117→                'note' => $note !== '' ? $note : null,
   118→   118→   118→            ]);
   119→   119→   119→        });
   120→   120→   120→
   121→   121→   121→        if ($newStatus !== $fromStatus) {
   122→   122→   122→            $this->recordStatusChange($task, $fromStatus, $newStatus, $me, $note !== '' ? $note : null);
   123→   123→   123→        }
   124→   124→   124→
   125→   125→   125→        $message = $this->taskLabel($task).' has been '.($reassign ? 'reassigned' : 'assigned')
   126→   126→   126→            .' to you.'.($note !== '' ? ' Note: '.$note : '');
   127→   127→   127→
   128→   128→   128→        Notifier::users(
   129→   129→   129→            [$staff->id],
   130→   130→   130→            $reassign ? 'TASK_REASSIGNED' : 'TASK_ASSIGNED',
   131→   131→   131→            $message,
   132→   132→   132→            $task->id,
   133→   133→   133→            $task->order_id
   134→   134→   134→        );
   135→   135→   135→
   136→   136→   136→        Audit::log($me, $reassign ? 'TASK_REASSIGNED' : 'TASK_ASSIGNED', 'TASK', $task->id, [
   137→   137→   137→            'staff' => $staff->name,
   138→   138→   138→            'note' => $note !== '' ? $note : null,
   139→   139→   139→        ]);
   140→   140→   140→
   141→   141→   141→        return redirect()
   142→   142→   142→            ->route('tasks.show', $task->id)
   143→   143→   143→            ->with('success', Trans::fill($reassign ? 'td.toast.reassignedTo' : 'td.toast.assignedTo', ['name' => $staff->name]));
   144→   144→   144→    }

   146→   146→   146→    /**
   147→   147→   147→     * POST /tasks/{id}/status — workflow transitions driven by the hidden
   148→   148→   148→     * `action` field: start|submit|ready|deliver|reopen|cancel (view
   149→   149→   149→     * contract). Gates, side effects, notifications and audit codes are
   150→   150→   150→     * ported from the SUBMIT / READY / DELIVER / CANCEL cases of
   151→   151→   151→     * tasks/[id]/route.ts; "start" and "reopen" implement the contract
   152→   152→   152→     * additions (reopen mirrors the REOPENED inspection decision).
   153→   153→   153→     */
   154→   154→   154→    public function status(Request $request, string $id): RedirectResponse
   155→   155→   155→    {
   156→   156→   156→        $me = $request->user();
   157→   157→   157→        $task = $this->loadTask($id);
   158→   158→   158→        $this->authorize($me, $task);
   159→   159→   159→
   160→   160→   160→        if ($task->isTerminal()) {
   161→   161→   161→            return $this->closedTaskError($task);
   162→   162→   162→        }
   163→   163→   163→
   164→   164→   164→        $data = $request->validate([
   165→   165→   165→            'action' => ['required', 'string', Rule::in(['start', 'submit', 'ready', 'deliver', 'reopen', 'cancel'])],
   166→   166→   166→            'note' => ['nullable', 'string', 'max:1000'],
   167→   167→   167→        ]);
   168→   168→   168→
   169→   169→   169→        $action = (string) $data['action'];
   170→   170→   170→        $note = trim((string) ($data['note'] ?? ''));
   171→   171→   171→        $supervisor = $this->isSupervisor($me);
   172→   172→   172→
   173→   173→   173→        switch ($action) {
   174→   174→   174→            case 'start': {
   175→   175→   175→                // Contract: assignee or supervisor, from ASSIGNED or
   176→   176→   176→                // CHANGES_REQUESTED -> ONGOING.
   177→   177→   177→                if (! in_array($task->status, ['ASSIGNED', 'CHANGES_REQUESTED'], true)) {
   178→   178→   178→                    return $this->actionError($task, __('td.actionFailed'));
   179→   179→   179→                }
   180→   180→   180→
   181→   181→   181→                $fromStatus = $task->status;
   182→   182→   182→                $task->forceFill(['status' => 'ONGOING'])->save();
   183→   183→   183→                $this->recordStatusChange($task, $fromStatus, 'ONGOING', $me, $note !== '' ? $note : null);
   184→   184→   184→
   185→   185→   185→                return redirect()
   186→   186→   186→                    ->route('tasks.show', $task->id)
   187→   187→   187→                    ->with('success', Trans::fill('td.hist.statusChange', [
   188→   188→   188→                        'from' => __('labels.status.'.$fromStatus),
   189→   189→   189→                        'to' => __('labels.status.ONGOING'),
   190→   190→   190→                    ]));
   191→   191→   191→            }
   192→   192→   192→
   193→   193→   193→            case 'submit': {
   194→   194→   194→                if (! in_array($task->status, self::IN_WORK_STATUSES, true)) {
   195→   195→   195→                    return $this->actionError($task, __('td.actionFailed'));
   196→   196→   196→                }
   197→   197→   197→
   198→   198→   198→                // Assignees must reach 100% (or complete every applicable
   199→   199→   199→                // stage) before submitting; supervisors may force-submit.
   200→   200→   200→                if (! $supervisor && ! $this->readyForInspection($task)) {
   201→   201→   201→                    return $this->actionError($task, __('td.reach100'));
   202→   202→   202→                }
   203→   203→   203→
   204→   204→   204→                $fromStatus = $task->status;
   205→   205→   205→                $task->forceFill(['status' => 'AWAITING_INSPECTION'])->save();
   206→   206→   206→                $this->recordStatusChange($task, $fromStatus, 'AWAITING_INSPECTION', $me, $note !== '' ? $note : null);
   207→   207→   207→
   208→   208→   208→                if ($note !== '') {
   209→   209→   209→                    TaskComment::query()->create([
   210→   210→   210→                        'task_id' => $task->id,
   211→   211→   211→                        'user_id' => $me->id,
   212→   212→   212→                        'user_name' => $me->name,
   213→   213→   213→                        'message' => 'Submitted for inspection: '.$note,
   214→   214→   214→                    ]);
   215→   215→   215→                }
   216→   216→   216→
   217→   217→   217→                Notifier::management(
   218→   218→   218→                    'SUBMITTED_FOR_INSPECTION',
   219→   219→   219→                    $me->name.' submitted '.$this->taskLabel($task).' for inspection.',
   220→   220→   220→                    $task->id,
   221→   221→   221→                    $task->order_id
   222→   222→   222→                );
   223→   223→   223→
   224→   224→   224→                Audit::log($me, 'SUBMITTED_FOR_INSPECTION', 'TASK', $task->id, [
   225→   225→   225→                    'note' => $note !== '' ? $note : null,
   226→   226→   226→                ]);
   227→   227→   227→
   228→   228→   228→                return redirect()
   229→   229→   229→                    ->route('tasks.show', $task->id)
   230→   230→   230→                    ->with('success', __('td.toast.submitted'));
   231→   231→   231→            }
   232→   232→   232→
   233→   233→   233→            case 'ready': {
   234→   234→   234→                if (! $supervisor || ! Permissions::allows($me, 'production.inspect')) {
   235→   235→   235→                    abort(403);
   236→   236→   236→                }
   237→   237→   237→
   238→   238→   238→                if ($task->status !== 'APPROVED') {
   239→   239→   239→                    return $this->actionError($task, __('td.actionFailed'));
   240→   240→   240→                }
   241→   241→   241→
   242→   242→   242→                $fromStatus = $task->status;
   243→   243→   243→                $task->forceFill(['status' => 'READY_FOR_PICKUP'])->save();
   244→   244→   244→                $this->recordStatusChange($task, $fromStatus, 'READY_FOR_PICKUP', $me);
   245→   245→   245→
   246→   246→   246→                Notifier::management(
   247→   247→   247→                    'READY_FOR_PICKUP',
   248→   248→   248→                    $this->taskLabel($task).' is approved and ready for customer pickup.',
   249→   249→   249→                    $task->id,
   250→   250→   250→                    $task->order_id
   251→   251→   251→                );
   252→   252→   252→
   253→   253→   253→                if ($task->assigned_staff_id !== null) {
   254→   254→   254→                    Notifier::users(
   255→   255→   255→                        [$task->assigned_staff_id],
   256→   256→   256→                        'READY_FOR_PICKUP',
   257→   257→   257→                        $this->taskLabel($task).' is ready for customer pickup.',
   258→   258→   258→                        $task->id,
   259→   259→   259→                        $task->order_id
   260→   260→   260→                    );
   261→   261→   261→                }
   262→   262→   262→
   263→   263→   263→                Audit::log($me, 'TASK_READY_FOR_PICKUP', 'TASK', $task->id);
   264→   264→   264→
   265→   265→   265→                return redirect()
   266→   266→   266→                    ->route('tasks.show', $task->id)
   267→   267→   267→                    ->with('success', __('td.toast.ready'));
   268→   268→   268→            }
   269→   269→   269→
   270→   270→   270→            case 'deliver': {
   271→   271→   271→                if (! $supervisor || ! Permissions::allows($me, 'production.inspect')) {
   272→   272→   272→                    abort(403);
   273→   273→   273→                }
   274→   274→   274→
   275→   275→   275→                if ($task->status !== 'READY_FOR_PICKUP') {
   276→   276→   276→                    return $this->actionError($task, __('td.actionFailed'));
   277→   277→   277→                }
   278→   278→   278→
   279→   279→   279→                $fromStatus = $task->status;
   280→   280→   280→                $task->forceFill([
   281→   281→   281→                    'status' => 'DELIVERED',
   282→   282→   282→                    'delivered_at' => now(),
   283→   283→   283→                ])->save();
   284→   284→   284→                $this->recordStatusChange($task, $fromStatus, 'DELIVERED', $me, $note !== '' ? $note : null);
   285→   285→   285→
   286→   286→   286→                $message = $this->taskLabel($task).' has been delivered/collected by the customer.';
   287→   287→   287→
   288→   288→   288→                Notifier::management('TASK_DELIVERED', $message, $task->id, $task->order_id);
   289→   289→   289→
   290→   290→   290→                if ($task->assigned_staff_id !== null) {
   291→   291→   291→                    Notifier::users([$task->assigned_staff_id], 'TASK_DELIVERED', $message, $task->id, $task->order_id);
   292→   292→   292→                }
   293→   293→   293→
   294→   294→   294→                Audit::log($me, 'TASK_DELIVERED', 'TASK', $task->id, [
   295→   295→   295→                    'note' => $note !== '' ? $note : null,
   296→   296→   296→                ]);
   297→   297→   297→
   298→   298→   298→                return redirect()
   299→   299→   299→                    ->route('tasks.show', $task->id)
   300→   300→   300→                    ->with('success', __('td.toast.delivered'));
   301→   301→   301→            }
   302→   302→   302→
   303→   303→   303→            case 'reopen': {
   304→   304→   304→                if (! $supervisor) {
   305→   305→   305→                    abort(403);
   306→   306→   306→                }
   307→   307→   307→
   308→   308→   308→                if (! in_array($task->status, ['APPROVED', 'READY_FOR_PICKUP', 'AWAITING_INSPECTION', 'CHANGES_REQUESTED'], true)) {
   309→   309→   309→                    return $this->actionError($task, __('td.actionFailed'));
   310→   310→   310→                }
   311→   311→   311→
   312→   312→   312→                $reason = trim((string) $request->input('reason', ''));
   313→   313→   313→
   314→   314→   314→                if ($reason === '' || mb_strlen($reason) < 3) {
   315→   315→   315→                    throw ValidationException::withMessages([
   316→   316→   316→                        'reason' => Trans::fill('td.toast.reasonRequiredDesc', [
   317→   317→   317→                            'field' => __('td.dialog.reopenPlaceholder'),
   318→   318→   318→                            'min' => 3,
   319→   319→   319→                        ]),
   320→   320→   320→                    ]);
   321→   321→   321→                }
   322→   322→   322→
   323→   323→   323→                $fromStatus = $task->status;
   324→   324→   324→                $task->forceFill([
   325→   325→   325→                    'status' => 'ONGOING',
   326→   326→   326→                    'completed_at' => null,
   327→   327→   327→                ])->save();
   328→   328→   328→                $this->recordStatusChange($task, $fromStatus, 'ONGOING', $me, $reason);
   329→   329→   329→
   330→   330→   330→                if ($task->assigned_staff_id !== null) {
   331→   331→   331→                    Notifier::users(
   332→   332→   332→                        [$task->assigned_staff_id],
   333→   333→   333→                        'TASK_REOPENED',
   334→   334→   334→                        $this->taskLabel($task).' has been reopened for production.',
   335→   335→   335→                        $task->id,
   336→   336→   336→                        $task->order_id
   337→   337→   337→                    );
   338→   338→   338→                }
   339→   339→   339→
   340→   340→   340→                return redirect()
   341→   341→   341→                    ->route('tasks.show', $task->id)
   342→   342→   342→                    ->with('success', __('td.toast.reopened'));
   343→   343→   343→            }
   344→   344→   344→
   345→   345→   345→            case 'cancel': {
   346→   346→   346→                if (! $supervisor || ! Permissions::allows($me, 'production.cancel')) {
   347→   347→   347→                    abort(403);
   348→   348→   348→                }
   349→   349→   349→
   350→   350→   350→                $reason = trim((string) $request->input('reason', ''));
   351→   351→   351→
   352→   352→   352→                if ($reason === '' || mb_strlen($reason) < 3) {
   353→   353→   353→                    throw ValidationException::withMessages([
   354→   354→   354→                        'reason' => Trans::fill('td.toast.reasonRequiredDesc', [
   355→   355→   355→                            'field' => __('td.reason.cancel'),
   356→   356→   356→                            'min' => 3,
   357→   357→   357→                        ]),
   358→   358→   358→                    ]);
   359→   359→   359→                }
   360→   360→   360→
   361→   361→   361→                $fromStatus = $task->status;
   362→   362→   362→                $task->forceFill([
   363→   363→   363→                    'status' => 'CANCELLED',
   364→   364→   364→                    'cancel_reason' => $reason,
   365→   365→   365→                ])->save();
   366→   366→   366→                $this->recordStatusChange($task, $fromStatus, 'CANCELLED', $me, $reason);
   367→   367→   367→
   368→   368→   368→                if ($task->assigned_staff_id !== null) {
   369→   369→   369→                    Notifier::users(
   370→   370→   370→                        [$task->assigned_staff_id],
   371→   371→   371→                        'TASK_CANCELLED',
   372→   372→   372→                        $this->taskLabel($task).' has been cancelled. Reason: '.$reason,
   373→   373→   373→                        $task->id,
   374→   374→   374→                        $task->order_id
   375→   375→   375→                    );
   376→   376→   376→                }
   377→   377→   377→
   378→   378→   378→                Audit::log($me, 'TASK_CANCELLED', 'TASK', $task->id, ['reason' => $reason]);
   379→   379→   379→
   380→   380→   380→                return redirect()
   381→   381→   381→                    ->route('tasks.show', $task->id)
   382→   382→   382→                    ->with('success', __('td.toast.cancelled'));
   383→   383→   383→            }
   384→   384→   384→        }
   385→   385→   385→
   386→   386→   386→        return $this->actionError($task, __('td.actionFailed'));
   387→   387→   387→    }

   389→   389→   389→    /**
   390→   390→   390→     * POST /tasks/{id}/progress — manual numerical progress update
   391→   391→   391→     * (ported from tasks/[id]/progress/route.ts). 100% never auto-approves:
   392→   392→   392→     * inspection submission stays an explicit action. The first positive
   393→   393→   393→     * value on an ASSIGNED task moves it to ONGOING.
   394→   394→   394→     */
