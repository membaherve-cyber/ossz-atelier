    29→    29→/**
    30→    30→ * Task workflow actions (spec §3: POST /tasks/{id}/assign|status|progress|
    31→    31→ * stages|inspect|comment|block|unblock|update).
    32→    32→ *
    33→    33→ * Every action is a 1:1 port of the Next.js reference implementation:
    34→    34→ * - PATCH src/app/api/atelier/tasks/[id]/route.ts — assign, update, cancel,
    35→    35→ *   block, unblock, submit, inspect, ready, deliver (status-transition
    36→    36→ *   gates, notifications, audit codes ported verbatim);
    37→    37→ * - POST src/app/api/atelier/tasks/[id]/progress/route.ts — progress;
    38→    38→ * - POST src/app/api/atelier/tasks/[id]/stages/route.ts — checklist;
    39→    39→ * - POST src/app/api/atelier/tasks/[id]/comments/route.ts — notes/issues.
    40→    40→ * The `status` endpoint additionally offers "start" and "reopen" verbs per
    41→    41→ * the controller/view contract.
    42→    42→ *
    43→    43→ * Access rules: a task may only be touched by its assignee or a supervisor
    44→    44→ * (MANAGEMENT / ADMIN); STAFF users reaching another user's task get 403.
    45→    45→ * Terminal tasks (DELIVERED / CANCELLED) reject every action, like the
    46→    46→ * outer gate of the Next.js PATCH handler.
    47→    47→ *
    48→    48→ * Deliberately does not extend a base Controller class, mirroring
    49→    49→ * LoginController / LocaleController.
    50→    50→ */
    51→    51→class TaskActionController

    53→    53→    /**

    57→    57→    private const IN_WORK_STATUSES = ['ASSIGNED', 'ONGOING', 'CHANGES_REQUESTED'];

    59→    59→    /**
    60→    60→     * POST /tasks/{id}/assign — assign or reassign the task (route perm
    61→    61→     * production.assign: MANAGEMENT / ADMIN only).
    62→    62→     *
    63→    63→     * Ported from the ASSIGN case of tasks/[id]/route.ts: the new status
    64→    64→     * only moves UNASSIGNED -> ASSIGNED; the assignment history row is
    65→    65→     * typed ASSIGN or REASSIGN; the new assignee is notified (message
    66→    66→     * mirrors the reference implementation exactly — the old assignee is
    67→    67→     * NOT notified, matching route.ts).
    68→    68→     */
    69→    69→    public function assign(Request $request, string $id): RedirectResponse
    70→    70→    {
    71→    71→        $me = $request->user();
    72→    72→        $task = $this->loadTask($id);
    73→    73→        $this->authorize($me, $task);
    74→    74→
    75→    75→        if ($task->isTerminal()) {
    76→    76→            return $this->closedTaskError($task);
    77→    77→        }
    78→    78→
    79→    79→        $data = $request->validate([
    80→    80→            'staffId' => [
    81→    81→                'required',
    82→    82→                'string',
    83→    83→                Rule::exists('users', 'id')->where('is_active', true),
    84→    84→            ],
    85→    85→            'note' => ['nullable', 'string', 'max:1000'],
    86→    86→        ]);
    87→    87→
    88→    88→        $staff = User::query()->findOrFail($data['staffId']);
    89→    89→
    90→    90→        if (! in_array($staff->role, Permissions::assignableRoles(), true)) {
    91→    91→            return $this->actionError($task, __('td.toast.selectStaff'));
    92→    92→        }
    93→    93→
    94→    94→        if ($staff->id === $task->assigned_staff_id) {
    95→    95→            return $this->actionError($task, __('td.toast.selectStaff'));
    96→    96→        }
    97→    97→
    98→    98→        $reassign = $task->assigned_staff_id !== null;
    99→    99→        $fromStatus = $task->status;
   100→   100→        $newStatus = $fromStatus === 'UNASSIGNED' ? 'ASSIGNED' : $fromStatus;
   101→   101→        $note = trim((string) ($data['note'] ?? ''));
   102→   102→
   103→   103→        DB::transaction(function () use ($task, $staff, $me, $reassign, $newStatus, $note): void {
   104→   104→            $task->forceFill([
   105→   105→                'assigned_staff_id' => $staff->id,
   106→   106→                'status' => $newStatus,
   107→   107→            ])->save();
   108→   108→
   109→   109→            TaskAssignment::query()->create([
   110→   110→                'task_id' => $task->id,
   111→   111→                'staff_id' => $staff->id,
   112→   112→                'staff_name' => $staff->name,
   113→   113→                'assigned_by_id' => $me->id,
   114→   114→                'assigned_by_name' => $me->name,
   115→   115→                'type' => $reassign ? 'REASSIGN' : 'ASSIGN',
   116→   116→                'note' => $note !== '' ? $note : null,
   117→   117→            ]);
   118→   118→        });
   119→   119→
   120→   120→        if ($newStatus !== $fromStatus) {
   121→   121→            $this->recordStatusChange($task, $fromStatus, $newStatus, $me, $note !== '' ? $note : null);
   122→   122→        }
   123→   123→
   124→   124→        $message = $this->taskLabel($task).' has been '.($reassign ? 'reassigned' : 'assigned')
   125→   125→            .' to you.'.($note !== '' ? ' Note: '.$note : '');
   126→   126→
   127→   127→        Notifier::users(
   128→   128→            [$staff->id],
   129→   129→            $reassign ? 'TASK_REASSIGNED' : 'TASK_ASSIGNED',
   130→   130→            $message,
   131→   131→            $task->id,
   132→   132→            $task->order_id
   133→   133→        );
   134→   134→
   135→   135→        Audit::log($me, $reassign ? 'TASK_REASSIGNED' : 'TASK_ASSIGNED', 'TASK', $task->id, [
   136→   136→            'staff' => $staff->name,
   137→   137→            'note' => $note !== '' ? $note : null,
   138→   138→        ]);
   139→   139→
   140→   140→        return redirect()
   141→   141→            ->route('tasks.show', $task->id)
   142→   142→            ->with('success', __($reassign ? 'td.toast.reassignedTo' : 'td.toast.assignedTo', ['name' => $staff->name]));
   143→   143→    }

   145→   145→    /**
   146→   146→     * POST /tasks/{id}/status — workflow transitions driven by the hidden
   147→   147→     * `action` field: start|submit|ready|deliver|reopen|cancel (view
   148→   148→     * contract). Gates, side effects, notifications and audit codes are
   149→   149→     * ported from the SUBMIT / READY / DELIVER / CANCEL cases of
   150→   150→     * tasks/[id]/route.ts; "start" and "reopen" implement the contract
   151→   151→     * additions (reopen mirrors the REOPENED inspection decision).
   152→   152→     */
   153→   153→    public function status(Request $request, string $id): RedirectResponse
   154→   154→    {
   155→   155→        $me = $request->user();
   156→   156→        $task = $this->loadTask($id);
   157→   157→        $this->authorize($me, $task);
   158→   158→
   159→   159→        if ($task->isTerminal()) {
   160→   160→            return $this->closedTaskError($task);
   161→   161→        }
   162→   162→
   163→   163→        $data = $request->validate([
   164→   164→            'action' => ['required', 'string', Rule::in(['start', 'submit', 'ready', 'deliver', 'reopen', 'cancel'])],
   165→   165→            'note' => ['nullable', 'string', 'max:1000'],
   166→   166→        ]);
   167→   167→
   168→   168→        $action = (string) $data['action'];
   169→   169→        $note = trim((string) ($data['note'] ?? ''));
   170→   170→        $supervisor = $this->isSupervisor($me);
   171→   171→
   172→   172→        switch ($action) {
   173→   173→            case 'start': {
   174→   174→                // Contract: assignee or supervisor, from ASSIGNED or
   175→   175→                // CHANGES_REQUESTED -> ONGOING.
   176→   176→                if (! in_array($task->status, ['ASSIGNED', 'CHANGES_REQUESTED'], true)) {
   177→   177→                    return $this->actionError($task, __('td.actionFailed'));
   178→   178→                }
   179→   179→
   180→   180→                $fromStatus = $task->status;
   181→   181→                $task->forceFill(['status' => 'ONGOING'])->save();
   182→   182→                $this->recordStatusChange($task, $fromStatus, 'ONGOING', $me, $note !== '' ? $note : null);
   183→   183→
   184→   184→                return redirect()
   185→   185→                    ->route('tasks.show', $task->id)
   186→   186→                    ->with('success', __('td.hist.statusChange', [
   187→   187→                        'from' => __('labels.status.'.$fromStatus),
   188→   188→                        'to' => __('labels.status.ONGOING'),
   189→   189→                    ]));
   190→   190→            }
   191→   191→
   192→   192→            case 'submit': {
   193→   193→                if (! in_array($task->status, self::IN_WORK_STATUSES, true)) {
   194→   194→                    return $this->actionError($task, __('td.actionFailed'));
   195→   195→                }
   196→   196→
   197→   197→                // Assignees must reach 100% (or complete every applicable
   198→   198→                // stage) before submitting; supervisors may force-submit.
   199→   199→                if (! $supervisor && ! $this->readyForInspection($task)) {
   200→   200→                    return $this->actionError($task, __('td.reach100'));
   201→   201→                }
   202→   202→
   203→   203→                $fromStatus = $task->status;
   204→   204→                $task->forceFill(['status' => 'AWAITING_INSPECTION'])->save();
   205→   205→                $this->recordStatusChange($task, $fromStatus, 'AWAITING_INSPECTION', $me, $note !== '' ? $note : null);
   206→   206→
   207→   207→                if ($note !== '') {
   208→   208→                    TaskComment::query()->create([
   209→   209→                        'task_id' => $task->id,
   210→   210→                        'user_id' => $me->id,
   211→   211→                        'user_name' => $me->name,
   212→   212→                        'message' => 'Submitted for inspection: '.$note,
   213→   213→                    ]);
   214→   214→                }
   215→   215→
   216→   216→                Notifier::management(
   217→   217→                    'SUBMITTED_FOR_INSPECTION',
   218→   218→                    $me->name.' submitted '.$this->taskLabel($task).' for inspection.',
   219→   219→                    $task->id,
   220→   220→                    $task->order_id
   221→   221→                );
   222→   222→
   223→   223→                Audit::log($me, 'SUBMITTED_FOR_INSPECTION', 'TASK', $task->id, [
   224→   224→                    'note' => $note !== '' ? $note : null,
   225→   225→                ]);
   226→   226→
   227→   227→                return redirect()
   228→   228→                    ->route('tasks.show', $task->id)
   229→   229→                    ->with('success', __('td.toast.submitted'));
   230→   230→            }
   231→   231→
   232→   232→            case 'ready': {
   233→   233→                if (! $supervisor || ! Permissions::allows($me, 'production.inspect')) {
   234→   234→                    abort(403);
   235→   235→                }
   236→   236→
   237→   237→                if ($task->status !== 'APPROVED') {
   238→   238→                    return $this->actionError($task, __('td.actionFailed'));
   239→   239→                }
   240→   240→
   241→   241→                $fromStatus = $task->status;
   242→   242→                $task->forceFill(['status' => 'READY_FOR_PICKUP'])->save();
   243→   243→                $this->recordStatusChange($task, $fromStatus, 'READY_FOR_PICKUP', $me);
   244→   244→
   245→   245→                Notifier::management(
   246→   246→                    'READY_FOR_PICKUP',
   247→   247→                    $this->taskLabel($task).' is approved and ready for customer pickup.',
   248→   248→                    $task->id,
   249→   249→                    $task->order_id
   250→   250→                );
   251→   251→
   252→   252→                if ($task->assigned_staff_id !== null) {
   253→   253→                    Notifier::users(
   254→   254→                        [$task->assigned_staff_id],
   255→   255→                        'READY_FOR_PICKUP',
   256→   256→                        $this->taskLabel($task).' is ready for customer pickup.',
   257→   257→                        $task->id,
   258→   258→                        $task->order_id
   259→   259→                    );
   260→   260→                }
   261→   261→
   262→   262→                Audit::log($me, 'TASK_READY_FOR_PICKUP', 'TASK', $task->id);
   263→   263→
   264→   264→                return redirect()
   265→   265→                    ->route('tasks.show', $task->id)
   266→   266→                    ->with('success', __('td.toast.ready'));
   267→   267→            }
   268→   268→
   269→   269→            case 'deliver': {
   270→   270→                if (! $supervisor || ! Permissions::allows($me, 'production.inspect')) {
   271→   271→                    abort(403);
   272→   272→                }
   273→   273→
   274→   274→                if ($task->status !== 'READY_FOR_PICKUP') {
   275→   275→                    return $this->actionError($task, __('td.actionFailed'));
   276→   276→                }
   277→   277→
   278→   278→                $fromStatus = $task->status;
   279→   279→                $task->forceFill([
   280→   280→                    'status' => 'DELIVERED',
   281→   281→                    'delivered_at' => now(),
   282→   282→                ])->save();
   283→   283→                $this->recordStatusChange($task, $fromStatus, 'DELIVERED', $me, $note !== '' ? $note : null);
   284→   284→
   285→   285→                $message = $this->taskLabel($task).' has been delivered/collected by the customer.';
   286→   286→
   287→   287→                Notifier::management('TASK_DELIVERED', $message, $task->id, $task->order_id);
   288→   288→
   289→   289→                if ($task->assigned_staff_id !== null) {
   290→   290→                    Notifier::users([$task->assigned_staff_id], 'TASK_DELIVERED', $message, $task->id, $task->order_id);
   291→   291→                }
   292→   292→
   293→   293→                Audit::log($me, 'TASK_DELIVERED', 'TASK', $task->id, [
   294→   294→                    'note' => $note !== '' ? $note : null,
   295→   295→                ]);
   296→   296→
   297→   297→                return redirect()
   298→   298→                    ->route('tasks.show', $task->id)
   299→   299→                    ->with('success', __('td.toast.delivered'));
   300→   300→            }
   301→   301→
   302→   302→            case 'reopen': {
   303→   303→                if (! $supervisor) {
   304→   304→                    abort(403);
   305→   305→                }
   306→   306→
   307→   307→                if (! in_array($task->status, ['APPROVED', 'READY_FOR_PICKUP', 'AWAITING_INSPECTION', 'CHANGES_REQUESTED'], true)) {
   308→   308→                    return $this->actionError($task, __('td.actionFailed'));
   309→   309→                }
   310→   310→
   311→   311→                $reason = trim((string) $request->input('reason', ''));
   312→   312→
   313→   313→                if ($reason === '' || mb_strlen($reason) < 3) {
   314→   314→                    throw ValidationException::withMessages([
   315→   315→                        'reason' => __('td.toast.reasonRequiredDesc', [
   316→   316→                            'field' => __('td.dialog.reopenPlaceholder'),
   317→   317→                            'min' => 3,
   318→   318→                        ]),
   319→   319→                    ]);
   320→   320→                }
   321→   321→
   322→   322→                $fromStatus = $task->status;
   323→   323→                $task->forceFill([
   324→   324→                    'status' => 'ONGOING',
   325→   325→                    'completed_at' => null,
   326→   326→                ])->save();
   327→   327→                $this->recordStatusChange($task, $fromStatus, 'ONGOING', $me, $reason);
   328→   328→
   329→   329→                if ($task->assigned_staff_id !== null) {
   330→   330→                    Notifier::users(
   331→   331→                        [$task->assigned_staff_id],
   332→   332→                        'TASK_REOPENED',
   333→   333→                        $this->taskLabel($task).' has been reopened for production.',
   334→   334→                        $task->id,
   335→   335→                        $task->order_id
   336→   336→                    );
   337→   337→                }
   338→   338→
   339→   339→                return redirect()
   340→   340→                    ->route('tasks.show', $task->id)
   341→   341→                    ->with('success', __('td.toast.reopened'));
   342→   342→            }
   343→   343→
   344→   344→            case 'cancel': {
   345→   345→                if (! $supervisor || ! Permissions::allows($me, 'production.cancel')) {
   346→   346→                    abort(403);
   347→   347→                }
   348→   348→
   349→   349→                $reason = trim((string) $request->input('reason', ''));
   350→   350→
   351→   351→                if ($reason === '' || mb_strlen($reason) < 3) {
   352→   352→                    throw ValidationException::withMessages([
   353→   353→                        'reason' => __('td.toast.reasonRequiredDesc', [
   354→   354→                            'field' => __('td.reason.cancel'),
   355→   355→                            'min' => 3,
   356→   356→                        ]),
   357→   357→                    ]);
   358→   358→                }
   359→   359→
   360→   360→                $fromStatus = $task->status;
   361→   361→                $task->forceFill([
   362→   362→                    'status' => 'CANCELLED',
   363→   363→                    'cancel_reason' => $reason,
   364→   364→                ])->save();
   365→   365→                $this->recordStatusChange($task, $fromStatus, 'CANCELLED', $me, $reason);
   366→   366→
   367→   367→                if ($task->assigned_staff_id !== null) {
   368→   368→                    Notifier::users(
   369→   369→                        [$task->assigned_staff_id],
   370→   370→                        'TASK_CANCELLED',
   371→   371→                        $this->taskLabel($task).' has been cancelled. Reason: '.$reason,
   372→   372→                        $task->id,
   373→   373→                        $task->order_id
   374→   374→                    );
   375→   375→                }
   376→   376→
   377→   377→                Audit::log($me, 'TASK_CANCELLED', 'TASK', $task->id, ['reason' => $reason]);
   378→   378→
   379→   379→                return redirect()
   380→   380→                    ->route('tasks.show', $task->id)
   381→   381→                    ->with('success', __('td.toast.cancelled'));
   382→   382→            }
   383→   383→        }
   384→   384→
   385→   385→        return $this->actionError($task, __('td.actionFailed'));
   386→   386→    }
   387→   387→
   388→   388→    /**
   389→   389→     * POST /tasks/{id}/progress — manual numerical progress update
   390→   390→     * (ported from tasks/[id]/progress/route.ts). 100% never auto-approves:
   391→   391→     * inspection submission stays an explicit action. The first positive
   392→   392→     * value on an ASSIGNED task moves it to ONGOING.
   393→   393→     */
   394→   394→    public function progress(Request $request, string $id): RedirectResponse
   395→   395→    {
   396→   396→        $me = $request->user();
   397→   397→        $task = $this->loadTask($id);
   398→   398→        $this->authorize($me, $task);
   399→   399→
   400→   400→        $supervisor = $this->isSupervisor($me);
   401→   401→
   402→   402→        if ($task->isTerminal()) {
   403→   403→            return $this->closedTaskError($task);
   404→   404→        }
   405→   405→
   406→   406→        if (! $supervisor && ! in_array($task->status, self::IN_WORK_STATUSES, true)) {
   407→   407→            return $this->actionError($task, __('td.toast.progressFailed'));
   408→   408→        }
   409→   409→
   410→   410→        // View contract names the field "value"; fall back to the
   411→   411→        // reference implementation's "progress" name just in case.
   412→   412→        if ($request->missing('value') && $request->has('progress')) {
   413→   413→            $request->merge(['value' => $request->input('progress')]);
   414→   414→        }
   415→   415→
   416→   416→        $data = $request->validate([
   417→   417→            'value' => ['required', 'integer', 'min:0', 'max:100'],
   418→   418→            'note' => ['nullable', 'string', 'max:500'],
   419→   419→        ]);
   420→   420→
   421→   421→        $value = (int) $data['value'];
   422→   422→        $previous = (int) $task->progress;
   423→   423→
   424→   424→        if ($value === $previous) {
   425→   425→            return $this->actionError($task, __('td.toast.progressFailed'));
   426→   426→        }
   427→   427→
   428→   428→        $note = trim((string) ($data['note'] ?? ''));
   429→   429→        $fromStatus = $task->status;
   430→   430→        $newStatus = $fromStatus === 'ASSIGNED' && $value > 0 ? 'ONGOING' : $fromStatus;
   431→   431→
   432→   432→        DB::transaction(function () use ($task, $me, $value, $previous, $newStatus, $note): void {
   433→   433→            $task->forceFill([
   434→   434→                'progress' => $value,
   435→   435→                'status' => $newStatus,
   436→   436→            ])->save();
   437→   437→
   438→   438→            TaskProgressHistory::query()->create([
   439→   439→                'task_id' => $task->id,
   440→   440→                'user_id' => $me->id,
   441→   441→                'user_name' => $me->name,
   442→   442→                'previous_progress' => $previous,
   443→   443→                'new_progress' => $value,
   444→   444→                'source' => 'MANUAL',
   445→   445→                'note' => $note !== '' ? $note : null,
   446→   446→            ]);
   447→   447→        });
   448→   448→
   449→   449→        if ($newStatus !== $fromStatus) {
   450→   450→            $this->recordStatusChange($task, $fromStatus, $newStatus, $me, 'Production started');
   451→   451→        }
   452→   452→
   453→   453→        Audit::log($me, 'PROGRESS_UPDATED', 'TASK', $task->id, [
   454→   454→            'from' => $previous,
   455→   455→            'to' => $value,
   456→   456→            'note' => $note !== '' ? $note : null,
   457→   457→            'correction' => $supervisor,
   458→   458→        ]);
   459→   459→
   460→   460→        return redirect()
   461→   461→            ->route('tasks.show', $task->id)
   462→   462→            ->with('success', __('td.toast.progressUpdated', ['value' => $value]));
   463→   463→    }

   465→   465→    /**

   928→   928→        }
   929→   929→
   930→   930→        $instructionsInput = trim((string) ($data['instructions'] ?? ''));
   931→   931→
   932→   932→        if ($instructionsInput !== '' && $instructionsInput !== (string) $task->instructions) {
   933→   933→            $update['instructions'] = $instructionsInput;
   934→   934→            $changes['instructions'] = 'updated';
   935→   935→            $instructionsChanged = true;
   936→   936→        }
   937→   937→
   938→   938→        $garmentPieceInput = trim((string) ($data['garmentPiece'] ?? ''));
   939→   939→
   940→   940→        if ($garmentPieceInput !== '' && $garmentPieceInput !== (string) $task->garment_piece) {
   941→   941→            $clash = ProductionTask::query()
   942→   942→                ->where('order_id', $task->order_id)
   943→   943→                ->where('garment_piece', $garmentPieceInput)
   944→   944→                ->where('status', '!=', 'CANCELLED')
   945→   945→                ->where('id', '!=', $task->id)
   946→   946→                ->exists();
   947→   947→
   948→   948→            if ($clash) {
   949→   949→                throw ValidationException::withMessages([
   950→   950→                    'garmentPiece' => 'Piece "'.$garmentPieceInput.'" already exists on this order.',
   951→   951→                ]);
   952→   952→            }
   953→   953→
   954→   954→            $update['garment_piece'] = $garmentPieceInput;
   955→   955→        }
   956→   956→
   957→   957→        if ($update === []) {
   958→   958→            return $this->actionError($task, __('td.toast.nothingToUpdate'));
   959→   959→        }
   960→   960→
   961→   961→        $task->forceFill($update)->save();
   962→   962→
   963→   963→        if ($deadlineChanged && $task->assigned_staff_id !== null) {
   964→   964→            Notifier::users(
   965→   965→                [$task->assigned_staff_id],
   966→   966→                'DEADLINE_CHANGED',
   967→   967→                'The deadline of '.$this->taskLabel($task).' has been changed.',
   968→   968→                $task->id,
   969→   969→                $task->order_id
   970→   970→            );
   971→   971→        }
   972→   972→
   973→   973→        if ($instructionsChanged && $task->assigned_staff_id !== null) {
   974→   974→            Notifier::users(
   975→   975→                [$task->assigned_staff_id],
   976→   976→                'NEW_INSTRUCTIONS',
   977→   977→                'New production instructions were added to '.$this->taskLabel($task).'.',
   978→   978→                $task->id,
   979→   979→                $task->order_id
   980→   980→            );
   981→   981→        }
   982→   982→
   983→   983→        Audit::log($me, 'TASK_UPDATED', 'TASK', $task->id, $changes);
   984→   984→
   985→   985→        return redirect()
   986→   986→            ->route('tasks.show', $task->id)
   987→   987→            ->with('success', __('td.toast.detailsUpdated'));
   988→   988→    }

  1053→  1053→    /**
  1054→  1054→     * Whether an assignee may submit the task for inspection: manual
  1055→  1055→     * progress reached 100% OR every applicable stage is COMPLETED (view
  1056→  1056→     * contract; supervisors bypass this check like in route.ts).
  1057→  1057→     */
  1058→  1058→    private function readyForInspection(ProductionTask $task): bool
  1059→  1059→    {
  1060→  1060→        if ((int) $task->progress >= 100) {
  1061→  1061→            return true;
  1062→  1062→        }
  1063→  1063→
  1064→  1064→        $stages = ProductionStage::query()
  1065→  1065→            ->where('is_active', true)
  1066→  1066→            ->orderBy('sort_order')
  1067→  1067→            ->get()
  1068→  1068→            ->filter(
  1069→  1069→                static fn (ProductionStage $stage): bool => Atelier::stageAppliesTo(
  1070→  1070→                    (string) $stage->applies_to,
  1071→  1071→                    (string) $task->category
  1072→  1072→                )
  1073→  1073→            );
  1074→  1074→
  1075→  1075→        if ($stages->isEmpty()) {
  1076→  1076→            return false;
  1077→  1077→        }
  1078→  1078→
  1079→  1079→        $states = TaskStageState::query()
  1080→  1080→            ->where('task_id', $task->id)
  1081→  1081→            ->get()
  1082→  1082→            ->keyBy('stage_id');
  1083→  1083→
  1084→  1084→        foreach ($stages as $stage) {
  1085→  1085→            $state = $states->has($stage->id)
  1086→  1086→                ? (string) $states->get($stage->id)->state
  1087→  1087→                : 'NOT_STARTED';
  1088→  1088→
  1089→  1089→            if ($state !== 'COMPLETED') {
  1090→  1090→                return false;
  1091→  1091→            }
  1092→  1092→        }
  1093→  1093→
  1094→  1094→        return true;
  1095→  1095→    }
