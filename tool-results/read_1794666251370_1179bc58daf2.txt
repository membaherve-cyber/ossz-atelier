    16→function hashPassword(password: string): string {
    17→  const salt = randomBytes(16).toString("hex");
    18→  const hash = scryptSync(password, salt, 64).toString("hex");
    19→  return `${salt}:${hash}`;
    20→}

   306→    };

   308→    const users: Record<string, string> = {
   309→      hamidou: hamidou.name,

   311→      aminata: aminata.name,
   312→      david: david.name,
   313→    };
   314→    for (const [code, name] of stageDefs) names[code] = name;
   315→    for (const [stageCode, state] of opts.stageStates) {
   316→      if (!applies(stageCode)) continue;
   317→      await db.taskStageState.create({
   318→        data: {
   319→          taskId: task.id,
   320→          stageId: stages[stageCode],
   321→          state,
   322→          updatedById: opts.assignedStaffId ?? null,
   323→          updatedByName: opts.assignedStaffId === hamidou.id ? hamidou.name : lawal.name,
   324→          updatedAt: opts.createdAt,
   325→        },
   326→      });
   327→    }
   328→    if (opts.stageUpdates) {
   329→      for (const u of opts.stageUpdates) {
   330→        if (!applies(u.stageCode)) continue;
   331→        await db.taskStageUpdate.create({
   332→          data: {
   333→            taskId: task.id,
   334→            stageId: stages[u.stageCode],
   335→            stageName: names[u.stageCode],
   336→            userId: u.by === "hamidou" ? hamidou.id : u.by === "lawal" ? lawal.id : aminata.id,
   337→            userName: users[u.by],
   338→            previousState: u.from,
   339→            newState: u.to,
   340→            createdAt: u.at,
   341→          },
   342→        });
   343→      }
   344→    }
   345→    if (opts.statusChain) {
   346→      let prev = "UNASSIGNED";
   347→      for (const c of opts.statusChain) {
   348→        await db.taskStatusChange.create({
   349→          data: {
   350→            taskId: task.id,
   351→            userId: c.by === "aminata" ? aminata.id : c.by === "david" ? david.id : c.by === "hamidou" ? hamidou.id : lawal.id,
   352→            userName: users[c.by] ?? aminata.name,
   353→            fromStatus: prev,
   354→            toStatus: c.to,
   355→            reason: c.reason ?? null,
   356→            createdAt: c.at,
   357→          },
   358→        });
   359→        prev = c.to;
   360→      }
   361→    }
   362→    if (opts.progressHistory) {
   363→      for (const p of opts.progressHistory) {
   364→        await db.taskProgressHistory.create({
   365→          data: {
   366→            taskId: task.id,
   367→            userId: p.by === "hamidou" ? hamidou.id : lawal.id,
   368→            userName: users[p.by],
   369→            previousProgress: p.from,
   370→            newProgress: p.to,
   371→            source: "MANUAL",
   372→            note: p.note ?? null,
   373→            createdAt: p.at,
   374→          },
   375→        });
   376→      }
   377→    }
   378→    if (opts.assignedStaffId) {
   379→      await db.taskAssignment.create({
   380→        data: {
   381→          taskId: task.id,
   382→          staffId: opts.assignedStaffId,
   383→          staffName:
   384→            opts.assignedStaffId === hamidou.id ? hamidou.name : opts.assignedStaffId === lawal.id ? lawal.name : "",
   385→          assignedById: aminata.id,
   386→          assignedByName: aminata.name,
   387→          type: "ASSIGN",
   388→          createdAt: opts.createdAt,
   389→        },
   390→      });
   391→    }
   392→    return task;
   393→  }
