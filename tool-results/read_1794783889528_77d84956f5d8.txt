   127→const WORK_STATUSES = ["ASSIGNED", "ONGOING", "CHANGES_REQUESTED"];

   129→/** Selectable checklist stage states, localized on demand. */
   130→function stageStateOptions(lang: Lang): { value: StageState; label: string }[] {
   131→  return STAGE_STATES.map((s) => ({
   132→    value: s,
   133→    label: s === "NOT_APPLICABLE" ? "N/A" : stageStateLabel(s, lang),
   134→  }));
   135→}
   136→
   137→/** Display label for a progress-history source (falls back to the raw value). */
   138→function progressSourceLabel(source: string, t: TranslateFn): string {
   139→  const known: Record<string, string> = {
   140→    MANUAL: t("td.source.MANUAL"),
   141→    STAGE: t("td.source.STAGE"),
   142→    INSPECTION: t("td.source.INSPECTION"),
   143→  };
   144→  return known[source] ?? source;
   145→}
   146→
   147→/** Convert an ISO timestamp to a `datetime-local` input value (local time). */
   148→function toDatetimeLocal(iso: string | null): string {
   149→  if (!iso) return "";
   150→  const d = new Date(iso);
   151→  if (Number.isNaN(d.getTime())) return "";
   152→  return format(d, "yyyy-MM-dd'T'HH:mm");
   153→}

   155→function MetaRow({ label, children }: { label: string; children: ReactNode }) {

   160→      </span>
   161→      <span className="text-sm text-right min-w-0">{children}</span>
   162→    </div>
   163→  );
   164→}

   591→      toast({
   592→        title: t("td.toast.nothingToUpdate"),
   593→        description: t("td.toast.nothingToUpdateDesc"),

   602→      api.patch(url, { action: "SUBMIT", note: note.trim() || undefined })
