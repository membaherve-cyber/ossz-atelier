    33→    33→ * the assignee or a supervisor (MANAGEMENT/ADMIN); actions gated by a
    34→    34→ * permission in the reference re-check it here because the routes only
    35→    35→ * carry `auth`.
    36→    36→ *
    37→    37→ * Status transitions (EXACT reference rules):
    38→    38→ * - start:    ASSIGNED → ONGOING (assignee or supervisor)
    39→    39→ * - submit:   ASSIGNED|ONGOING|CHANGES_REQUESTED → AWAITING_INSPECTION

    41→    41→ * - inspect:  APPROVED|CHANGES_REQUESTED only from AWAITING_INSPECTION;
    42→    42→ *             REOPENED only from APPROVED|READY_FOR_PICKUP
    43→    43→ * - ready:    APPROVED → READY_FOR_PICKUP (supervisor only)
    44→    44→ * - deliver:  READY_FOR_PICKUP → DELIVERED (supervisor only)
    45→    45→ * - reopen:   APPROVED|READY_FOR_PICKUP → ONGOING (supervisor only)
    46→    46→ * - cancel:   any non-terminal status → CANCELLED (supervisor only, reason)
    47→    47→ * - block:    ASSIGNED|ONGOING|CHANGES_REQUESTED → BLOCKED (reason)
    48→    48→ * - unblock:  BLOCKED → ONGOING (progress > 0) or ASSIGNED
    49→    49→ */

    57→    57→     * POST /tasks/{id}/assign — assign or reassign the task.
    58→    58→     */
    59→    59→    public function assign(Request $request, string $id): RedirectResponse
    60→    60→    {
    61→    61→        $data = $request->validate([
    62→    62→            'staffId' => ['required', 'string'],
    63→    63→            'note' => ['nullable', 'string', 'max:1000'],
    64→    64→        ]);
    65→    65→
    66→    66→        /** @var User $user */
    67→    67→        $user = $request->user();
    68→    68→        $task = $this->taskOrFail($id);
    69→    69→
    70→    70→        if (! $this->canAccess($user, $task)) {
    71→    71→            abort(403);
    72→    72→        }
    73→    73→
    74→    74→        if ($task->isTerminal()) {
    75→    75→            return $this->closedTask($id);
    76→    76→        }
    77→    77→
    78→    78→        $staff = User::query()->find($data['staffId']);
    79→    79→
    80→    80→        if ($staff === null || ! $staff->is_active) {
    81→    81→            return $this->taskError($id, __('flash.invalidStaffAccount'));
    82→    82→        }
    83→    83→
    84→    84→        if ($staff->id === $task->assigned_staff_id) {
    85→    85→            return $this->taskError($id, __('flash.taskAlreadyAssigned'));
    86→    86→        }
    87→    87→
    88→    88→        $reassign = $task->assigned_staff_id !== null;
    89→    89→        $oldAssigneeId = $task->assigned_staff_id;
    90→    90→        $newStatus = $task->status === 'UNASSIGNED' ? 'ASSIGNED' : $task->status;
    91→    91→        $note = $this->text($data, 'note');
    92→    92→
    93→    93→        DB::transaction(function () use ($task, $staff, $user, $reassign, $newStatus, $note): void {
    94→    94→            $task->forceFill([
    95→    95→                'assigned_staff_id' => $staff->id,
    96→    96→                'status' => $newStatus,
    97→    97→            ])->save();
    98→    98→
    99→    99→            TaskAssignment::query()->create([
   100→   100→                'task_id' => $task->id,
   101→   101→                'staff_id' => $staff->id,
   102→   102→                'staff_name' => $staff->name,
   103→   103→                'assigned_by_id' => $user->id,
   104→   104→                'assigned_by_name' => $user->name,
   105→   105→                'type' => $reassign ? 'REASSIGN' : 'ASSIGN',
   106→   106→                'note' => $note,
   107→   107→            ]);
   108→   108→        });
   109→   109→
   110→   110→        if ($newStatus !== $task->status) {
   111→   111→            $this->recordStatusChange($task, $task->status, $newStatus, $note, $user);
   112→   112→        }
   113→   113→
   114→   114→        $label = $this->taskLabel($task);
   115→   115→
   116→   116→        Notifier::users(
   117→   117→            [$staff->id],
   118→   118→            $reassign ? 'TASK_REASSIGNED' : 'TASK_ASSIGNED',
   119→   119→            $label.' has been '.($reassign ? 'reassigned' : 'assigned').' to you.'
   120→   120→            .($note !== null ? ' Note: '.$note : ''),
   121→   121→            $task->id,
   122→   122→            $task->order_id
   123→   123→        );
   124→   124→
   125→   125→        // Spec §5: reassignment notifies BOTH the old and the new assignee.
   126→   126→        if ($reassign && $oldAssigneeId !== null) {
   127→   127→            Notifier::users(
   128→   128→                [$oldAssigneeId],
   129→   129→                'TASK_REASSIGNED',
   130→   130→                $label.' has been reassigned to '.$staff->name.'.',
   131→   131→                $task->id,
   132→   132→                $task->order_id
   133→   133→            );
   134→   134→        }
   135→   135→
   136→   136→        Audit::log($user, $reassign ? 'TASK_REASSIGNED' : 'TASK_ASSIGNED', 'TASK', $task->id, [
   137→   137→            'staff' => $staff->name,
   138→   138→            'note' => $note,
   139→   139→        ]);
   140→   140→
   141→   141→        return $this->taskSuccess(
   142→   142→            $id,
   143→   143→            __($reassign ? 'td.toast.reassignedTo' : 'td.toast.assignedTo', ['name' => $staff->name])
   144→   144→        );
   145→   145→    }

   154→   154→    public function status(Request $request, string $id): RedirectResponse

   157→   157→            'action' => ['required', 'string', 'in:start,submit,ready,deliver,reopen,cancel'],
   158→   158→            'reason' => ['nullable', 'string', 'max:2000'],
   159→   159→            'note' => ['nullable', 'string', 'max:1000'],
   160→   160→            'comments' => ['nullable', 'string', 'max:2000'],
   161→   161→        ]);
   162→   162→
   163→   163→        /** @var User $user */
   164→   164→        $user = $request->user();
   165→   165→        $task = $this->taskOrFail($id);
   166→   166→        $supervisor = $this->supervisor($user);
   167→   167→
   168→   168→        if (! $this->canAccess($user, $task)) {
   169→   169→            abort(403);
   170→   170→        }
   171→   171→
   172→   172→        if ($task->isTerminal()) {
   173→   173→            return $this->closedTask($id);
   174→   174→        }
   175→   175→
   176→   176→        return match ($data['action']) {
   177→   177→            'start' => $this->startTask($task, $user),

   182→   182→            'cancel' => $this->cancelTask($task, $user, $supervisor, $this->text($data, 'reason')),
   183→   183→            default => $this->taskError($id, __('td.actionFailed')),
   184→   184→        };
   185→   185→    }

   187→   187→    /**
   188→   188→     * action=start — ASSIGNED → ONGOING (assignee or supervisor).
   189→   189→     */
   190→   190→    private function startTask(ProductionTask $task, User $user): RedirectResponse
   191→   191→    {
   192→   192→        if ($task->status !== 'ASSIGNED') {
   193→   193→            return $this->taskError($task->id, __('flash.statusNoStart'));
   194→   194→        }

   202→   202→    /**
   203→   203→     * action=submit — ASSIGNED|ONGOING|CHANGES_REQUESTED → AWAITING_INSPECTION.
   204→   204→     * Non-supervisors must have reached 100% progress.
   205→   205→     */
   206→   206→    private function submitTask(ProductionTask $task, User $user, ?string $note): RedirectResponse
   207→   207→    {
   208→   208→        $supervisor = $this->supervisor($user);
   209→   209→
   210→   210→        if (! in_array($task->status, ['ASSIGNED', 'ONGOING', 'CHANGES_REQUESTED'], true)) {
   211→   211→            return $this->taskError(
   212→   212→                $task->id,
   213→   213→                __('flash.statusNoSubmit', ['status' => __('labels.status.'.$task->status)])
   214→   214→            );
   215→   215→        }
   216→   216→
   217→   217→        if (! $supervisor && $task->progress < 100) {
   218→   218→            return $this->taskError($task->id, __('td.toast.reach100'));
   219→   219→        }
   220→   220→
   221→   221→        $task->forceFill(['status' => 'AWAITING_INSPECTION'])->save();
   222→   222→        $this->recordStatusChange($task, $task->status, 'AWAITING_INSPECTION', $note, $user);
   223→   223→
   224→   224→        if ($note !== null) {
   225→   225→            TaskComment::query()->create([
   226→   226→                'task_id' => $task->id,
   227→   227→                'user_id' => $user->id,
   228→   228→                'user_name' => $user->name,
   229→   229→                'message' => 'Submitted for inspection: '.$note,
   230→   230→            ]);
   231→   231→        }
   232→   232→
   233→   233→        Notifier::management(
   234→   234→            'SUBMITTED_FOR_INSPECTION',
   235→   235→            $user->name.' submitted '.$this->taskLabel($task).' for inspection.',
   236→   236→            $task->id,
   237→   237→            $task->order_id
   238→   238→        );
   239→   239→
   240→   240→        Audit::log($user, 'SUBMITTED_FOR_INSPECTION', 'TASK', $task->id, ['note' => $note]);
   241→   241→
   242→   242→        return $this->taskSuccess($task->id, __('td.toast.submitted'));
   243→   243→    }

   420→   420→    /**
   421→   421→     * POST /tasks/{id}/progress — numerical progress update (append-only
   422→   422→     * history; 100% never auto-approves — inspection is explicit).
   423→   423→     */
   424→   424→    public function progress(Request $request, string $id): RedirectResponse
   425→   425→    {
   426→   426→        $data = $request->validate([
   427→   427→            'progress' => ['required', 'integer', 'min:0', 'max:100'],
   428→   428→            'note' => ['nullable', 'string', 'max:500'],
   429→   429→        ]);
   430→   430→
   431→   431→        /** @var User $user */
   432→   432→        $user = $request->user();
   433→   433→        $task = $this->taskOrFail($id);
   434→   434→        $supervisor = $this->supervisor($user);
   435→   435→
   436→   436→        if (! $this->canAccess($user, $task)) {
   437→   437→            abort(403);
   438→   438→        }
   439→   439→
   440→   440→        if ($task->isTerminal()) {
   441→   441→            return $this->closedTask($id);
   442→   442→        }
   443→   443→
   444→   444→        if (! $supervisor && $task->status === 'BLOCKED') {
   445→   445→            return $this->taskError($id, __('flash.resolveBlockerFirst'));
   446→   446→        }
   447→   447→
   448→   448→        if (! $supervisor && ! in_array($task->status, ['ASSIGNED', 'ONGOING', 'CHANGES_REQUESTED'], true)) {
   449→   449→            return $this->taskError(
   450→   450→                $id,
   451→   451→                __('flash.progressLocked', ['status' => __('labels.status.'.$task->status)])
   452→   452→            );
   453→   453→        }
   454→   454→
   455→   455→        $progress = (int) $data['progress'];
   456→   456→
   457→   457→        if ($progress === $task->progress) {
   458→   458→            return $this->taskError($id, __('flash.progressUnchanged', ['value' => $task->progress]));
   459→   459→        }
   460→   460→
   461→   461→        $note = $this->text($data, 'note');
   462→   462→        $newStatus = $task->status === 'ASSIGNED' && $progress > 0 ? 'ONGOING' : $task->status;
   463→   463→
   464→   464→        DB::transaction(function () use ($task, $user, $progress, $note, $newStatus): void {
   465→   465→            $task->forceFill([

  1018→  1018→
  1019→  1019→        Audit::log($user, 'TASK_UPDATED', 'TASK', $task->id, $changes);
  1020→  1020→
