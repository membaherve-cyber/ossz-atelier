    30→export function monthNow(): string {
    31→  const now = new Date();
    32→  return `${now.getUTCFullYear()}-${String(now.getUTCMonth() + 1).padStart(2, "0")}`;
    33→}
    34→
    35→export function yearNow(): string {
    36→  return String(new Date().getUTCFullYear());
    37→}
    38→
    39→function quarterNow(): string {
    40→  const now = new Date();
    41→  return `${now.getUTCFullYear()}-Q${Math.floor(now.getUTCMonth() / 3) + 1}`;
    42→}
    43→
    44→function shiftMonth(value: string, delta: number): string {
    45→  const m = /^(\d{4})-(\d{2})$/.exec(value);
    46→  if (!m) return value;
    47→  const d = new Date(Date.UTC(Number(m[1]), Number(m[2]) - 1 + delta, 1));
    48→  return `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, "0")}`;
    49→}
    50→
    51→function shiftQuarter(value: string, delta: number): string {
    52→  const m = /^(\d{4})-Q([1-4])$/.exec(value);
    53→  if (!m) return value;
    54→  const total = Number(m[1]) * 4 + (Number(m[2]) - 1) + delta;
    55→  const year = Math.floor(total / 4);
    56→  return `${year}-Q${total - year * 4 + 1}`;
    57→}
    58→
    59→function shiftYear(value: string, delta: number): string {
    60→  const m = /^\d{4}$/.exec(value);
    61→  if (!m) return value;
    62→  return String(Number(m[1]) + delta);
    63→}

    69→export function scoreTone(value: number): string {
    70→  if (value >= 75) return "text-emerald-700";
    71→  if (value >= 50) return "text-amber-700";
    72→  return "text-red-700";
    73→}

    75→function csvCell(value: string | number | null | undefined): string {
    76→  const s = value == null ? "" : String(value);
    77→  return /[",\n]/.test(s) ? `"${s.replaceAll('"', '""')}"` : s;
    78→}

    80→export function downloadCsv(
    81→  filename: string,
    82→  rows: Array<Array<string | number | null | undefined>>
    83→): void {
    84→  const csv = rows.map((r) => r.map(csvCell).join(",")).join("\n");
    85→  const blob = new Blob([`\uFEFF${csv}`], { type: "text/csv;charset=utf-8" });
    86→  const url = URL.createObjectURL(blob);
    87→  const link = document.createElement("a");
    88→  link.href = url;
    89→  link.download = filename;
    90→  document.body.appendChild(link);
    91→  link.click();
    92→  link.remove();
    93→  URL.revokeObjectURL(url);
    94→}
