    32→    32→export function monthNow(): string {
    33→    33→  const now = new Date();
    34→    34→  return `${now.getUTCFullYear()}-${String(now.getUTCMonth() + 1).padStart(2, "0")}`;
    35→    35→}
    36→    36→
    37→    37→export function yearNow(): string {
    38→    38→  return String(new Date().getUTCFullYear());
    39→    39→}
    40→    40→
    41→    41→function quarterNow(): string {
    42→    42→  const now = new Date();
    43→    43→  return `${now.getUTCFullYear()}-Q${Math.floor(now.getUTCMonth() / 3) + 1}`;
    44→    44→}
    45→    45→
    46→    46→function shiftMonth(value: string, delta: number): string {
    47→    47→  const m = /^(\d{4})-(\d{2})$/.exec(value);
    48→    48→  if (!m) return value;
    49→    49→  const d = new Date(Date.UTC(Number(m[1]), Number(m[2]) - 1 + delta, 1));
    50→    50→  return `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, "0")}`;
    51→    51→}
    52→    52→
    53→    53→function shiftQuarter(value: string, delta: number): string {
    54→    54→  const m = /^(\d{4})-Q([1-4])$/.exec(value);
    55→    55→  if (!m) return value;
    56→    56→  const total = Number(m[1]) * 4 + (Number(m[2]) - 1) + delta;
    57→    57→  const year = Math.floor(total / 4);
    58→    58→  return `${year}-Q${total - year * 4 + 1}`;
    59→    59→}
    60→    60→
    61→    61→function shiftYear(value: string, delta: number): string {
    62→    62→  const m = /^\d{4}$/.exec(value);
    63→    63→  if (!m) return value;
    64→    64→  return String(Number(m[1]) + delta);
    65→    65→}

    77→    77→function csvCell(value: string | number | null | undefined): string {
    78→    78→  const s = value == null ? "" : String(value);
    79→    79→  return /[",\n]/.test(s) ? `"${s.replaceAll('"', '""')}"` : s;
    80→    80→}

    82→    82→export function downloadCsv(
    83→    83→  filename: string,
    84→    84→  rows: Array<Array<string | number | null | undefined>>
    85→    85→): void {
    86→    86→  const csv = rows.map((r) => r.map(csvCell).join(",")).join("\n");
    87→    87→  const blob = new Blob([`\uFEFF${csv}`], { type: "text/csv;charset=utf-8" });
    88→    88→  const url = URL.createObjectURL(blob);
    89→    89→  const link = document.createElement("a");
    90→    90→  link.href = url;
    91→    91→  link.download = filename;
    92→    92→  document.body.appendChild(link);
    93→    93→  link.click();
    94→    94→  link.remove();
    95→    95→  URL.revokeObjectURL(url);
    96→    96→}

    99→    99→export function periodLabel(
   100→   100→  t: TranslateFn,
   101→   101→  lang: Lang,
   102→   102→  range: { type: string; value: string; label: string }
   103→   103→): string {
   104→   104→  if (range.type === "MONTH") {
   105→   105→    const m = /^(\d{4})-(\d{2})$/.exec(range.value);
   106→   106→    if (m) {
   107→   107→      const month = new Intl.DateTimeFormat(lang === "fr" ? "fr-FR" : "en-US", {
   108→   108→        month: "long",
   109→   109→        timeZone: "UTC",
   110→   110→      }).format(new Date(Date.UTC(Number(m[1]), Number(m[2]) - 1, 1)));
   111→   111→      return t("perf.rangeMonth", { month, year: m[1] });
   112→   112→    }
   113→   113→  }
   114→   114→  if (range.type === "QUARTER") {
   115→   115→    const q = /^(\d{4})-Q([1-4])$/.exec(range.value.toUpperCase());
   116→   116→    if (q) return t("perf.rangeQuarter", { quarter: q[2], year: q[1] });
   117→   117→  }
   118→   118→  if (range.type === "YEAR" && /^\d{4}$/.test(range.value)) {
   119→   119→    return t("perf.rangeYear", { year: range.value });
   120→   120→  }
   121→   121→  return range.label;
   122→   122→}
